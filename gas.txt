function buildHeaderIndexMap_(headers) {
  const map = {};
  headers.forEach((h, i) => map[h] = i);
  return map;
}
function headerRow_() { return CONFIG.HEADER_ROW; }
function dataStartRow_() { return CONFIG.DATA_START_ROW; }
function getSheetDataRowCount_(sh) { return Math.max(0, sh.getLastRow() - headerRow_()); }
function getSafeAppendStartRow_(sh) { return Math.max(dataStartRow_(), sh.getLastRow() + 1); }
function getHeaderValues_(sh, headersLength) { return sh.getRange(headerRow_(), 1, 1, headersLength).getValues()[0]; }
function headerMatches_(actual, expected) {
  if (!actual || actual.length < expected.length) return false;
  for (let i = 0; i < expected.length; i++) if (String(actual[i] || "") !== String(expected[i])) return false;
  return true;
}
function ensureTopRowsAndHeader_(sh, headers) {
  if (sh.getMaxColumns() < headers.length) sh.insertColumnsAfter(sh.getMaxColumns(), headers.length - sh.getMaxColumns());
  if (sh.getMaxRows() < headerRow_()) sh.insertRowsAfter(sh.getMaxRows(), headerRow_() - sh.getMaxRows());
  const current = getHeaderValues_(sh, headers.length);
  if (!headerMatches_(current, headers)) sh.getRange(headerRow_(), 1, 1, headers.length).setValues([headers]);
  sh.setFrozenRows(headerRow_());
}
function getOrCreateSheet_(spreadsheetId, name, headers) {
  const ss = SpreadsheetApp.openById(spreadsheetId);
  let sh = ss.getSheetByName(name);
  if (!sh) sh = ss.insertSheet(name);
  ensureTopRowsAndHeader_(sh, headers);
  return sh;
}
function clearSheetDataKeepHeader_(sh, headers) {
  const lastRow     = sh.getLastRow();
  const maxRows     = sh.getMaxRows();
  const numToDelete = lastRow - headerRow_();

  if (numToDelete > 0) {
    // 削除後に非固定行がゼロになる場合は deleteRows できないため
    // コンテンツのみクリアする（行自体は残す）
    if (numToDelete >= maxRows - headerRow_()) {
      sh.getRange(dataStartRow_(), 1, numToDelete, Math.max(1, sh.getMaxColumns())).clearContent();
    } else {
      sh.deleteRows(dataStartRow_(), numToDelete);
    }
  }
  ensureTopRowsAndHeader_(sh, headers);
}

function migrateCurrentSchemaToMidtermTracking_() {
  const app = getAppConfig_();
  const ss = SpreadsheetApp.openById(app.spreadsheetId);
  let rawSh = ss.getSheetByName(CONFIG.RAW_SHEET);
  if (!rawSh) {
    rawSh = ss.insertSheet(CONFIG.RAW_SHEET);
    ensureTopRowsAndHeader_(rawSh, RAW_HEADERS);
    return { ok: true, migrated: false, reason: "raw created new" };
  }
  const reportSh = getOrCreateSheet_(app.spreadsheetId, CONFIG.REPORT_SHEET_NAME, REPORT_HEADERS);
  const holidaySh = getOrCreateSheet_(app.spreadsheetId, CONFIG.HOLIDAY_SHEET_NAME, ["date", "name", "source"]);
  clearSheetDataKeepHeader_(reportSh, REPORT_HEADERS);
  ensureTopRowsAndHeader_(holidaySh, ["date", "name", "source"]);
  const lastRow = rawSh.getLastRow();
  if (lastRow < dataStartRow_()) {
    ensureTopRowsAndHeader_(rawSh, RAW_HEADERS);
    return { ok: true, migrated: false, reason: "raw empty" };
  }
  const headerLegacy = getHeaderValues_(rawSh, LEGACY_RAW_HEADERS.length);
  const headerNew = getHeaderValues_(rawSh, RAW_HEADERS.length);
  const isLegacy = headerMatches_(headerLegacy, LEGACY_RAW_HEADERS);
  const isNew = headerMatches_(headerNew, RAW_HEADERS);
  if (isNew) {
    ensureTopRowsAndHeader_(rawSh, RAW_HEADERS);
    return { ok: true, migrated: false, reason: "already new schema" };
  }
  if (!isLegacy) {
    ensureTopRowsAndHeader_(rawSh, RAW_HEADERS);
    return { ok: true, migrated: false, reason: "unknown header, only rewrote header row" };
  }
  const dataRows = getSheetDataRowCount_(rawSh);
  const oldValues = rawSh.getRange(dataStartRow_(), 1, dataRows, LEGACY_RAW_HEADERS.length).getValues();
  const newRows = oldValues.map((r) => {
    const signalType = String(r[4] || "").trim().toUpperCase();
    const signalDate = r[2];
    const row = [
      r[0], r[1], r[2], r[3], signalType, r[5], r[6], r[7], r[8], r[9], r[10],
      r[11] instanceof Date ? r[11] : (signalDate instanceof Date ? addBusinessDaysJst_(signalDate, 5) : ""),
      r[12], r[13], r[14], (r[18] === true || String(r[18] || "").toUpperCase() === "TRUE"),
      signalDate instanceof Date ? addBusinessDaysJst_(signalDate, 10) : "", "", "", "", false,
      signalDate instanceof Date ? addBusinessDaysJst_(signalDate, 20) : "", "", "", "", false,
      signalDate instanceof Date ? addBusinessDaysJst_(signalDate, 40) : "", "", "", "", false,
      "", appendTag_(String(r[16] || ""), "MIGRATED_V2"), r[17] || new Date()
    ];
    row[RAW_IDX.status] = computeRowStatusFromArray_(row);
    return row;
  });
  rawSh.clearContents();
  ensureTopRowsAndHeader_(rawSh, RAW_HEADERS);
  if (newRows.length > 0) {
    rawSh.getRange(dataStartRow_(), 1, newRows.length, RAW_HEADERS.length).setValues(newRows);
    applyRawFormats_(rawSh, dataStartRow_(), newRows.length);
  }
  return { ok: true, migrated: true, convertedRows: newRows.length };
}

function doPost(e) {
  let lock = null;
  try {
    debugLogToSheet_("doPost.start", { hasPostData: !!(e && e.postData), contentType: e && e.postData ? e.postData.type : "", bodyLength: e && e.postData && e.postData.contents ? e.postData.contents.length : 0 });
    const app = getAppConfig_();
    const envelope = parseEnvelope_(e);
    debugLogToSheet_("doPost.envelope", { v: envelope.v || "", ts: envelope.ts || "", hasPayloadJson: !!envelope.payloadJson, payloadJsonLength: envelope.payloadJson ? String(envelope.payloadJson).length : 0 });
    verifyEnvelope_(envelope, app.gasSharedSecret);
    debugLogToSheet_("doPost.verify.ok", {});
    const payload = JSON.parse(envelope.payloadJson);
    const parsed = sanitizeAlertsWithDiagnostics_(payload);
    lock = LockService.getScriptLock();
    lock.waitLock(30000);
    const rawSh = getOrCreateSheet_(app.spreadsheetId, CONFIG.RAW_SHEET, RAW_HEADERS);
    const existingIds = getExistingAlertIdSet_(rawSh);
    const seenInBatch = new Set();
    const invalidAlerts = parsed.invalidAlerts;
    const duplicateAlerts = [];
    const alertsToInsert = [];
    let skippedUnsupported = 0;
    parsed.validAlerts.forEach((alert) => {
      if (!looksLikeJpxCode_(alert.symbolCode)) { skippedUnsupported += 1; return; }
      const id = String(alert.alertId || "");
      if (!id) { duplicateAlerts.push({ stage: "dedupe", reason: "empty alertId", alert: alert }); return; }
      if (existingIds.has(id)) { duplicateAlerts.push({ stage: "dedupe", reason: "already exists in sheet", alertId: id, symbolCode: alert.symbolCode, symbolName: alert.symbolName }); return; }
      if (seenInBatch.has(id)) { duplicateAlerts.push({ stage: "dedupe", reason: "duplicated inside batch", alertId: id, symbolCode: alert.symbolCode, symbolName: alert.symbolName }); return; }
      seenInBatch.add(id);
      alertsToInsert.push(alert);
    });
    if (invalidAlerts.length > 0) debugLogToSheet_("doPost.invalidAlerts", invalidAlerts);
    if (duplicateAlerts.length > 0) debugLogToSheet_("doPost.duplicateAlerts", duplicateAlerts);
    const appendResult = appendAlertRowsSafe_(rawSh, alertsToInsert);
    debugLogToSheet_("doPost.final.summary", { received: parsed.receivedCount, valid: parsed.validAlerts.length, invalid: invalidAlerts.length, duplicates: duplicateAlerts.length, skippedUnsupported: skippedUnsupported, insertTarget: alertsToInsert.length, inserted: appendResult.inserted });
    return jsonOut_({ ok: true, received: parsed.receivedCount, valid: parsed.validAlerts.length, invalid: invalidAlerts.length, duplicates: duplicateAlerts.length, skippedUnsupported: skippedUnsupported, insertTarget: alertsToInsert.length, inserted: appendResult.inserted, buildFailed: appendResult.failedRows.filter(x => x.stage === "buildRow").length, writeFailed: appendResult.failedRows.filter(x => x.stage === "writeRow").length });
  } catch (err) {
    debugLogToSheet_("doPost.error", { message: err && err.message ? err.message : String(err), stack: err && err.stack ? String(err.stack) : "" });
    return jsonOut_({ ok: false, error: err && err.message ? err.message : String(err) });
  } finally { if (lock && lock.hasLock()) lock.releaseLock(); }
}

function sanitizeAlertsWithDiagnostics_(body) {
  const sourceAlerts = Array.isArray(body.alerts) ? body.alerts : (body.alert && typeof body.alert === "object" ? [body.alert] : null);
  if (!sourceAlerts) throw new Error("Missing alert or alerts");
  if (!sourceAlerts.length) throw new Error("alerts is empty");
  const validAlerts = [], invalidAlerts = [];
  sourceAlerts.forEach((rawAlert, idx) => {
    try { validAlerts.push(sanitizeAlert_(rawAlert)); } catch (err) { invalidAlerts.push({ stage: "sanitize", index: idx, reason: err && err.message ? err.message : String(err), rawAlert: rawAlert }); }
  });
  return { receivedCount: sourceAlerts.length, validAlerts: validAlerts, invalidAlerts: invalidAlerts };
}

function buildAndSendWeeklyReportInternal_(options) {
  let lock = null;
  options = options || {};
  try {
    lock = LockService.getScriptLock();
    lock.waitLock(30000);
    SpreadsheetApp.flush();
    const app = getAppConfig_();
    if (!app.discordStatsWebhookUrl) throw new Error("Missing Script Property: DISCORD_STATS_WEBHOOK_URL");
    const rawSh    = getOrCreateSheet_(app.spreadsheetId, CONFIG.RAW_SHEET,    RAW_HEADERS);
    const reportSh = getOrCreateSheet_(app.spreadsheetId, CONFIG.REPORT_SHEET_NAME, REPORT_HEADERS);
    const reportRange      = options.reportRange || getClosedBusinessWeekRangeJst_(new Date());
    const weekKey          = formatYmd_(reportRange.start);
    const force             = options.force           === true;
    const ignoreReported    = options.ignoreReported  === true;
    const markReported      = options.markReported    !== false;
    const updateLastSentWeek = options.updateLastSentWeek !== false;
    const props          = PropertiesService.getScriptProperties();
    const lastSentWeek   = props.getProperty("LAST_WEEKLY_REPORT_WEEK");
    if (!force && lastSentWeek === weekKey) return { ok: true, skipped: true, reason: "already sent", week: weekKey };
    const rows       = loadRecentMappedRawRows_(rawSh, CONFIG.WEEKLY_SCAN_MAX_ROWS);
    const bottomRows = rows.filter((r) => r.signalType === "BOTTOM");
    const statsList = CHECKPOINTS.map((cp) => {
      const targetRows = bottomRows.filter((r) => {
        const c = r.checkpoints[cp.days];
        return (c.evalDate instanceof Date && isDateBetweenInclusive_(c.evalDate, reportRange.start, reportRange.end) && Number.isFinite(c.perf) && (ignoreReported || !c.reported));
      });
      return buildCheckpointStats_(targetRows, cp);
    });
    const summaryText   = buildWeeklySummaryText_(reportRange.start, reportRange.end, statsList, markReported);
    const archiveMessage = buildWeeklyArchiveMessage_(reportRange.start, reportRange.end, statsList, summaryText);
    sendDiscordWeeklyEmbeds_(app.discordStatsWebhookUrl, reportRange.start, reportRange.end, statsList, summaryText);
    saveWeeklyReportRows_(reportSh, reportRange.start, reportRange.end, statsList, summaryText, archiveMessage);
    let markResult = { ok: true, updated: 0 };
    if (markReported) markResult = markCheckpointRowsReported_(rawSh, statsList);
    if (updateLastSentWeek) props.setProperty("LAST_WEEKLY_REPORT_WEEK", weekKey);
    const purgeResult = purgeArchivedRawRows_(rawSh);
    return { ok: true, sent: true, week: weekKey, reportStart: formatYmd_(reportRange.start), reportEnd: formatYmd_(reportRange.end), ignoreReported, markReported, updateLastSentWeek, markResult, purgeResult, counts: statsList.map((s) => ({ days: s.checkpoint.days, count: s.count })) };
  } catch (err) { logError_("weekly.error", err); throw err; } finally { if (lock && lock.hasLock()) lock.releaseLock(); }
}
function getClosedBusinessWeekRangeJst_(now) {
  const d = makeJstMidnightDate_(now);
  const weekday = Number(Utilities.formatDate(d, CONFIG.TIMEZONE, "u"));
  const thisWeekStart = startOfWeekJst_(d);
  const weekStart = (weekday >= 6) ? thisWeekStart : addDaysJst_(thisWeekStart, -7);
  return { start: weekStart, end: addDaysJst_(weekStart, 4) };
}
function getCurrentBusinessWeekRangeJst_(now) {
  const d = makeJstMidnightDate_(now);
  const weekStart = startOfWeekJst_(d);
  return { start: weekStart, end: addDaysJst_(weekStart, 4) };
}
function loadRecentMappedRawRows_(sh, maxRows) {
  const lastRow = sh.getLastRow();
  if (lastRow < dataStartRow_()) return [];
  const dataRows = getSheetDataRowCount_(sh);
  const takeRows = Math.min(dataRows, maxRows);
  const startRow = Math.max(dataStartRow_(), lastRow - takeRows + 1);
  const values = sh.getRange(startRow, 1, takeRows, RAW_HEADERS.length).getValues();
  return values.map((r, idx) => { const mapped = mapRawRow_(r); mapped.rowNum = startRow + idx; return mapped; }).filter((r) => Number.isFinite(r.entryPrice));
}
function buildCheckpointStats_(rows, checkpoint) {
  const count = rows.length;
  const wins = rows.filter((r) => r.checkpoints[checkpoint.days].winFlag === true).length;
  const ties = rows.filter((r) => isTiePerf_(r.checkpoints[checkpoint.days].perf)).length;
  const losses = rows.filter((r) => r.checkpoints[checkpoint.days].winFlag === false && !isTiePerf_(r.checkpoints[checkpoint.days].perf)).length;
  const decisiveCount = wins + losses;
  const winRate = count === 0 ? NaN : (decisiveCount === 0 ? 0 : wins / decisiveCount);
  const avgPerf = count ? rows.reduce((sum, r) => sum + r.checkpoints[checkpoint.days].perf, 0) / count : NaN;
  const bestBySymbol = {};
  rows.forEach((r) => {
    const key = r.symbolCode || r.symbolName || Utilities.getUuid();
    const perf = r.checkpoints[checkpoint.days].perf;
    if (!bestBySymbol[key] || perf > bestBySymbol[key].perf) bestBySymbol[key] = { rowNum: r.rowNum, symbolCode: r.symbolCode, symbolName: r.symbolName, perf: perf };
  });
  const top3 = Object.values(bestBySymbol).sort((a, b) => b.perf - a.perf).slice(0,3).map((x, idx) => ({ rank: idx+1, rowNum: x.rowNum, symbolCode: x.symbolCode, symbolName: x.symbolName, perf: x.perf }));
  return { checkpoint, count, wins, losses, ties, decisiveCount, winRate, avgPerf, top3, rowNums: rows.map(r => r.rowNum) };
}
function meanSafe_(values) { const nums = values.filter(v => Number.isFinite(v)); if (!nums.length) return NaN; return nums.reduce((a,b)=>a+b,0)/nums.length; }
function sampleBucket_(count) { if (count < 30) return "small"; if (count < 80) return "mid"; return "large"; }

function loadVariantHistory_() { try { const raw = PropertiesService.getScriptProperties().getProperty(VARIANT_HISTORY_PROP_KEY); return raw ? JSON.parse(raw) : {}; } catch(e) { return {}; } }
function saveVariantHistory_(history, usedMap) {
  if (!usedMap || !Object.keys(usedMap).length) return;
  const updated = Object.assign({}, history || {});
  Object.keys(usedMap).forEach(sectionKey => {
    const text = usedMap[sectionKey];
    if (!text) return;
    const prev = updated[sectionKey] || [];
    const next = [text].concat(prev.filter(t => t !== text)).slice(0, VARIANT_HISTORY_MAX_PER_SECTION);
    updated[sectionKey] = next;
  });
  try { PropertiesService.getScriptProperties().setProperty(VARIANT_HISTORY_PROP_KEY, JSON.stringify(updated)); } catch(e) {}
}
function normalizeSummaryText_(text) {
  return String(text || "").replace(/\s+/g,"").replace(/5営業日後時点/g,"CP").replace(/10営業日後時点/g,"CP").replace(/20営業日後時点/g,"CP").replace(/40営業日後時点/g,"CP").replace(/5営業日後/g,"CP").replace(/10営業日後/g,"CP").replace(/20営業日後/g,"CP").replace(/40営業日後/g,"CP").replace(/今週集計できた各評価地点/g,"評価地点").replace(/今週集計できた評価地点/g,"評価地点").replace(/到達済みの評価地点/g,"評価地点").replace(/全体平均/g,"全体").replace(/平均値/g,"全体").replace(/全体の数字/g,"全体").replace(/全体として/g,"全体").replace(/全体には/g,"全体").replace(/重たい内容/g,"弱い").replace(/重たさが残る/g,"弱い").replace(/上値の重さが残る/g,"弱い").replace(/戻りの鈍さが残る/g,"弱い").replace(/弱さが見られました/g,"弱い").replace(/弱めの数字が並んだ/g,"弱い").replace(/目を引き/g,"目立つ").replace(/存在感を見せました/g,"目立つ").replace(/際立ちました/g,"目立つ").replace(/印象的で/g,"目立つ").replace(/見どころの多い週でした/g,"目立つ週").replace(/個別の強さが際立ちました/g,"個別強い").replace(/個別物色の色が感じられる週でした/g,"個別強い").replace(/一部銘柄の強さがランキングに表れました/g,"個別強い").replace(/内容でした/g,"内容").replace(/週でした/g,"週").replace(/となりました/g,"週");
}
function summaryThemes_(text) {
  const t = normalizeSummaryText_(text);
  const themes = [];
  if (/全体|評価地点/.test(t)) themes.push("market");
  if (/弱い|慎重|マイナス/.test(t)) themes.push("weak");
  if (/プラス|前向き|底堅い/.test(t)) themes.push("positive");
  if (/目立つ|ランキング|個別強い/.test(t)) themes.push("top");
  if (/初動|立ち上がり|稼働初期|サンプル/.test(t)) themes.push("early");
  if (/横ばい|小動き|方向感/.test(t)) themes.push("flat");
  return themes;
}
function pushUniqueVariant_(parts, key, variants, options) {
  options = options || {};
  const allowSkip = options.allowSkip === true;
  const keyParts = key.split("|");
  const sectionKey = keyParts.slice(2).join("|") || key;
  const recentlyUsed = VARIANT_HISTORY_ ? (VARIANT_HISTORY_[sectionKey] || []) : [];
  const sorted = variants.map((v,i) => ({ v, score: Math.abs(hashCode_(key+"|"+i)) })).sort((a,b)=>a.score-b.score).map(x=>x.v);
  const existingThemes = parts.flatMap(summaryThemes_);
  for (let pass = 0; pass < 2; pass++) {
    for (const candidate of sorted) {
      if (pass === 0 && recentlyUsed.indexOf(candidate) !== -1) continue;
      const norm = normalizeSummaryText_(candidate);
      const candidateThemes = summaryThemes_(candidate);
      const duplicated = parts.some(p => { const prev = normalizeSummaryText_(p); return prev === norm || prev.includes(norm) || norm.includes(prev); });
      if (duplicated) continue;
      const weakDup = candidateThemes.includes("weak") && existingThemes.includes("weak");
      const topDup  = candidateThemes.includes("top")  && existingThemes.includes("top");

    }
  }
  if (!allowSkip && sorted.length) { parts.push(sorted[0]); if (VARIANT_USED_ !== null) VARIANT_USED_[sectionKey] = sorted[0]; }
}

const MAX_SLEEP_MS = 5 * 60 * 1000;   // Utilities.sleep() の最大許容時間（5分）

/**
 * Discord Webhook に POST する（レート制限対策＋長期再送トリガーつき）
 */
function postDiscordPayload_(webhookUrl, payload, maxRetries = 3) {
  const url = String(webhookUrl || "").trim();
  if (!url) throw new Error("Discord webhook URL is empty");

  const props = PropertiesService.getScriptProperties();

  for (let attempt = 0; attempt <= maxRetries; attempt++) {
    const res = UrlFetchApp.fetch(url, {
      method: "post",
      contentType: "application/json; charset=utf-8",
      payload: JSON.stringify(payload),
      muteHttpExceptions: true
    });

    const code = res.getResponseCode();
    if (code >= 200 && code < 300) {
      return { ok: true, status: code };
    }

    if (code === 429) {
      const retryAfterSec = Number(res.getHeaders()['Retry-After']) || 0;
      const waitMs = retryAfterSec > 0 ? retryAfterSec * 1000 : Math.pow(2, attempt) * 1000;

      if (waitMs > MAX_SLEEP_MS) {
        // 長時間スリープ不可 -> 後で実行するトリガーに任せる
        const deferredKey = "DEFERRED_DISCORD_PAYLOAD";
        const deferredData = JSON.stringify({
          webhookUrl: url,
          payload: payload,
          retryAfterMs: waitMs
        });
        props.setProperty(deferredKey, deferredData);

        // 既存の再送トリガーを削除して新規登録
        deleteTriggersByHandler_("sendDeferredDiscordPayload");
        ScriptApp.newTrigger("sendDeferredDiscordPayload")
          .timeBased()
          .after(waitMs)
          .create();

        writeProcessLog_("Discord通知", "Discordのレート制限に当たったため、待機時間後に再送するトリガーへ回します。", { waitMs: waitMs });
        return { ok: true, deferred: true, retryAfterMs: waitMs };
      }

      // 待機可能な長さならその場でスリープ
      writeProcessLog_("Discord通知", "Discordのレート制限に当たったため、短時間待って再送します。", {
        waitMs: waitMs,
        attempt: attempt + 1,
        maxAttempts: maxRetries + 1
      });
      Utilities.sleep(waitMs);
      continue;
    }

    // その他のエラー
    const body = res.getContentText();
    throw new Error(`Discord webhook failed: HTTP ${code} / ${body}`);
  }
}

/**
 * 延期された Discord ペイロードを再送する（トリガーから呼ばれる）
 */
function sendDeferredDiscordPayload() {
  const props = PropertiesService.getScriptProperties();
  const deferredKey = "DEFERRED_DISCORD_PAYLOAD";
  const raw = props.getProperty(deferredKey);
  if (!raw) return;

  let data;
  try {
    data = JSON.parse(raw);
  } catch (e) {
    props.deleteProperty(deferredKey);
    return;
  }

  try {
    const result = postDiscordPayload_(data.webhookUrl, data.payload);
    if (!result.deferred) {
      // 成功 or 致命的エラー（429以外）ならプロパティ削除
      props.deleteProperty(deferredKey);
    }
    // deferred が true なら再びトリガーが設定されるので放置
  } catch (e) {
    writeProcessLog_("Discord通知", "延期していたDiscord通知の再送に失敗しました。保存済みペイロードを削除します。", { message: e.message });
    // 最終的にあきらめる場合はプロパティ削除
    props.deleteProperty(deferredKey);
  }
}

function runDailyMaintenanceInternal_(isResume, newCount, refreshId) {
  let lock = null;
  const startedAt = Date.now();
  try {
    debugLogToSheet_("daily.start", { isResume: isResume, startedAt: new Date(startedAt).toISOString() });
    deleteTriggersByHandler_(CONFIG.DAILY_CONT_TRIGGER_HANDLER);
    if (!isResume) {
      clearDailyMaintenanceState_();
      saveDailyMaintenanceMeta_(newCount, refreshId);
    }
    const meta = loadDailyMaintenanceMeta_(newCount, refreshId);
    // 強制終了に備えて先に保険トリガーを置く。正常完了時は削除する。
    ensureSingleContinuationTrigger_(CONFIG.DAILY_SAFETY_AFTER_MS);
    lock = LockService.getScriptLock();
    lock.waitLock(30000);
    SpreadsheetApp.flush();
    const result = finalizeDueEvaluationsChunked_(startedAt, isResume);
    debugLogToSheet_("daily.result", result);
    if (result.hasMore) {
      ensureSingleContinuationTrigger_(CONFIG.DAILY_RESUME_AFTER_MS);
      return { ok: true, evaluation: result, continuation: true };
    }

    clearDailyMaintenanceState_();
    deleteTriggersByHandler_(CONFIG.DAILY_CONT_TRIGGER_HANDLER);

    // Discord通知 & GitHub Actions起動
    const displayCount = meta.newCount !== "" ? String(meta.newCount) : "0";
    const tag = (meta.refreshId && typeof meta.refreshId === "string") ? (meta.refreshId.includes("AM") ? "AM" : "PM") : "AUTO";
    sendCompletionNotice_(displayCount, "0", tag);
    triggerGitHubActionsOptimize_();

    // quickRepairRecentGaps は直接呼ばず、1分後のトリガーに分離する。
    deleteTriggersByHandler_("quickRepairTrigger");
    ScriptApp.newTrigger("quickRepairTrigger")
      .timeBased()
      .after(60 * 1000)
      .create();
    writeProcessLog_("日次メンテナンス", "評価更新が完了したため、ギャップ修復を1分後に起動します。", { handlerName: "quickRepairTrigger" });

    const purgeDebugResult = purgeOldDebugLogs_();
    return { ok: true, evaluation: result, purgeDebugResult: purgeDebugResult };
  } catch (err) {
    debugLogToSheet_("daily.error", { message: err && err.message ? err.message : String(err), stack: err && err.stack ? String(err.stack) : "" });
    if (isDailyMaintenanceRetriableError_(err)) {
      ensureSingleContinuationTrigger_(CONFIG.DAILY_RESUME_AFTER_MS);
      return { ok: false, transient: true, message: err && err.message ? err.message : String(err) };
    }
    deleteTriggersByHandler_(CONFIG.DAILY_CONT_TRIGGER_HANDLER);
    logError_("daily.error", err);
    throw err;
  } finally {
    if (lock && lock.hasLock()) lock.releaseLock();
    debugLogToSheet_("daily.end", { elapsedMs: Date.now() - startedAt });
  }
}

// 【修正1】finalizeDueEvaluationsChunked_
// 変更した行だけを個別に書き戻す（全行setValuesをやめる）
// =====================================
function finalizeDueEvaluationsChunked_(startedAtMs, isResume) {
  const app = getAppConfig_();
  const sh = getOrCreateSheet_(app.spreadsheetId, CONFIG.RAW_SHEET, RAW_HEADERS);
  const props = PropertiesService.getScriptProperties();
  const priceMemo = {};
  const startRow = dataStartRow_();
  const lastRow = sh.getLastRow();
  if (lastRow < startRow) {
    props.deleteProperty(DAILY_MAINT_CURSOR_KEY);
    return { ok: true, hasMore: false, scannedRows: 0, updatedRows: 0, dueRows: 0 };
  }

  let cursor = Math.max(startRow, parseInt(props.getProperty(DAILY_MAINT_CURSOR_KEY) || String(startRow), 10));
  if (!Number.isFinite(cursor) || cursor < startRow) cursor = startRow;
  if (cursor > lastRow) {
    props.deleteProperty(DAILY_MAINT_CURSOR_KEY);
    return { ok: true, hasMore: false, scannedRows: 0, updatedRows: 0, dueRows: 0 };
  }

  const todayKey = getOhlcvBusinessDateKey_();
  writeProcessLog_("日次メンテナンス", `評価日判定の基準日は ${todayKey} です。`, { businessDate: todayKey, isManual: !!getManualOhlcvBusinessDateKey_() });
  const ABNORMAL_ENTRY_PRICE_THRESHOLD = 10000000;
  let scannedRows = 0;
  let updatedRows = 0;
  let dueRows = 0;
  let chunks = 0;

  while (cursor <= lastRow) {
    const elapsed = Date.now() - startedAtMs;
    if (elapsed > CONFIG.DAILY_SOFT_LIMIT_MS - DAILY_MAINT_FETCH_GUARD_MS) {
      props.setProperty(DAILY_MAINT_CURSOR_KEY, String(cursor));
      return { ok: true, hasMore: true, nextRow: cursor, lastRow: lastRow, scannedRows: scannedRows, updatedRows: updatedRows, dueRows: dueRows, chunks: chunks, elapsedMs: elapsed };
    }

    const chunkSize = Math.min(DAILY_MAINT_ROW_CHUNK_SIZE, lastRow - cursor + 1);
    const values = sh.getRange(cursor, 1, chunkSize, RAW_HEADERS.length).getValues();
    const dirtyIndices = [];

    for (let i = 0; i < values.length; i++) {
      const elapsedBeforeRow = Date.now() - startedAtMs;
      if (elapsedBeforeRow > CONFIG.DAILY_SOFT_LIMIT_MS - DAILY_MAINT_FETCH_GUARD_MS) {
        writeDirtyRawRows_(sh, cursor, values, dirtyIndices);
        props.setProperty(DAILY_MAINT_CURSOR_KEY, String(cursor + i));
        return { ok: true, hasMore: true, nextRow: cursor + i, lastRow: lastRow, scannedRows: scannedRows, updatedRows: updatedRows, dueRows: dueRows, chunks: chunks, elapsedMs: elapsedBeforeRow };
      }

      scannedRows++;
      const row = values[i];
      const symbolCode = String(row[RAW_IDX.symbol_code] || "").trim();
      const status = row[RAW_IDX.status];
      if (!symbolCode || (status !== "PENDING" && status !== "NO_PRICE")) continue;

      let entryPrice = Number(row[RAW_IDX.entry_price]);
      if (Number.isFinite(entryPrice) && entryPrice > ABNORMAL_ENTRY_PRICE_THRESHOLD) {
        row[RAW_IDX.status] = "COMPLETE";
        CHECKPOINTS.forEach(cp => { row[RAW_IDX[cp.reportedKey]] = true; });
        row[RAW_IDX.note] = appendTag_(row[RAW_IDX.note], "MBO_ABNORMAL_PRICE");
        dirtyIndices.push(i);
        updatedRows++;
        continue;
      }

      const dueCheckpoints = CHECKPOINTS.filter(cp => {
        const evalDate = row[RAW_IDX[cp.evalDateKey]];
        const hasPrice = Number.isFinite(toFiniteNumberOrNaN_(row[RAW_IDX[cp.evalCloseKey]]));
        return (evalDate instanceof Date) && (formatYmd_(evalDate) <= todayKey) && !hasPrice;
      });
      if (dueCheckpoints.length === 0) continue;
      dueRows++;

      const market = getLatestMarketDataCached_(symbolCode, priceMemo);
      if (market.error === "DELISTED") {
        dueCheckpoints.forEach(cp => { row[RAW_IDX[cp.evalCloseKey]] = "上場廃止"; });
        row[RAW_IDX.status] = "COMPLETE";
        row[RAW_IDX.note] = appendTag_(row[RAW_IDX.note], "DELISTED");
        dirtyIndices.push(i);
        updatedRows++;
        continue;
      }
      if (!market.price) {
        row[RAW_IDX.status] = "NO_PRICE";
        dirtyIndices.push(i);
        updatedRows++;
        continue;
      }
      if (market.splitRatio && market.splitRatio !== 1) {
        const splitTag = `SPLIT_ADJUSTED_${market.splitDate}`;

        // 分割日がシグナル日より後の場合のみ調整する。
        const splitDateJst = Utilities.formatDate(
          new Date((market.splitDate || 0) * 1000), CONFIG.TIMEZONE, "yyyy-MM-dd"
        );
        const signalDate = row[RAW_IDX.signal_date];
        const signalDateJst = signalDate instanceof Date
          ? Utilities.formatDate(signalDate, CONFIG.TIMEZONE, "yyyy-MM-dd")
          : "";
        const splitIsAfterSignal = signalDateJst !== "" && splitDateJst > signalDateJst;

        if (String(row[RAW_IDX.note]).indexOf(splitTag) === -1 && splitIsAfterSignal) {
          entryPrice = entryPrice / market.splitRatio;
          row[RAW_IDX.entry_price] = entryPrice;
          row[RAW_IDX.note] = appendTag_(row[RAW_IDX.note], splitTag);
        }
      }
      dueCheckpoints.forEach(cp => {
        const perf = (market.price - entryPrice) / entryPrice;
        row[RAW_IDX[cp.evalCloseKey]] = market.price;
        row[RAW_IDX[cp.perfKey]] = perf;
        row[RAW_IDX[cp.winKey]] = (row[RAW_IDX.signal_type] === "TOP") ? (perf < 0) : (perf > 0);
      });
      row[RAW_IDX.status] = computeRowStatusFromArray_(row);
      dirtyIndices.push(i);
      updatedRows++;
    }

    writeDirtyRawRows_(sh, cursor, values, dirtyIndices);
    cursor += chunkSize;
    chunks++;
    props.setProperty(DAILY_MAINT_CURSOR_KEY, String(cursor));
  }

  props.deleteProperty(DAILY_MAINT_CURSOR_KEY);
  return { ok: true, hasMore: false, nextRow: cursor, lastRow: lastRow, scannedRows: scannedRows, updatedRows: updatedRows, dueRows: dueRows, chunks: chunks, elapsedMs: Date.now() - startedAtMs };
}

function writeDirtyRawRows_(sh, chunkStartRow, values, dirtyIndices) {
  if (!dirtyIndices || dirtyIndices.length === 0) return;
  dirtyIndices.forEach(i => {
    sh.getRange(chunkStartRow + i, 1, 1, RAW_HEADERS.length).setValues([values[i]]);
  });
}
function getLatestMarketDataCached_(symbolCode, runMemo, options) {
  const code = String(symbolCode || "").trim().toUpperCase();
  const forceRefresh = options?.forceRefresh === true;
  if (!forceRefresh && runMemo && runMemo[code]) return runMemo[code];
  const result = fetchYahooFinanceDataAPI_(code);
  if (result.price === Number(code.replace(".T", ""))) { result.error = "DELISTED"; result.price = null; }
  if (runMemo) runMemo[code] = result;
  return result;
}
function computeRowStatusFromArray_(row) { const allDone = CHECKPOINTS.every(cp => Number.isFinite(toFiniteNumberOrNaN_(row[RAW_IDX[cp.evalCloseKey]]))); return allDone ? "COMPLETE" : "PENDING"; }
/**
 * DAILY_MAINTENANCE 用のトリガー作成も同様に堅牢化（念のため）
 */
function ensureSingleContinuationTrigger_(delayMs) {
  const handlerName = CONFIG.DAILY_CONT_TRIGGER_HANDLER;
  const triggerDelayMs = Number.isFinite(delayMs) ? delayMs : CONFIG.DAILY_RESUME_AFTER_MS;
  deleteTriggersByHandler_(handlerName);
  
  const maxRetries = 3;
  for (let i = 0; i < maxRetries; i++) {
    try {
      ScriptApp.newTrigger(handlerName)
        .timeBased()
        .after(triggerDelayMs)
        .create();
      return;
    } catch (e) {
      if (i === maxRetries - 1) throw e;
      writeProcessLog_("日次メンテナンス", "再開トリガーの作成に失敗したため、少し待って再試行します。", {
        attempt: i + 1,
        maxRetries: maxRetries,
        message: e.message
      });
      Utilities.sleep(2000 * (i + 1));
    }
  }
}

function purgeArchivedRawRows_(sh) {
  try {
    const lastRow = sh.getLastRow();
    if (lastRow < CONFIG.DATA_START_ROW) return { ok: true, deleted: 0 };
    const data = sh.getRange(CONFIG.DATA_START_ROW, 1, lastRow - CONFIG.HEADER_ROW, RAW_HEADERS.length).getValues();
    const now = new Date();
    const deleteRowNums = [], deleteAlertIds = [];
    for (let i = 0; i < data.length; i++) {
      const row = data[i];
      const status = row[RAW_IDX.status];
      const signalType = row[RAW_IDX.signal_type];
      const loggedAt = row[RAW_IDX.logged_at] instanceof Date ? row[RAW_IDX.logged_at] : null;
      if (status !== "COMPLETE" || !loggedAt) continue;
      const retentionDays = (signalType === "TOP") ? CONFIG.TOP_COMPLETED_RETENTION_DAYS : 7;
      const diffDays = (now - loggedAt) / (1000 * 60 * 60 * 24);
      if (diffDays >= retentionDays) {
        deleteRowNums.push(CONFIG.DATA_START_ROW + i);
        const alertId = row[RAW_IDX.alert_id];
        if (alertId) deleteAlertIds.push(alertId);
      }
    }
    if (deleteRowNums.length === 0) return { ok: true, deleted: 0 };
    if (deleteAlertIds.length > 0) purgeOhlcvRowsByAlertIds_(deleteAlertIds);
    deleteRowsByRowNums_(sh, deleteRowNums);
    return { ok: true, deleted: deleteRowNums.length };
  } catch (err) { return { ok: false, error: err && err.message ? err.message : String(err) }; }
}
function purgeOhlcvRowsByAlertIds_(alertIds) {
  const app = getAppConfig_();
  const ss = SpreadsheetApp.openById(app.spreadsheetId);
  const ohlcvSh = ss.getSheetByName(CONFIG.OHLCV_SHEET);
  if (!ohlcvSh) return;
  const lastRow = ohlcvSh.getLastRow();
  if (lastRow < CONFIG.DATA_START_ROW) return;
  const data = ohlcvSh.getRange(CONFIG.DATA_START_ROW, 1, lastRow - CONFIG.HEADER_ROW, 1).getValues();
  const idSet = new Set(alertIds.map(id => String(id)));
  const deleteRowNums = [];
  for (let i = 0; i < data.length; i++) if (idSet.has(String(data[i][0]))) deleteRowNums.push(CONFIG.DATA_START_ROW + i);
  if (deleteRowNums.length > 0) { deleteRowsByRowNums_(ohlcvSh, deleteRowNums); debugLogToSheet_("purge.ohlcv", { deleted: deleteRowNums.length, ids: alertIds.length }); }
}
function deleteRowsByRowNums_(sh, rowNums) {
  if (!rowNums || !rowNums.length) return;
  const sorted = rowNums.slice().sort((a, b) => b - a);
  let start = sorted[0], count = 1;
  for (let i = 1; i < sorted.length; i++) {
    if (sorted[i] === start - count) count++;
    else { sh.deleteRows(start - count + 1, count); start = sorted[i]; count = 1; }
  }
  sh.deleteRows(start - count + 1, count);
}

/**
 * ohlcv_4h シートから365日以上経過した古いデータを削除する（バッチ処理・再開可能）
 * - シートのA列（timestamp）が昇順ソートされている前提
 * - 1回の実行で最大3.5分間処理し、終わらなければ1分後に自分自身を再呼び出しする
 * - 毎日深夜のトリガーから呼び出すことを想定
 */
function purgeOldOhlcvDataDaily() {
  const startTime = Date.now();
  const MAX_RUN_MS = 3.5 * 60 * 1000;
  const BATCH_SIZE = 5000;
  const RETRY_DELAY_MS = 60 * 1000;

  const app = getAppConfig_();
  const ss = SpreadsheetApp.openById(app.spreadsheetId);
  const sheet = ss.getSheetByName(CONFIG.OHLCV_SHEET);
  if (!sheet) {
    console.error("ohlcv_4h sheet not found");
    return;
  }

  const HEADER_ROW = 1;
  const firstDataRow = HEADER_ROW + 1;
  const lastRow = sheet.getLastRow();
  if (lastRow < firstDataRow) return;

  const cutoff = new Date();
  cutoff.setDate(cutoff.getDate() - 365);
  cutoff.setHours(0, 0, 0, 0);
  console.log(`[purgeOldOhlcv] 基準日: ${formatYmd_(cutoff)} より古いデータを削除対象とします`);

  let totalDeleted = 0;
  let totalBatches = 0;

  while (Date.now() - startTime < MAX_RUN_MS) {
    const firstTs = sheet.getRange(firstDataRow, 1).getValue();
    if (!(firstTs instanceof Date)) break;
    if (firstTs >= cutoff) break;

    const timestamps = sheet.getRange(firstDataRow, 1, BATCH_SIZE, 1).getValues();

    let deleteCount = 0;
    let minDate = null, maxDate = null;
    for (let i = 0; i < timestamps.length; i++) {
      const ts = timestamps[i][0];
      if (ts instanceof Date && ts < cutoff) {
        deleteCount++;
        if (!minDate || ts < minDate) minDate = ts;
        if (!maxDate || ts > maxDate) maxDate = ts;
      } else break;
    }

    if (deleteCount === 0) break;

    const minDateStr = minDate ? formatYmd_(minDate) : "不明";
    const maxDateStr = maxDate ? formatYmd_(maxDate) : "不明";
    console.log(`[purgeOldOhlcv] 削除対象: ${minDateStr} 〜 ${maxDateStr} (${deleteCount}行)`);

    sheet.deleteRows(firstDataRow, deleteCount);
    totalDeleted += deleteCount;
    totalBatches++;
  }

  console.log(`[purgeOldOhlcv] 今回の実行結果: 合計 ${totalDeleted} 行削除 (${totalBatches}バッチ), 処理時間=${Date.now() - startTime}ms`);

  const newFirstTs = (sheet.getLastRow() >= firstDataRow) ? sheet.getRange(firstDataRow, 1).getValue() : null;
  const stillNeedsWork = (newFirstTs instanceof Date && newFirstTs < cutoff);

  if (stillNeedsWork) {
    console.log(`[purgeOldOhlcv] 削除が完了しませんでした。1分後に再開します。`);
    // 再開用は別ハンドラー名で登録（毎日トリガーを消さないため）
    deleteTriggersByHandler_("purgeOldOhlcvResumeTrigger");
    ScriptApp.newTrigger("purgeOldOhlcvResumeTrigger").timeBased().after(RETRY_DELAY_MS).create();
  } else {
    console.log(`[purgeOldOhlcv] 全ての古いデータを削除しました。`);
    // 再開用トリガーのみ削除（毎日トリガーは残す）
    deleteTriggersByHandler_("purgeOldOhlcvResumeTrigger");
  }
}

/**
 * purgeOldOhlcvDataDaily の再開用ラッパー（.after()トリガーから呼ばれる）
 */
function purgeOldOhlcvResumeTrigger() {
  deleteTriggersByHandler_("purgeOldOhlcvResumeTrigger");
  purgeOldOhlcvDataDaily();
}

function syncMarketHolidays() {
  const now = new Date();
  const currentYear = Number(Utilities.formatDate(now, CONFIG.TIMEZONE, "yyyy"));
  const fromYear = currentYear - CONFIG.HOLIDAY_SYNC_YEARS_BACK;
  const toYear = currentYear + CONFIG.HOLIDAY_SYNC_YEARS_FORWARD;
  return syncMarketHolidaysRange_(fromYear, toYear);
}
function syncMarketHolidaysRange_(fromYear, toYear) {
  if (!Number.isFinite(fromYear) || !Number.isFinite(toYear) || fromYear > toYear) throw new Error("invalid year range");
  const app = getAppConfig_();
  const sh = getOrCreateSheet_(app.spreadsheetId, CONFIG.HOLIDAY_SHEET_NAME, ["date", "name", "source"]);
  const allRows = [];
  const csvUrl = "https://www8.cao.go.jp/chosei/shukujitsu/syukujitsu.csv";
  const res = UrlFetchApp.fetch(csvUrl, { muteHttpExceptions: true });
  if (res.getResponseCode() !== 200) throw new Error(`failed to fetch holiday csv: ${res.getResponseCode()}`);
  const csvText = res.getBlob().getDataAsString("Shift_JIS");
  const csvRows = Utilities.parseCsv(csvText);
  for (let i = 1; i < csvRows.length; i++) {
    const dateStr = String(csvRows[i][0] || "").trim();
    const name = String(csvRows[i][1] || "").trim();
    if (!dateStr || !name) continue;
    const d = parseYmdFlexible_(dateStr);
    if (!d) continue;
    const y = Number(Utilities.formatDate(d, CONFIG.TIMEZONE, "yyyy"));
    if (y < fromYear || y > toYear) continue;
    allRows.push([formatYmdSlash_(d), name, "CABINET_OFFICE"]);
  }
  buildJpxExtraClosures_(fromYear, toYear).forEach((item) => allRows.push([formatYmdSlash_(item.date), item.name, "JPX_EXTRA"]));
  const dedup = {};
  allRows.forEach(row => { const key = String(row[0] || "").trim(); if (key && !dedup[key]) dedup[key] = row; });
  const finalRows = Object.keys(dedup).sort().map(k => dedup[k]);
  sh.clearContents();
  ensureTopRowsAndHeader_(sh, ["date", "name", "source"]);
  if (finalRows.length) {
    sh.getRange(dataStartRow_(), 1, finalRows.length, 3).setValues(finalRows);
    sh.getRange(dataStartRow_(), 1, finalRows.length, 1).setNumberFormat("yyyy/mm/dd");
  }
  MARKET_HOLIDAY_CACHE_ = null;
  return { ok: true, fromYear: fromYear, toYear: toYear, rows: finalRows.length };
}
function buildJpxExtraClosures_(fromYear, toYear) {
  const rows = [];
  for (let y = fromYear; y <= toYear; y++) {
    rows.push({ date: new Date(`${y}/01/02 00:00:00 +0900`), name: "年始休業日" });
    rows.push({ date: new Date(`${y}/01/03 00:00:00 +0900`), name: "年始休業日" });
    rows.push({ date: new Date(`${y}/12/31 00:00:00 +0900`), name: "年末休業日" });
  }
  return rows;
}

function parseEnvelope_(e) {
  if (!e || !e.postData || !e.postData.contents) throw new Error("Missing POST body");
  const contentType = String(e.postData.type || "");
  if (contentType && contentType.indexOf("application/json") === -1) throw new Error("Content-Type must be application/json");
  let body;
  try { body = JSON.parse(e.postData.contents); } catch (err) { throw new Error("Invalid JSON body"); }
  if (!body || typeof body !== "object") throw new Error("Invalid request body");
  return body;
}
function verifyEnvelope_(body, secret) {
  const v = String(body.v || ""), ts = Number(body.ts), sig = String(body.sig || ""), payloadJson = String(body.payloadJson || "");
  if (v !== CONFIG.SIGN_VERSION) throw new Error("Invalid sign version");
  if (!Number.isFinite(ts)) throw new Error("Invalid timestamp");
  if (!sig) throw new Error("Missing signature");
  if (!payloadJson) throw new Error("Missing payloadJson");
  const now = Date.now();
  if (Math.abs(now - ts) > CONFIG.MAX_CLOCK_SKEW_MS) throw new Error(`Expired request: now=${now}, ts=${ts}, diffMs=${Math.abs(now - ts)}`);
  const message = `${v}.${ts}.${payloadJson}`;
  const expected = computeHmacHex_(message, secret);
  debugLogToSheet_("verifyEnvelope.debug", { sigPrefix: sig.slice(0,16), expectedPrefix: expected.slice(0,16), sigLength: sig.length, expectedLength: expected.length, payloadJsonLength: payloadJson.length, secretLength: String(secret || "").length });
  if (!constantTimeEqual_(sig, expected)) throw new Error(`Invalid signature: sigLength=${sig.length}, expectedLength=${expected.length}`);
}
function computeHmacHex_(message, secret) { return Utilities.computeHmacSha256Signature(message, secret, Utilities.Charset.UTF_8).map(b => (b < 0 ? b+256 : b).toString(16).padStart(2,"0")).join(""); }
function constantTimeEqual_(a, b) {
  a = String(a || ""); b = String(b || "");
  const len = Math.max(a.length, b.length);
  let diff = a.length ^ b.length;
  for (let i = 0; i < len; i++) { const ca = i < a.length ? a.charCodeAt(i) : 0; const cb = i < b.length ? b.charCodeAt(i) : 0; diff |= (ca ^ cb); }
  return diff === 0;
}

function sanitizeAlert_(alert) {
  const signalType = toSafeCellString_(String(alert.signalType || "").trim().toUpperCase(), 30);
  const timeframe = toSafeCellString_(String(alert.timeframe || "").trim(), 30);
  const symbolCode = toSafeCellString_(String(alert.symbolCode || "").trim(), 30);

  const receivedAt = toSafeCellString_(String(alert.receivedAt || "").trim(), 80);

  if (!symbolCode) throw new Error("symbolCode is required");
  if (signalType !== "BOTTOM" && signalType !== "TOP") throw new Error("signalType must be BOTTOM or TOP");

function appendAlertRowsSafe_(sh, alerts) {
  if (!alerts || !alerts.length) return { inserted: 0, failedRows: [] };
  const buildOkItems = [], failedRows = [];
  alerts.forEach((alert, idx) => { try { buildOkItems.push({ alert, row: buildAlertRow_(alert), originalIndex: idx }); } catch (err) { failedRows.push({ stage: "buildRow", index: idx, reason: err && err.message ? err.message : String(err), alert }); } });
  if (!buildOkItems.length) return { inserted: 0, failedRows };
  const rows = buildOkItems.map(x => x.row);
  const startRow = getSafeAppendStartRow_(sh);
  try {
    sh.getRange(startRow, 1, rows.length, RAW_HEADERS.length).setValues(rows);
    applyRawFormats_(sh, startRow, rows.length);
    return { inserted: rows.length, failedRows };
  } catch (batchErr) {
    let inserted = 0;
    buildOkItems.forEach(item => {
      try { const rowNum = getSafeAppendStartRow_(sh); sh.getRange(rowNum, 1, 1, RAW_HEADERS.length).setValues([item.row]); applyRawFormats_(sh, rowNum, 1); inserted++; } catch (rowErr) { failedRows.push({ stage: "writeRow", index: item.originalIndex, reason: rowErr && rowErr.message ? rowErr.message : String(rowErr), alert: item.alert }); }
    });
    return { inserted, failedRows };
  }
}
function buildAlertRow_(alert) {
  const receivedAt = parseIsoDateOrThrow_(alert.receivedAt);
  const signalDate = makeJstMidnightDate_(receivedAt);
  const signalWeekStart = startOfWeekJst_(signalDate);

    alert.symbolCode, alert.symbolName, alert.closePrice, alert.volume, `TYO:${alert.symbolCode}`,
    addBusinessDaysJst_(signalDate, 5), "", "", "", false,
    addBusinessDaysJst_(signalDate, 10), "", "", "", false,
    addBusinessDaysJst_(signalDate, 20), "", "", "", false,
    addBusinessDaysJst_(signalDate, 40), "", "", "", false,
    "PENDING", "", new Date()
  ];
}

function parseIsoDateOrThrow_(value) { const d = new Date(value); if (isNaN(d.getTime())) throw new Error(`Invalid receivedAt: ${value}`); return d; }
function makeJstMidnightDate_(date) { const ymd = Utilities.formatDate(new Date(date), CONFIG.TIMEZONE, "yyyy-MM-dd"); return new Date(`${ymd}T00:00:00+09:00`); }
function addDaysJst_(date, days) { return new Date(date.getTime() + days * 24 * 60 * 60 * 1000); }
function addBusinessDaysJst_(date, businessDays) {
  let d = makeJstMidnightDate_(date);
  let added = 0;
  const holidaySet = getMarketHolidaySet_();
  while (added < businessDays) { d = addDaysJst_(d, 1); if (!isMarketHolidayJst_(d, holidaySet)) added++; }
  return d;
}
function startOfWeekJst_(date) { const d = makeJstMidnightDate_(date); const weekday = Number(Utilities.formatDate(d, CONFIG.TIMEZONE, "u")); return addDaysJst_(d, -(weekday - 1)); }

function isDateBetweenInclusive_(date, start, end) { if (!(date instanceof Date) || !(start instanceof Date) || !(end instanceof Date)) return false; const k = formatYmd_(date); return k >= formatYmd_(start) && k <= formatYmd_(end); }
function parseYmdFlexible_(s) {
  const text = String(s || "").trim();
  if (!text) return null;
  const m = text.match(/^(\d{4})[\/\-](\d{1,2})[\/\-](\d{1,2})$/);
  if (!m) return null;
  const y = Number(m[1]), mo = Number(m[2]), d = Number(m[3]);
  if (!Number.isFinite(y) || !Number.isFinite(mo) || !Number.isFinite(d)) return null;
  return new Date(`${y}/${pad2_(mo)}/${pad2_(d)} 00:00:00 +0900`);
}
function pad2_(n) { return String(n).padStart(2, "0"); }

// "-26.2008733624454%" のような文字列も、0.262... のような数値も両方対応
function parsePerfValue_(value) {
  if (value === '' || value == null) return NaN;
  const s = String(value).replace(/%/g, '').trim();
  const n = Number(s);
  if (!Number.isFinite(n)) return NaN;
  // % 記号が含まれていた場合は100で割って小数に変換
  // 含まれていない場合はそのまま（既に小数の場合）
  return String(value).includes('%') ? n / 100 : n;
}

function buildLastTsMapFromOhlcv_(sheet, symbols, repairSymsSet) {
  const lastRow = sheet.getLastRow();
  const map = {};
  if (lastRow < 2 || !symbols || symbols.length === 0) return map;

  const target = {};
  symbols.forEach(sym => {
    const s = String(sym || "").trim();
    if (s && !(repairSymsSet && repairSymsSet.has(s))) target[s] = true;
  });

  let found = 0;
  const targetCount = Object.keys(target).length;
  let row = lastRow;
  let chunks = 0;
  let scannedRows = 0;

  while (row >= 2 && found < targetCount && scannedRows < OHLCV_LAST_TS_SCAN_MAX_ROWS) {
    const numRows = Math.min(OHLCV_LAST_TS_SCAN_CHUNK_ROWS, row - 1);
    const startRow = row - numRows + 1;
    const values = sheet.getRange(startRow, 1, numRows, 3).getValues();
    chunks++;
    scannedRows += numRows;

    for (let i = values.length - 1; i >= 0; i--) {
      const ts = values[i][0];
      const sym = String(values[i][2] || "").trim();
      if (!target[sym] || map[sym] || !(ts instanceof Date)) continue;
      map[sym] = ts.getTime();
      found++;
      if (found >= targetCount) break;
    }

    row = startRow - 1;
  }

  debugLogToSheet_("ohlcv.lastTsMap.scan", {
    targetSymbols: targetCount,
    foundSymbols: found,
    chunks: chunks,
    scannedRows: scannedRows,
    maxRows: OHLCV_LAST_TS_SCAN_MAX_ROWS
  });
  return map;
}

function applyPhase1ResultRows_(symbol, resultRows, refreshId, lastTsMap) {
  if (!resultRows || resultRows.length === 0) return [];
  let skippedInvalidTs = 0;
  const skippedSamples = [];
  const validRows = [];
  resultRows.forEach((row, index) => {
    const ts = row && row.length ? row[0] : null;
    if (!isValidOhlcvTimestamp_(ts)) {
      skippedInvalidTs++;
      if (skippedSamples.length < 5) {
        skippedSamples.push({ index: index, timestamp: ts == null ? "" : String(ts), symbol: symbol });
      }
      return;
    }
    row[1] = refreshId;
    validRows.push(row);
  });
  if (skippedInvalidTs > 0) {
    debugLogToSheet_("ohlcv.phase1.skippedInvalidTimestamp", {
      symbol: symbol,
      skipped: skippedInvalidTs,
      samples: skippedSamples
    });
  }
  if (validRows.length > 0) {
    const lastTimestamp = validRows[validRows.length - 1][0];
    if (lastTimestamp instanceof Date) {
      const currentLast = lastTsMap[symbol];
      if (!currentLast || lastTimestamp.getTime() > currentLast) {
        lastTsMap[symbol] = lastTimestamp.getTime();
      }
    }
  }
  return validRows;
}

function applySplitAdjustToRawRow_(row, splitInfo) {
  if (!splitInfo || !splitInfo.splitRatio || splitInfo.splitRatio === 1 || !splitInfo.splitDate) return false;
  const splitTag = `SPLIT_ADJ_${splitInfo.splitDate}`;
  const note = String(row[RAW_IDX.note] || "");
  if (note.indexOf(splitTag) !== -1) return false;
  const alertDate = row[RAW_IDX.signal_date];
  if (!(alertDate instanceof Date) || (alertDate.getTime() / 1000 >= splitInfo.splitDate)) return false;
  const ratio = splitInfo.splitRatio;
  const priceCols = [RAW_IDX.entry_price, RAW_IDX.eval_close_5bd, RAW_IDX.eval_close_10bd, RAW_IDX.eval_close_20bd, RAW_IDX.eval_close_40bd];
  priceCols.forEach(idx => { if (Number.isFinite(row[idx]) && row[idx] > 0) row[idx] = Math.round((row[idx] / ratio) * 100) / 100; });
  const entryPrice = row[RAW_IDX.entry_price];
  if (Number.isFinite(entryPrice) && entryPrice > 0) {
    const cpCols = [
      { close: RAW_IDX.eval_close_5bd, perf: RAW_IDX.perf_5bd, win: RAW_IDX.win_flag_5bd },
      { close: RAW_IDX.eval_close_10bd, perf: RAW_IDX.perf_10bd, win: RAW_IDX.win_flag_10bd },
      { close: RAW_IDX.eval_close_20bd, perf: RAW_IDX.perf_20bd, win: RAW_IDX.win_flag_20bd },
      { close: RAW_IDX.eval_close_40bd, perf: RAW_IDX.perf_40bd, win: RAW_IDX.win_flag_40bd }
    ];
    const isTop = row[RAW_IDX.signal_type] === "TOP";
    cpCols.forEach(cp => {
      const evalClose = row[cp.close];
      if (Number.isFinite(evalClose) && evalClose > 0) {
        const perf = (evalClose - entryPrice) / entryPrice;
        row[cp.perf] = perf;
        row[cp.win] = isTop ? (perf < 0) : (perf > 0);
      }
    });
  }
  row[RAW_IDX.note] = appendTag_(note, splitTag);
  return true;
}

function fetchOHLCVForNewAlerts() {
  const entryTime = Date.now();
  let lock = null;
  writeProcessLog_("OHLCV", "fetchOHLCVForNewAlertsを開始しました。休場日判定、ロック取得、PHASE進行の順に処理します。", {
    businessDate: getOhlcvBusinessDateKey_(),
    isManual: !!getManualOhlcvBusinessDateKey_()
  });
  traceOhlcv_("fetch.entry", { entryTime: new Date(entryTime).toISOString() });
  try {
    // 入口ログ後の初期処理で強制終了しても、再開だけは残るように先に保険を置く。
    scheduleOhlcvResume_("resumeOHLCVFetch", OHLCV_SAFETY_RESUME_AFTER_MS);
    logOhlcvRuntime_("entry", { triggeredAt: new Date(entryTime).toISOString() });
    lock = LockService.getScriptLock();
    if (!lock.tryLock(OHLCV_LOCK_WAIT_MS)) {
      logOhlcvRuntime_("lock_busy", { waitMs: OHLCV_LOCK_WAIT_MS });
      setupResumeTrigger_("resumeOHLCVFetch");
      return { ok: false, skipped: true, reason: "lock_busy" };
    }
    return fetchOHLCVForNewAlertsInternal_(entryTime);
  } catch (e) {
    const errData = {
      message: e && e.message ? e.message : String(e),
      stack: e && e.stack ? String(e.stack) : "",
      elapsedMs: Date.now() - entryTime
    };
    console.error("[OHLCV] fetchOHLCVForNewAlerts fatal " + JSON.stringify(errData));
    debugLogToSheet_("ohlcv.fatal", errData);
    try {
      setupResumeTrigger_("resumeOHLCVFetch");
    } catch (scheduleErr) {
      console.error("[OHLCV] failed to schedule resume after fatal: " + (scheduleErr && scheduleErr.message ? scheduleErr.message : String(scheduleErr)));
    }
    throw e;
  } finally {
    if (lock && lock.hasLock()) lock.releaseLock();
    writeProcessLog_("OHLCV", "fetchOHLCVForNewAlertsの今回実行分を終了しました。未完了の場合は再開トリガーが続きます。", { elapsedMs: Date.now() - entryTime });
  }
}

      const rawData = rawSheet.getRange(5, 1, Math.max(1, rawSheet.getLastRow() - 4), RAW_HEADERS.length).getValues();
      const uniqueSymbols = new Set();
      rawData.forEach(r => { if (r[RAW_IDX.symbol_code]) uniqueSymbols.add(String(r[RAW_IDX.symbol_code]).trim()); });
      symbols = Array.from(uniqueSymbols);
      // ohlcv_4h 全行読みは巨大シートで開始直後タイムアウトの原因になるため、末尾から必要銘柄だけ探す。
      lastTsMap = buildLastTsMapFromOhlcv_(ohlcvSheet, symbols, repairSymsSet);
      
      props.setProperties({ "LAST_TS_MAP": JSON.stringify(lastTsMap), [SYMBOL_LIST_KEY]: JSON.stringify(symbols), [STATE_KEY]: "0" });
    }

    let startIndex = parseInt(props.getProperty(STATE_KEY) || "0");
    let allFetchedData = [];
    let batchNo = 0;
    for (let i = startIndex; i < symbols.length; i += OHLCV_BATCH_SIZE) {
      const elapsedBeforeBatch = Date.now() - startTime;
      if (elapsedBeforeBatch > LIMIT_MS || (LIMIT_MS - elapsedBeforeBatch) < FETCH_GUARD_MS) {
        if (allFetchedData.length > 0) appendRowsToSheet_(ohlcvSheet, allFetchedData);
        props.setProperty(STATE_KEY, i.toString());
        props.setProperty(NEW_ALERT_COUNT_KEY, newAlertCount.toString());
        props.setProperty("LAST_TS_MAP", JSON.stringify(lastTsMap));
        writeProcessLog_("OHLCV", `PHASE1: 実行時間が近づいたため、${i}/${symbols.length} 銘柄までの進捗を保存して再開します。`, {
          cursor: i,
          totalSymbols: symbols.length,
          elapsedMs: elapsedBeforeBatch
        });
        debugLogToSheet_("ohlcv.phase.summary", {
          event: "phase_pause",
          phase: "PHASE1",
          cursor: i,
          totalSymbols: symbols.length,
          elapsedMs: elapsedBeforeBatch
        });
        setupResumeTrigger_("resumeOHLCVFetch");
        return;
      }
      const batchStart = Date.now();
      const batch = symbols.slice(i, Math.min(i + OHLCV_BATCH_SIZE, symbols.length));
      const requests = [];
      const reqMeta = [];
      batch.forEach(sym => {
        try {
          const pair = buildOhlcvRequestPair_(sym, lastTsMap[sym] || null);
          reqMeta.push({ symbol: sym, intraIdx: requests.length });
          requests.push({ url: pair.intraUrl, muteHttpExceptions: true, headers: { "User-Agent": "Mozilla/5.0" } });
        } catch (e) {
          writeProcessLog_("OHLCV", `PHASE1: ${sym} のYahoo Finance取得URLを作れませんでした。`, { symbol: sym, message: e.message });
        }
      });
      batchNo++;
      let responses = null;
      let parseErrCount = 0;
      let fetchedRowsInBatch = 0;
      let fetchedSymbolsInBatch = 0;
      let batchFirstTs = null;
      let batchLastTs = null;
      const noRowsSymbols = [];
      const sparseResultSamples = [];
      try {
        responses = UrlFetchApp.fetchAll(requests);
      } catch (e) {
        if (isQuotaExhausted_(e)) {
          writeProcessLog_("OHLCV", `PHASE1: Google側の実行制限に当たったため、バッチ ${batchNo} から4時間後に再開します。`, { batchNo: batchNo, message: e.message });
          debugLogToSheet_("ohlcv.phase.error", { phase: "PHASE1", stage: "quota_exhausted", batchNo: batchNo, message: e.message });
          if (allFetchedData.length > 0) appendRowsToSheet_(ohlcvSheet, allFetchedData);
          props.setProperty(STATE_KEY, i.toString());
          props.setProperty(NEW_ALERT_COUNT_KEY, newAlertCount.toString());
          props.setProperty("LAST_TS_MAP", JSON.stringify(lastTsMap));
          deleteTriggersByHandler_("resumeOHLCVFetch");
          ScriptApp.newTrigger("resumeOHLCVFetch").timeBased().after(4 * 60 * 60 * 1000).create();
          return;
        }
        writeProcessLog_("OHLCV", `PHASE1: バッチ ${batchNo} のOHLCV取得に失敗しました。次の処理に進みます。`, { batchNo: batchNo, message: e.message });
        debugLogToSheet_("ohlcv.phase.error", { phase: "PHASE1", stage: "batch_fetch", batchNo: batchNo, message: e.message });
      }
      if (responses && responses.length > 0) {
        reqMeta.forEach(meta => {
          try {
            const intraRes = responses[meta.intraIdx];
            const result = parseIntraResponse_(meta.symbol, intraRes, null);
            const validRows = applyPhase1ResultRows_(meta.symbol, result, refreshId, lastTsMap);
            if (validRows.length > 0) {
              allFetchedData = allFetchedData.concat(validRows);
              fetchedRowsInBatch += validRows.length;
              fetchedSymbolsInBatch++;
              validRows.forEach(row => {
                const ts = row && row.length ? row[0] : null;
                if (!(ts instanceof Date)) return;
                if (!batchFirstTs || ts.getTime() < batchFirstTs.getTime()) batchFirstTs = ts;
                if (!batchLastTs || ts.getTime() > batchLastTs.getTime()) batchLastTs = ts;
              });
              if (validRows.length < 2 && sparseResultSamples.length < 5) {
                sparseResultSamples.push({ symbol: meta.symbol, rows: validRows.length });
              }
              newAlertCount++;
            } else if (noRowsSymbols.length < 5) {
              noRowsSymbols.push(meta.symbol);
            }
          } catch (e) {
            parseErrCount++;
            writeProcessLog_("OHLCV", `PHASE1: ${meta.symbol} のYahoo FinanceレスポンスをOHLCV行に変換できませんでした。`, { symbol: meta.symbol, message: e.message });
          }
        });
      }
      debugLogToSheet_("ohlcv.phase1.batch", {
        batchNo: batchNo,
        startIndex: i,
        symbolsInBatch: batch.length,
        requestCount: requests.length,
        fetchedRows: fetchedRowsInBatch,
        symbolsWithRows: fetchedSymbolsInBatch,
        symbolsWithoutRows: reqMeta.length - fetchedSymbolsInBatch,
        noRowsSamples: noRowsSymbols,
        sparseResultSamples: sparseResultSamples,
        dateRange: batchFirstTs && batchLastTs ? `${batchFirstTs.toISOString()} ~ ${batchLastTs.toISOString()}` : "none",
        parseErrCount: parseErrCount,
        elapsedMs: Date.now() - batchStart
      });
      Utilities.sleep(OHLCV_BATCH_SLEEP_MS);
    }

    if (allFetchedData.length > 0) appendRowsToSheet_(ohlcvSheet, allFetchedData);
    
    // ★★★ 追加: 修復対象銘柄リストをクリア ★★★
    if (repairSymsSet.size > 0) {
      props.deleteProperty("OHLCV_REPAIR_SYMBOLS");
      writeProcessLog_("OHLCV", "PHASE1: 修復対象銘柄リストをクリアしました。", { repairSymbols: repairSymsSet.size });
    }
    // ★★★ 追加ここまで ★★★

    props.setProperties({ [PHASE_KEY]: "PHASE2", [STATE_KEY]: "5", [NEW_ALERT_COUNT_KEY]: newAlertCount.toString(), "LAST_TS_MAP": JSON.stringify(lastTsMap) });
    writeProcessLog_("OHLCV", "PHASE1: OHLCV取得が完了しました。次は株式分割チェック（PHASE2）へ進みます。", {
      totalSymbols: symbols.length,
      fetchedRows: allFetchedData.length,
      newAlertCount: newAlertCount
    });
    debugLogToSheet_("ohlcv.phase.summary", {
      event: "phase_complete",
      phase: "PHASE1",
      totalSymbols: symbols.length,
      fetchedRows: allFetchedData.length,
      newAlertCount: newAlertCount,
      elapsedMs: Date.now() - startTime
    });
    setupResumeTrigger_("resumeOHLCVFetch");
    return;
  }

  if (phase === "PHASE2") {
    let cursor = Math.max(5, parseInt(props.getProperty(STATE_KEY) || "5"));
    const persistedSplitCache = JSON.parse(props.getProperty("OHLCV_SPLIT_CACHE") || "{}");
    const lastRow = rawSheet.getLastRow();
    if (lastRow < cursor) {
      props.setProperties({ [PHASE_KEY]: "PHASE3", [STATE_KEY]: "0" });
      debugLogToSheet_("ohlcv.phase.summary", {
        event: "phase_complete",
        phase: "PHASE2",
        elapsedMs: Date.now() - startTime
      });
      setupResumeTrigger_("resumeOHLCVFetch");
      return;
    }
    let chunkNo = 0;
    while (cursor <= lastRow) {
      const elapsedBeforeChunk = Date.now() - startTime;
      if (elapsedBeforeChunk > LIMIT_MS || (LIMIT_MS - elapsedBeforeChunk) < FETCH_GUARD_MS) {
        props.setProperty(STATE_KEY, cursor.toString());
        debugLogToSheet_("ohlcv.phase.summary", {
          event: "phase_pause",
          phase: "PHASE2",
          cursor: cursor,
          elapsedMs: elapsedBeforeChunk
        });
        setupResumeTrigger_("resumeOHLCVFetch");
        return;
      }
      const chunkStartRow = cursor;
      const chunkEndRow = Math.min(lastRow, chunkStartRow + PHASE2_ROW_CHUNK_SIZE - 1);
      const chunkSize = chunkEndRow - chunkStartRow + 1;
      const rawData = rawSheet.getRange(chunkStartRow, 1, chunkSize, RAW_HEADERS.length).getValues();
      const splitCache = {};
      const seen = {};
      rawData.forEach(r => {
        const sym = String(r[RAW_IDX.symbol_code] || "").trim();
        if (sym && !seen[sym]) {
          seen[sym] = true;
          if (persistedSplitCache[sym]) splitCache[sym] = expandSplitInfo_(persistedSplitCache[sym]);
        }
      });
      const uniqueSymbols = Object.keys(seen).filter(sym => !splitCache[sym]);
      let fetchBatches = 0;
      const fetchStart = Date.now();
      let timedOutDuringSplitFetch = false;
      for (let si = 0; si < uniqueSymbols.length; si += PHASE2_SPLIT_BATCH_SIZE) {
        const fetchElapsed = Date.now() - startTime;
        if (fetchElapsed > LIMIT_MS || (LIMIT_MS - fetchElapsed) < FETCH_GUARD_MS) {
          timedOutDuringSplitFetch = true;
          break;
        }
        const symBatch = uniqueSymbols.slice(si, Math.min(si + PHASE2_SPLIT_BATCH_SIZE, uniqueSymbols.length));
        const requests = symBatch.map(sym => {
          return {
            url: buildYahooFinanceDataUrl_(sym),
            muteHttpExceptions: true,
            headers: { "User-Agent": "Mozilla/5.0" }
          };
        });
        let responses = null;
        try {
          responses = UrlFetchApp.fetchAll(requests);
          fetchBatches++;
        } catch (e) {
          if (isQuotaExhausted_(e)) {
            writeProcessLog_("OHLCV", `PHASE2: 株式分割情報の取得中にGoogle側の実行制限へ当たりました。行 ${chunkStartRow} から4時間後に再開します。`, {
              rowStart: chunkStartRow,
              batchNo: fetchBatches + 1,
              message: e.message
            });
            debugLogToSheet_("ohlcv.phase.error", { phase: "PHASE2", stage: "quota_exhausted", rowStart: chunkStartRow, batchNo: fetchBatches + 1, message: e.message });
            props.setProperty(STATE_KEY, chunkStartRow.toString());
            deleteTriggersByHandler_("resumeOHLCVFetch");
            ScriptApp.newTrigger("resumeOHLCVFetch").timeBased().after(4 * 60 * 60 * 1000).create();
            return;
          }
          writeProcessLog_("OHLCV", `PHASE2: 行 ${chunkStartRow} からの株式分割情報バッチ取得に失敗しました。`, {
            rowStart: chunkStartRow,
            batchNo: fetchBatches + 1,
            message: e.message
          });
          debugLogToSheet_("ohlcv.phase.error", {
            phase: "PHASE2",
            stage: "split_batch_fetch",
            rowStart: chunkStartRow,
            batchNo: fetchBatches + 1,
            message: e.message
          });
        }
        if (responses) {
          symBatch.forEach((sym, idx) => {
            const info = parseYahooFinanceDataResponse_(sym, responses[idx]);
            splitCache[sym] = info;
            persistedSplitCache[sym] = minifySplitInfo_(info);
          });
          safeSetSplitCache_(props, persistedSplitCache, Object.keys(seen));
        }
        Utilities.sleep(PHASE2_SPLIT_BATCH_SLEEP_MS);
      }
      if (timedOutDuringSplitFetch) {
        props.setProperty(STATE_KEY, chunkStartRow.toString());
        debugLogToSheet_("ohlcv.phase.summary", {
          event: "phase_pause",
          phase: "PHASE2",
          cursor: chunkStartRow,
          reason: "split_fetch_guard",
          elapsedMs: Date.now() - startTime
        });
        setupResumeTrigger_("resumeOHLCVFetch");
        return;
      }
      const modifiedRows = [];
      let adjustedRows = 0;
      const adjustedSymbolMap = {};
      rawData.forEach((row, idx) => {
        const symbol = String(row[RAW_IDX.symbol_code] || "").trim();
        if (!symbol) return;
        const splitInfo = splitCache[symbol];
        if (!splitInfo) return;
        if (applySplitAdjustToRawRow_(row, splitInfo)) {
          addSplitQueueItem_(props, symbol, splitInfo.splitRatio, splitInfo.splitDate);
          modifiedRows.push(idx);
          adjustedRows++;
          const splitKey = `${symbol}|${splitInfo.splitDate}|${splitInfo.splitRatio}`;
          if (!adjustedSymbolMap[splitKey]) {
            adjustedSymbolMap[splitKey] = {
              symbol: symbol,
              splitDate: splitInfo.splitDate,
              splitDateKey: splitInfo.splitDate ? Utilities.formatDate(new Date(splitInfo.splitDate * 1000), CONFIG.TIMEZONE, "yyyy-MM-dd") : "",
              ratio: splitInfo.splitRatio,
              adjustedRows: 0
            };
          }
          adjustedSymbolMap[splitKey].adjustedRows++;
        }
      });
      modifiedRows.forEach(idx => {
        rawSheet.getRange(chunkStartRow + idx, 1, 1, RAW_HEADERS.length).setValues([rawData[idx]]);
      });
      const nextCursor = chunkEndRow + 1;
      props.setProperty(STATE_KEY, nextCursor.toString());
      chunkNo++;
      const adjustedSymbols = Object.keys(adjustedSymbolMap).sort().map(key => adjustedSymbolMap[key]);
      debugLogToSheet_("ohlcv.phase2.chunk", {
        chunkNo: chunkNo,
        rowStart: chunkStartRow,
        rowEnd: chunkEndRow,
        uniqueSymbols: uniqueSymbols.length,
        splitFetchBatches: fetchBatches,
        splitFetchMs: Date.now() - fetchStart,
        adjustedRows: adjustedRows,
        adjustedSymbolCount: adjustedSymbols.length,
        adjustedSymbols: adjustedSymbols,
        nextCursor: nextCursor,
        elapsedMs: Date.now() - startTime
      });
      cursor = nextCursor;
    }
    props.deleteProperty("OHLCV_SPLIT_CACHE");
    props.setProperties({ [PHASE_KEY]: "PHASE3", [STATE_KEY]: "0" });
    debugLogToSheet_("ohlcv.phase.summary", {
      event: "phase_complete",
      phase: "PHASE2",
      elapsedMs: Date.now() - startTime
    });
    setupResumeTrigger_("resumeOHLCVFetch");
    return;
  }

  if (phase === "PHASE3") {
    const isDone = processSplitQueueBatch_(startTime, LIMIT_MS);
    if (!isDone) {
      writeProcessLog_("OHLCV", "PHASE3: 分割調整キューの適用が途中です。進捗を保存して再開します。", {});
      setupResumeTrigger_("resumeOHLCVFetch");
      return;
    }
    writeProcessLog_("OHLCV", "PHASE3: 分割調整キューの適用が完了しました。次は重複排除と並べ替え（PHASE4）です。", {});
    props.setProperties({ [PHASE_KEY]: "PHASE4", [STATE_KEY]: "0" });
    setupResumeTrigger_("resumeOHLCVFetch");
    return;
  }

// fetchOHLCVForNewAlerts 内の PHASE4 ブロック（1150行目付近）
if (phase === "PHASE4") {
  writeProcessLog_("OHLCV", "PHASE4: OHLCVシートの重複行を削除し、timestamp昇順に並べ替えます。", {});
  const lastRow = ohlcvSheet.getLastRow();
  debugLogToSheet_("ohlcv.phase4.start", { lastRow: lastRow, elapsedMs: Date.now() - startTime });
  if (lastRow > 1) {
    // ソート: 銘柄(3列目)順 -> 日時(1列目)順 -> alert_id(8列目)降順
    debugLogToSheet_("ohlcv.phase4.sort_before_dedupe.start", { rows: lastRow - 1 });
    ohlcvSheet.getRange(2, 1, lastRow - 1, 8).sort([{column:3, ascending:true}, {column:1, ascending:true}, {column:8, ascending:false}]);
    // 重複削除: 日時(1) と 銘柄(3) が一致する行を削除
    debugLogToSheet_("ohlcv.phase4.remove_duplicates.start", { rows: lastRow - 1 });
    ohlcvSheet.getRange(2, 1, lastRow - 1, 8).removeDuplicates([1,3]);
    const newLastRow = ohlcvSheet.getLastRow();
    debugLogToSheet_("ohlcv.phase4.sort_final.start", { rows: Math.max(0, newLastRow - 1) });
    if (newLastRow > 1) ohlcvSheet.getRange(2, 1, newLastRow - 1, 8).sort([{column:1, ascending:true}, {column:3, ascending:true}]);
  }
  const newCount = props.getProperty(NEW_ALERT_COUNT_KEY) || "0";
  resetProcessProperties_();
  writeProcessLog_("OHLCV", "OHLCV取得フローが完了しました。続いて日次メンテナンスを1分後に起動します。", { newRowsOrSymbols: newCount, refreshId: refreshId });
  debugLogToSheet_("ohlcv.phase4.complete", { newCount: newCount, elapsedMs: Date.now() - startTime });

  // ★ 変更：runDailyMaintenance を直接呼ばず、1分後のトリガーを設定
  props.setProperty(DAILY_MAINT_NEW_COUNT_KEY, newCount);
  props.setProperty(DAILY_MAINT_REFRESH_ID_KEY, refreshId);
  deleteTriggersByHandler_("runDailyMaintenanceTrigger");
  ScriptApp.newTrigger("runDailyMaintenanceTrigger")
    .timeBased()
    .after(60 * 1000) // 1分後
    .create();
  writeProcessLog_("OHLCV", "日次メンテナンスのワンショットトリガーを1分後に設定しました。", { handlerName: "runDailyMaintenanceTrigger" });
}
}

function processSplitQueueBatch_(startTime, limitMs) {
  const props = PropertiesService.getScriptProperties();
  let queue = JSON.parse(props.getProperty("SPLIT_QUEUE") || "[]");
  if (queue.length === 0) return true;
  const ohlcvSheet = openAppSpreadsheet_().getSheetByName(CONFIG.OHLCV_SHEET);
  const lastRow = ohlcvSheet.getLastRow();
  if (lastRow < 2) return true;

  while (queue.length > 0) {
    const target = queue[0];
    let startIndex = parseInt(props.getProperty("SPLIT_INDEX") || "2");
    if (!Number.isFinite(startIndex) || startIndex < 2) startIndex = 2;

    const matches = ohlcvSheet
      .getRange(2, 3, lastRow - 1, 1)
      .createTextFinder(String(target.symbol || ""))
      .matchEntireCell(true)
      .findAll();
    const rowNums = matches
      .map(r => r.getRow())
      .filter(rowNum => rowNum >= startIndex)
      .sort((a, b) => a - b);

    debugLogToSheet_("ohlcv.phase3.target", {
      symbol: target.symbol,
      splitDate: target.date,
      ratio: target.ratio,
      startIndex: startIndex,
      matchedRows: rowNums.length,
      lastRow: lastRow
    });

    let updatedRows = 0;
    for (let idx = 0; idx < rowNums.length; idx++) {
      const elapsed = Date.now() - startTime;
      if (elapsed > limitMs - FETCH_GUARD_MS) {
        props.setProperty("SPLIT_INDEX", String(startIndex));
        debugLogToSheet_("ohlcv.phase3.pause", {
          symbol: target.symbol,
          splitIndex: startIndex,
          lastRow: lastRow,
          elapsedMs: elapsed
        });
        return false;
      }

      const rowNum = rowNums[idx];
      const row = ohlcvSheet.getRange(rowNum, 1, 1, 8).getValues()[0];
      const ts = row[0];
      if ((ts instanceof Date) && (ts.getTime() / 1000 < target.date)) {
        for (let c = 3; c <= 6; c++) if (Number.isFinite(row[c])) row[c] = Math.round((row[c] / target.ratio) * 100) / 100;
        if (Number.isFinite(row[7])) row[7] = Math.round(row[7] * target.ratio);
        ohlcvSheet.getRange(rowNum, 4, 1, 5).setValues([row.slice(3, 8)]);
        updatedRows++;
      }
      startIndex = rowNum + 1;
      props.setProperty("SPLIT_INDEX", String(startIndex));
    }

    debugLogToSheet_("ohlcv.phase3.targetComplete", {
      symbol: target.symbol,
      matchedRows: rowNums.length,
      updatedRows: updatedRows,
      nextQueue: Math.max(0, queue.length - 1)
    });
    queue.shift();
    props.setProperty("SPLIT_QUEUE", JSON.stringify(queue));
    props.setProperty("SPLIT_INDEX", "2");
  }

  return true;
}

}
function isValidOhlcvTimestamp_(ts) {
  const MIN_VALID_DATE = new Date("1990-01-01T00:00:00Z");
  return ts instanceof Date && !isNaN(ts.getTime()) && ts >= MIN_VALID_DATE;
}

function normalizeOhlcvTimestamp_(value) {
  if (isValidOhlcvTimestamp_(value)) return value;
  if (typeof value === "number" && Number.isFinite(value)) {
    const ms = value > 1000000000000 ? value : value * 1000;
    const d = new Date(ms);
    return isValidOhlcvTimestamp_(d) ? d : null;
  }
  if (typeof value === "string" && value.trim()) {
    const raw = value.trim();
    const m = raw.replace(/-/g, "/").match(/^(\d{4})\/(\d{1,2})\/(\d{1,2})(?:\s+(\d{1,2}):(\d{2})(?::(\d{2}))?)?/);
    if (m) {
      const iso = m[1] + "-" + String(m[2]).padStart(2, "0") + "-" + String(m[3]).padStart(2, "0") +
        "T" + String(m[4] || "0").padStart(2, "0") + ":" + String(m[5] || "0").padStart(2, "0") + ":" + String(m[6] || "0").padStart(2, "0") + "+09:00";
      const jst = new Date(iso);
      if (isValidOhlcvTimestamp_(jst)) return jst;
    }
    const d = new Date(raw);
    if (isValidOhlcvTimestamp_(d)) return d;
  }
  return null;
}

function normalizeOhlcvRowForWrite_(row) {
  if (!row || !row.length) return null;
  const out = row.slice(0, 8);
  while (out.length < 8) out.push("");
  const ts = normalizeOhlcvTimestamp_(out[0]);
  if (!ts) return null;
  if (!isOhlcvSessionTimestamp_(ts)) return null;
  out[0] = ts;
  return out;
}

function appendRowsToSheet_(sheet, data) {
  if (!data || data.length === 0) return 0;
  const validData = [];
  const skippedSamples = [];
  data.forEach((row, index) => {
    const normalized = normalizeOhlcvRowForWrite_(row);
    if (normalized) {
      validData.push(normalized);
    } else if (skippedSamples.length < 5) {
      skippedSamples.push({ index: index, timestamp: row && row.length ? String(row[0]) : "", symbol: row && row.length > 2 ? row[2] : "" });
    }
  });
  const skipped = data.length - validData.length;
  if (skipped > 0) {
    debugLogToSheet_("appendRowsToSheet_.skippedInvalidTs", {
      total: data.length, valid: validData.length, skipped: skipped, samples: skippedSamples
    });
  }
  if (validData.length === 0) return 0;
  const startRow = sheet.getLastRow() + 1;
  sheet.getRange(startRow, 1, validData.length, 8).setValues(validData);
  sheet.getRange(startRow, 1, validData.length, 1).setNumberFormat("yyyy/mm/dd hh:mm");
  return validData.length;
}

function scheduleOhlcvResume_(handlerName, delayMs) {
  deleteTriggersByHandler_(handlerName);
  
  const maxRetries = 3;
  const triggerDelayMs = Number.isFinite(delayMs) ? delayMs : 60 * 1000;
  
  for (let i = 0; i < maxRetries; i++) {
    try {
      ScriptApp.newTrigger(handlerName)
        .timeBased()
        .after(triggerDelayMs)
        .create();
      writeProcessLog_("OHLCV", `${handlerName} を ${Math.round(triggerDelayMs / 1000)} 秒後に実行する再開トリガーとして設定しました。`, {
        handlerName: handlerName,
        delayMs: triggerDelayMs,
        attempt: i + 1
      });
      traceOhlcv_("resume.scheduled", {
        handlerName: handlerName,
        delayMs: triggerDelayMs,
        attempt: i + 1
      });
      debugLogToSheet_("ohlcv.resume.scheduled", {
        handlerName: handlerName,
        delayMs: triggerDelayMs,
        attempt: i + 1
      });
      return;
    } catch (e) {
      if (i === maxRetries - 1) {
        writeProcessLog_("OHLCV", `再開トリガーを ${maxRetries} 回試しましたが作成できませんでした。`, { handlerName: handlerName, message: e.message });
        throw e;
      }
      writeProcessLog_("OHLCV", `再開トリガー作成に失敗しました。少し待って再試行します。`, {
        handlerName: handlerName,
        attempt: i + 1,
        maxRetries: maxRetries,
        message: e.message
      });
      Utilities.sleep(2000 * (i + 1));
    }
  }
}
/**
 * 次回の実行を予約する関数の堅牢化版
 * トリガー作成自体が Google サーバーエラーで失敗した場合にリトライします。
 */
function setupResumeTrigger_(handlerName) {
  scheduleOhlcvResume_(handlerName, 60 * 1000);
}

/**
 * 継続実行のエントリポイント
 * fetchOHLCVForNewAlerts 自体がエラーで止まっても、自動リトライを仕掛けます。
 */
function resumeOHLCVFetch() {
  writeProcessLog_("OHLCV", "resumeOHLCVFetchが起動しました。保存済みのフェーズとカーソルからOHLCV取得を再開します。", {
    businessDate: getOhlcvBusinessDateKey_(),
    isManual: !!getManualOhlcvBusinessDateKey_()
  });
  traceOhlcv_("resume.entry", {});
  debugLogToSheet_("ohlcv.resume.entry", getOhlcvRuntimeState_());
  try {
    return fetchOHLCVForNewAlerts();
  } catch (e) {
    const errData = {
      message: e && e.message ? e.message : String(e),
      stack: e && e.stack ? String(e.stack) : "",
      state: getOhlcvRuntimeState_()
    };
    console.error("[OHLCV] resumeOHLCVFetch error " + JSON.stringify(errData));
    debugLogToSheet_("ohlcv.resume.error", errData);
    writeProcessLog_("OHLCV", "再開処理中にエラーが発生しました。1分後に自動リトライします。", { message: errData.message });
    
    // エラー発生時も、stateが保存されていれば setupResumeTrigger_ により
    // 次回の実行（リトライ）がスケジュールされます。
    setupResumeTrigger_("resumeOHLCVFetch");
    return { ok: false, retried: true, message: errData.message };
  }
}

// ★ 追加: URL ペア構築（fetchAll でバッチ実行するための部品）
//    fetchOHLCVRecent の URL 生成ロジックをそのまま抽出した関数
function buildOhlcvRequestPair_(tvSymbol, lastTs, optStartSec, optEndSec) {
  const yahooSymbol = convertToYahooSymbol(tvSymbol);
  let timeParam = "";
  if (optStartSec && optEndSec) {
    timeParam = `period1=${optStartSec}&period2=${optEndSec}`;
  } else if (lastTs) {
    const nowSec = Math.floor(getOhlcvRunEndMillis_() / 1000);
    const lastTsSec = Math.floor(lastTs / 1000);
    const diffDays = (nowSec - lastTsSec) / (24 * 60 * 60);
    const lookbackSec = OHLCV_DEFAULT_LOOKBACK_DAYS * 24 * 60 * 60;
    if (diffDays > 60) {
      const p1 = nowSec - lookbackSec;
      const p2 = nowSec;
      timeParam = `period1=${p1}&period2=${p2}`;
      if (isVerboseOhlcvFetchLogEnabled_()) {
        debugLogToSheet_("fetchOHLCVRecent.fullRefresh", { symbol: tvSymbol, diffDays: diffDays.toFixed(1), lookbackDays: OHLCV_DEFAULT_LOOKBACK_DAYS });
      }
    } else {
      const minBackSec = 7 * 24 * 60 * 60;
      const maxBackSec = lookbackSec;
      let p1 = lastTsSec - minBackSec;
      const p1Limit = nowSec - maxBackSec;
      if (p1 < p1Limit) p1 = p1Limit;
      const p2 = nowSec;
      timeParam = `period1=${p1}&period2=${p2}`;
      if (isVerboseOhlcvFetchLogEnabled_()) {
        debugLogToSheet_("fetchOHLCVRecent.period", { symbol: tvSymbol, lastTs: new Date(lastTs).toISOString(), p1: new Date(p1*1000).toISOString(), p2: new Date(p2*1000).toISOString(), lookbackDays: OHLCV_DEFAULT_LOOKBACK_DAYS });
      }
    }
  } else {
    const p2 = Math.floor(getOhlcvRunEndMillis_() / 1000);
    const p1 = p2 - OHLCV_DEFAULT_LOOKBACK_DAYS * 24 * 60 * 60;
    timeParam = `period1=${p1}&period2=${p2}`;
    if (isVerboseOhlcvFetchLogEnabled_()) {
      debugLogToSheet_("fetchOHLCVRecent.period", { symbol: tvSymbol, p1: new Date(p1*1000).toISOString(), p2: new Date(p2*1000).toISOString(), lookbackDays: OHLCV_DEFAULT_LOOKBACK_DAYS });
    }
  }
  const dailyUrl = `https://query1.finance.yahoo.com/v8/finance/chart/${yahooSymbol}?interval=1d&${timeParam}`;
  const intraUrl = `https://query1.finance.yahoo.com/v8/finance/chart/${yahooSymbol}?interval=1h&${timeParam}`;
  return { dailyUrl: dailyUrl, intraUrl: intraUrl };
}

// ★ 追加: Daily レスポンスの解析（factorMap を返す）。失敗時は null
//    fetchOHLCVRecent の解析ロジックをそのまま抽出
function parseDailyResponse_(tvSymbol, dailyRes) {
  let dailyJson, dailyResult;
  try {
    if (!dailyRes) { debugLogToSheet_("fetchOHLCVRecent.dailyException", { symbol: tvSymbol, error: "no response" }); return null; }
    if (dailyRes.getResponseCode() !== 200) { debugLogToSheet_("fetchOHLCVRecent.dailyError", { symbol: tvSymbol, status: dailyRes.getResponseCode() }); return null; }
    dailyJson = JSON.parse(dailyRes.getContentText());
    dailyResult = dailyJson.chart.result?.[0];
    if (!dailyResult) { debugLogToSheet_("fetchOHLCVRecent.noDailyResult", { symbol: tvSymbol }); return null; }
  } catch(e) { debugLogToSheet_("fetchOHLCVRecent.dailyException", { symbol: tvSymbol, error: e.message }); return null; }

  const dTs = dailyResult.timestamp;
  const dClose = dailyResult.indicators.quote[0].close;
  const dAdj = dailyResult.indicators.adjclose?.[0]?.adjclose;
  const factorMap = {};
  for (let i = 0; i < dTs.length; i++) {
    if (dClose[i] == null) continue;
    const date = new Date(dTs[i] * 1000);
    const dateKey = Utilities.formatDate(date, "JST", "yyyy-MM-dd");
    const adj = (dAdj && dAdj[i] != null) ? dAdj[i] : dClose[i];
    const factor = dClose[i] ? adj / dClose[i] : 1;
    factorMap[dateKey] = factor;
  }
  return factorMap;
}

// ★ 追加: 1h レスポンスの解析（OHLCV 行配列を返す）
//    fetchOHLCVRecent の解析ロジックをそのまま抽出（GAP_REPAIR 修正込み）
function getYahooIntraSessionName_(hour, minute) {
  // Yahoo の JPX 1h 足は 10:00/11:00/... のように区間末尾側の時刻で返る。
  // 13:00 足は 12:00-13:00 相当として前場（9:00-13:00）に含める。
  if (hour < 9 || hour > 15) return "";
  if (hour < 13 || (hour === 13 && minute === 0)) return "AM";
  return "PM";
}

function parseIntraResponse_(tvSymbol, intraRes, factorMap) {
  let res = intraRes;
  try {
    if (!res) { debugLogToSheet_("fetchOHLCVRecent.intraException", { symbol: tvSymbol, error: "no response" }); return []; }
    if (res.getResponseCode() !== 200) { debugLogToSheet_("fetchOHLCVRecent.intraError", { symbol: tvSymbol, status: res.getResponseCode() }); return []; }
  } catch(e) { debugLogToSheet_("fetchOHLCVRecent.intraException", { symbol: tvSymbol, error: e.message }); return []; }

  const json = JSON.parse(res.getContentText());
  const result = json.chart.result?.[0];
  if (!result || !result.timestamp) { debugLogToSheet_("fetchOHLCVRecent.noIntraResult", { symbol: tvSymbol }); return []; }

  const ts = result.timestamp;
  const q = result.indicators.quote[0];
  const candlesByPeriod = {};
  for (let i = 0; i < ts.length; i++) {
    if (q.open[i] == null) continue;

    // ★ 追加: ts[i] が無効な値（0・負・1990年以前）の場合スキップ
    const MIN_VALID_TS = 631152000; // 1990-01-01 00:00:00 UTC
    if (!ts[i] || ts[i] < MIN_VALID_TS) {
      debugLogToSheet_("fetchOHLCVRecent.invalidTs", {
        symbol: tvSymbol, tsValue: ts[i], index: i
      });
      continue;
    }

    const dateObj = new Date(ts[i] * 1000);
    if (isNaN(dateObj.getTime())) {
      debugLogToSheet_("fetchOHLCVRecent.invalidTimestamp", {
        symbol: tvSymbol, tsValue: ts[i], index: i
      });
      continue;
    }
    const dateKey = Utilities.formatDate(dateObj, "JST", "yyyy-MM-dd");
    const hour = parseInt(Utilities.formatDate(dateObj, "JST", "HH"), 10);
    const minute = parseInt(Utilities.formatDate(dateObj, "JST", "mm"), 10);
    const sessionName = getYahooIntraSessionName_(hour, minute);
    if (!sessionName) continue;
    const key = `${dateKey}_${sessionName}`;
    const round = (val) => Math.round(val * 100) / 100;
    // 生の価格をそのまま使用（調整なし）
    const open  = q.open[i];
    const high  = q.high[i];
    const low   = q.low[i];
    const close = q.close[i];
    const volume = q.volume[i];
    const isFlatZeroVolume = (volume === 0 || !Number.isFinite(volume)) && open === high && high === low && low === close;
    const isClosingSnapshot = isFlatZeroVolume && hour === 15 && minute === 30;
    // 15:30 JST の終値スナップショットは PM の close だけ更新する。
    // high/low/volume には混ぜないことで、レンジと出来高の汚染を避ける。
    if (isClosingSnapshot) {
      if (candlesByPeriod[key]) candlesByPeriod[key].close = round(close);
      continue;
    }
    // その他の V=0 かつ OHLC 同値バーは、従来どおりダミーとしてスキップする。
    if (isFlatZeroVolume) {
      continue;
    }
    if (!candlesByPeriod[key]) {
      // ★ 変更: AM は 09:00 JST、PM は 13:00 JST に timestamp を正規化する
      //    こうすると同日・同セッションを後から再 fetch しても timestamp が一致し、
      //    removeDuplicates([timestamp, symbol]) が正しく重複排除できる
      //    注意: setHours は UTC 時刻を操作するため、JST 固定の ISO 文字列で構築する
      const sessionHour = (sessionName === "AM") ? 9 : 13;
      const dateStr = Utilities.formatDate(dateObj, CONFIG.TIMEZONE, "yyyy-MM-dd");
      const normalizedTs = new Date(`${dateStr}T${String(sessionHour).padStart(2, '0')}:00:00+09:00`);
      candlesByPeriod[key] = { timestamp: normalizedTs, symbol: tvSymbol, open: round(open), high: round(high), low: round(low), close: round(close), volume: volume };
    } else {
      const p = candlesByPeriod[key];
      p.high = round(Math.max(p.high, high));
      p.low  = round(Math.min(p.low, low));
      p.close = round(close);
      p.volume += volume;
    }
  }
  const resultCandles = Object.values(candlesByPeriod).sort((a,b) => a.timestamp - b.timestamp).map(p => [p.timestamp, "", p.symbol, p.open, p.high, p.low, p.close, p.volume]);
  if (isVerboseOhlcvFetchLogEnabled_()) {
    debugLogToSheet_("fetchOHLCVRecent.result", { symbol: tvSymbol, candlesCount: resultCandles.length, dateRange: resultCandles.length ? `${resultCandles[0][0].toISOString()} ~ ${resultCandles[resultCandles.length-1][0].toISOString()}` : "none" });
  }
  return resultCandles;
}

// ★ 変更: fetchOHLCVRecent は URL構築 → 1h fetch → 解析の薄いラッパー
//    既存シグネチャは互換。1h取得成功時の生成行は従来同等。
function fetchOHLCVRecent(tvSymbol, lastTs, optStartSec, optEndSec) {
  const pair = buildOhlcvRequestPair_(tvSymbol, lastTs, optStartSec, optEndSec);
  let intraRes;
  try {
    intraRes = UrlFetchApp.fetch(pair.intraUrl, { muteHttpExceptions: true, headers: { "User-Agent": "Mozilla/5.0" } });
  } catch(e) { debugLogToSheet_("fetchOHLCVRecent.intraException", { symbol: tvSymbol, error: e.message }); return []; }
  return parseIntraResponse_(tvSymbol, intraRes, null);
}

function convertToYahooSymbol(tvSymbol) {
  let code = String(tvSymbol || "").trim();
  if (!code) return "";
  if (code.startsWith("TYO:")) return code.replace("TYO:", "") + ".T";
  if (/^\d{4}$/.test(code) || /^\d{3}[A-Z\d]$/.test(code)) return code + ".T";
  return code;
}

function safeSetSplitCache_(props, obj, keepKeys) {
  const json = JSON.stringify(obj);
  if (json.length <= 8192) {
    props.setProperty("OHLCV_SPLIT_CACHE", json);
  } else {
    const keep = {};
    (keepKeys || []).forEach(key => {
      if (Object.prototype.hasOwnProperty.call(obj, key)) keep[key] = obj[key];
    });
    const keepJson = JSON.stringify(keep);
    if (keepJson.length <= 8192 && keepKeys && keepKeys.length > 0) {
      console.log(`[OHLCV_SPLIT_CACHE] ${json.length}B > 8192B - current chunk cache only (${keepJson.length}B)`);
      props.setProperty("OHLCV_SPLIT_CACHE", keepJson);
    } else {
      console.log(`[OHLCV_SPLIT_CACHE] ${json.length}B > 8192B - cache dropped`);
      props.deleteProperty("OHLCV_SPLIT_CACHE");
    }
  }
}

// ★ 追加: 分割情報 API の URL 構築（fetchAll でバッチ実行するための部品）
function buildYahooFinanceDataUrl_(symbolCode) {
  const code = String(symbolCode || "").trim().toUpperCase();
  if (!code) return null;
  const yahooSymbol = code.endsWith(".T") ? code : code + ".T";
  return `https://query1.finance.yahoo.com/v8/finance/chart/${yahooSymbol}?interval=1d&range=3mo&events=splits`;
}

// ★ 追加: 分割情報レスポンスの解析
//    fetchYahooFinanceDataAPI_ の解析ロジックをそのまま抽出
function parseYahooFinanceDataResponse_(symbolCode, res) {
  const code = String(symbolCode || "").trim().toUpperCase();
  if (!code) return { price: null, splitRatio: null, splitDate: null, error: "NO_CODE" };
  try {
    if (!res) return { price: null, splitRatio: null, splitDate: null, error: "FETCH_ERROR" };
    const responseCode = res.getResponseCode();
    if (responseCode === 404) {
      return { price: null, splitRatio: null, splitDate: null, error: "DELISTED" };
    }
    const json = JSON.parse(res.getContentText());
    const result = json.chart.result?.[0];
    if (!result) return { price: null, splitRatio: null, splitDate: null, error: "NO_DATA" };

    let currentPrice = result.meta?.regularMarketPrice;
    let splitRatio = null;
    let splitDate = null;
    if (result.events && result.events.splits) {
      const splitsArray = Object.values(result.events.splits).sort((a, b) => b.date - a.date);
      if (splitsArray.length > 0) {
        const latestSplit = splitsArray[0];
        splitRatio = latestSplit.numerator / latestSplit.denominator;
        splitDate = latestSplit.date;
      }
    }
    return { price: currentPrice, splitRatio: splitRatio, splitDate: splitDate, error: null };
  } catch (e) {
    return { price: null, splitRatio: null, splitDate: null, error: "FETCH_ERROR" };
  }
}

// ★ 変更: URL 構築 → fetch → 解析の薄いラッパー（既存シグネチャ互換）
function fetchYahooFinanceDataAPI_(symbolCode) {
  const code = String(symbolCode || "").trim().toUpperCase();
  if (!code) return { price: null, splitRatio: null, splitDate: null, error: "NO_CODE" };
  const url = buildYahooFinanceDataUrl_(code);
  try {
    const res = UrlFetchApp.fetch(url, {
      muteHttpExceptions: true,
      headers: { "User-Agent": "Mozilla/5.0" }
    });
    return parseYahooFinanceDataResponse_(code, res);
  } catch (e) {
    return { price: null, splitRatio: null, splitDate: null, error: "FETCH_ERROR" };
  }
}

    console.log("debugFetchOhlcvVerbose: 銘柄コードを指定してください。例: debugFetchOhlcvVerbose(\"2146\")");
    return null;
  }
  const pair = buildOhlcvRequestPair_(normalizedSymbol, null);
  const out = {
    symbol: normalizedSymbol,
    yahooSymbol: convertToYahooSymbol(normalizedSymbol),
    dailyUrl: pair.dailyUrl,
    intraUrl: pair.intraUrl,
    dailyStatus: null,
    intraStatus: null,
    dailyHasResult: false,
    intraHasResult: false,
    intraTimestampCount: 0,
    factorMapSize: 0,
    candleCount: 0
  };
  try {
    const dailyRes = UrlFetchApp.fetch(pair.dailyUrl, { muteHttpExceptions: true });
    out.dailyStatus = dailyRes.getResponseCode();
    const dailyJson = JSON.parse(dailyRes.getContentText());
    out.dailyHasResult = !!dailyJson.chart?.result?.[0];
    const factorMap = parseDailyResponse_(normalizedSymbol, dailyRes);
    out.factorMapSize = factorMap ? Object.keys(factorMap).length : 0;
    const intraRes = UrlFetchApp.fetch(pair.intraUrl, { muteHttpExceptions: true, headers: { "User-Agent": "Mozilla/5.0" } });
    out.intraStatus = intraRes.getResponseCode();
    const intraJson = JSON.parse(intraRes.getContentText());
    const intraResult = intraJson.chart?.result?.[0];
    out.intraHasResult = !!intraResult;
    out.intraTimestampCount = intraResult && intraResult.timestamp ? intraResult.timestamp.length : 0;
    out.candleCount = factorMap ? parseIntraResponse_(normalizedSymbol, intraRes, factorMap).length : 0;
  } catch (e) {
    out.error = e && e.message ? e.message : String(e);
  }
  console.log(JSON.stringify(out, null, 2));
  debugLogToSheet_("fetchOHLCVRecent.verbose", out);
  return out;
}

// 過去のFALSEをまとめてTRUEにする補修用関数（1回実行したら削除してOK）
function fixMissingReportedFlags() {
  const app = getAppConfig_();
  const sh = getOrCreateSheet_(app.spreadsheetId, CONFIG.RAW_SHEET, RAW_HEADERS);
  const lastRow = sh.getLastRow();
  if (lastRow < dataStartRow_()) return;
  
  const values = sh.getRange(dataStartRow_(), 1, lastRow - dataStartRow_() + 1, RAW_HEADERS.length).getValues();
  let updated = 0;

  values.forEach((row, i) => {
    if (String(row[RAW_IDX.signal_type] || '') !== 'BOTTOM') return;
    let dirty = false;
    CHECKPOINTS.forEach(cp => {
      const perf = row[RAW_IDX[cp.perfKey]];
      const reported = row[RAW_IDX[cp.reportedKey]];
      // perfが入っていてreportedがFALSEなら修正
      if (reported !== true && String(reported).toUpperCase() !== 'TRUE') {
        const perfStr = String(perf || '');
        const perfNum = Number(perfStr.replace(/%/g, '').trim());
        if (perfStr !== '' && Number.isFinite(perfNum)) {
          row[RAW_IDX[cp.reportedKey]] = true;
          dirty = true;
        }
      }
    });
    if (dirty) {
      sh.getRange(dataStartRow_() + i, 1, 1, RAW_HEADERS.length).setValues([row]);
      updated++;
    }
  });
  console.log(`補修完了: ${updated}行をTRUEに更新`);
}

// ─────────────────────────────────────────────────────────────
// 【修正3】repairEmptyTimestampRows（書き直し版）
//
// 使い方:
//   repairEmptyTimestampRows()          // 本番実行
//   repairEmptyTimestampRows(true)      // DryRun（書き込みなし）
//
// 処理内容:
//   1. ohlcv_4h シートを全読み込み
//   2. タイムスタンプ空欄（epoch含む）の行を検出
//   3. DryRunでは削除候補と対象銘柄を返す
//   4. 本番では空timestamp行を削除し、対象銘柄を OHLCV_REPAIR_SYMBOLS に記録
//   5. 日付推定による書き換えは行わず、次回の正規取得で補填する
// ─────────────────────────────────────────────────────────────
function repairEmptyTimestampRows(dryRun) {
  dryRun = dryRun === true;
  const ss = SpreadsheetApp.openById(
    PropertiesService.getScriptProperties().getProperty("SPREADSHEET_ID")
  );
  const sh = ss.getSheetByName(CONFIG.OHLCV_SHEET);
  if (!sh) { console.log("ERROR: ohlcv_4h が見つかりません"); return; }
 
  const DATA_ROW  = 2;  // ohlcv_4h は1行目ヘッダー、2行目からデータ
  const NUM_COLS  = 8;
  const lastRow   = sh.getLastRow();
  if (lastRow < DATA_ROW) { console.log("データなし"); return; }
 
  console.log(`[repair] 開始 lastRow=${lastRow} dryRun=${dryRun}`);
 
  const numDataRows = lastRow - DATA_ROW + 1;
  const values = sh.getRange(DATA_ROW, 1, numDataRows, NUM_COLS).getValues();
 
  const deleteRowNums = [];
  const repairSymbolSet = {};
  const samples = [];
  for (let i = 0; i < values.length; i++) {
    const ts = values[i][0];
    if (isValidOhlcvTimestamp_(ts)) continue;
    const sheetRow = DATA_ROW + i;
    const sym = String(values[i][2] || "").trim();
    deleteRowNums.push(sheetRow);
    if (sym) repairSymbolSet[sym] = true;
    if (samples.length < 20) {
      samples.push({
        sheetRow: sheetRow,
        timestamp: ts == null ? "" : String(ts),
        symbol: sym,
        alertId: String(values[i][1] || ""),
        open: values[i][3],
        close: values[i][6]
      });
    }
  }

  const repairSymbols = Object.keys(repairSymbolSet).sort();
  const summary = {
    ok: true,
    dryRun: dryRun,
    lastRow: lastRow,
    invalidTimestampRows: deleteRowNums.length,
    deleteCandidates: deleteRowNums.length,
    repairSymbols: repairSymbols.length,
    sampleSymbols: repairSymbols.slice(0, 20),
    samples: samples
  };
  console.log(`[repair] 空/無効timestamp行: ${deleteRowNums.length}, 対象銘柄: ${repairSymbols.length}`);
  if (deleteRowNums.length === 0) {
    console.log("[repair] 修正不要");
    return summary;
  }

  if (dryRun) {
    console.log("[repair] DryRun: 削除と再取得キュー登録をスキップ");
    console.log(`[repair] DryRun summary: ${JSON.stringify(summary)}`);
    return summary;
  }

  if (deleteRowNums.length > 0) {
    const uniqueDeleteRows = [...new Set(deleteRowNums)];
    deleteRowsByRowNums_(sh, uniqueDeleteRows);
    SpreadsheetApp.flush();
    console.log(`[repair] ${uniqueDeleteRows.length}行を削除しました`);
    summary.deletedRows = uniqueDeleteRows.length;
  }
 
  const newLastRow = sh.getLastRow();
  if (newLastRow > DATA_ROW) {
    sh.getRange(DATA_ROW, 1, newLastRow - DATA_ROW + 1, NUM_COLS)
      .sort([{ column: 1, ascending: true }, { column: 3, ascending: true }]);
    console.log("[repair] ソート完了");
  }

  if (repairSymbols.length > 0) {
    setOhlcvRepairSymbols_(repairSymbols);
    console.log(`[repair] ${repairSymbols.length}銘柄を OHLCV_REPAIR_SYMBOLS に記録しました`);
  }
 
  console.log(`[repair] 完了`);
  return summary;
}

function diagOneSessionDays(options) {
  options = options || {};
  const daysBack  = options.daysBack  || 14;  // 直近何日間を調べるか
  const minSessions = options.minSessions || 2; // 通常何セッションあるべきか
 
  const ss = SpreadsheetApp.openById(
    PropertiesService.getScriptProperties().getProperty("SPREADSHEET_ID")
  );
  const sh = ss.getSheetByName(CONFIG.OHLCV_SHEET);
  if (!sh) { console.log("ohlcv_4h が見つかりません"); return; }
 
  const now = new Date();
  const cutoff = new Date(now.getTime() - daysBack * 24 * 60 * 60 * 1000);
  const cutoffStr = Utilities.formatDate(cutoff, CONFIG.TIMEZONE, "yyyy/MM/dd");
  const lastRow = sh.getLastRow();
  if (lastRow < 2) return;
 
  const values = sh.getRange(2, 1, lastRow - 1, 3).getValues();
 
  // 銘柄×日付 の行数をカウント
  const countMap = {};
  for (const row of values) {
    const ts = row[0], sym = String(row[2] || "").trim();
    if (!(ts instanceof Date) || !sym) continue;
    const dateStr = Utilities.formatDate(ts, CONFIG.TIMEZONE, "yyyy/MM/dd");
    if (dateStr < cutoffStr) continue;
    const key = sym + "|" + dateStr;
    countMap[key] = (countMap[key] || 0) + 1;
  }
 
  // 1セッションのみの銘柄を抽出
  const missing = {};
  for (const key of Object.keys(countMap)) {
    if (countMap[key] < minSessions) {
      const [sym, date] = key.split("|");
      if (!missing[sym]) missing[sym] = [];
      missing[sym].push(date);
    }
  }
 
  const symList = Object.keys(missing).sort();
  console.log(`[diagOneSessionDays] 直近${daysBack}日で片方セッション欠落: ${symList.length}銘柄`);
  symList.forEach(sym => console.log(`  ${sym}: ${missing[sym].join(", ")}`));
  return { count: symList.length, details: missing };
}

function refetchSymbolRange(options) {
  options = options || {};
  const daysBack = options.daysBack || 21;
 
  const ss = SpreadsheetApp.openById(
    PropertiesService.getScriptProperties().getProperty("SPREADSHEET_ID")
  );
  const rawSh  = ss.getSheetByName(CONFIG.RAW_SHEET);
  const ohlcvSh = ss.getSheetByName(CONFIG.OHLCV_SHEET);
  if (!rawSh || !ohlcvSh) { console.log("シートが見つかりません"); return; }
 
  // 対象銘柄の決定
  let symbols;
  if (options.symbols && options.symbols.length > 0) {
    symbols = options.symbols;
  } else {
    // alerts_raw の全銘柄
    const rawData = rawSh.getRange(
      CONFIG.DATA_START_ROW, 1,
      Math.max(1, rawSh.getLastRow() - CONFIG.DATA_START_ROW + 1),
      RAW_HEADERS.length
    ).getValues();
    const symSet = new Set();
    rawData.forEach(r => {
      const s = String(r[RAW_IDX.symbol_code] || "").trim();
      if (s) symSet.add(s);
    });
    symbols = Array.from(symSet);
  }
 
  console.log(`[refetchSymbolRange] 対象: ${symbols.length}銘柄 / 直近${daysBack}日`);
 
  const nowSec = Math.floor(Date.now() / 1000);
  const p1 = nowSec - daysBack * 24 * 60 * 60;
  const p2 = nowSec;
 
  const MIN_VALID_DATE = new Date("1990-01-01T00:00:00Z");
  const LIMIT_MS = 4 * 60 * 1000;
  const startMs = Date.now();
 
  let fetchedSyms = 0;
  const allData = [];
 
  for (const sym of symbols) {
    if (Date.now() - startMs > LIMIT_MS) {
      console.log(`[refetchSymbolRange] タイムアウト。${fetchedSyms}/${symbols.length}件完了`);
      break;
    }
 
    try {
      const result = fetchOHLCVRecent(sym, null, p1, p2);
      if (result && result.length > 0) {
        // タイムスタンプ検証
        const valid = result.filter(row => {
          const ts = row[0];
          return ts instanceof Date && !isNaN(ts.getTime()) && ts >= MIN_VALID_DATE;
        });
        if (valid.length > 0) {
          allData.push(...valid);
          fetchedSyms++;
        }
      }
    } catch(e) {
      console.log(`[refetchSymbolRange] ${sym}: エラー ${e.message}`);
    }
 
    Utilities.sleep(300);
  }
 
  console.log(`[refetchSymbolRange] 取得完了: ${fetchedSyms}銘柄 / ${allData.length}行`);
 
  if (allData.length === 0) {
    console.log("[refetchSymbolRange] 追加データなし");
    return { ok: true, fetched: 0 };
  }
 
  // シートに追記
  appendRowsToSheet_(ohlcvSh, allData);
  SpreadsheetApp.flush();
 
  // 重複削除 & ソート
  const newLastRow = ohlcvSh.getLastRow();
  if (newLastRow > 1) {
    ohlcvSh.getRange(2, 1, newLastRow - 1, 8)
      .removeDuplicates([1, 3]);  // timestamp + symbol
    const afterDedup = ohlcvSh.getLastRow();
    if (afterDedup > 1) {
      ohlcvSh.getRange(2, 1, afterDedup - 1, 8)
        .sort([{ column: 1, ascending: true }, { column: 3, ascending: true }]);
    }
    console.log(`[refetchSymbolRange] 重複削除後: ${afterDedup - 1}行`);
  }
 
  console.log("[refetchSymbolRange] 完了");
  return { ok: true, fetched: fetchedSyms, rows: allData.length };
}

/**
 * 単一銘柄の特定期間のOHLCVデータを強制再取得し、シートに反映する。
 * データが取得できた場合は B列に "GAP_REPAIR" を記録。
 * データが取得できなかった場合は、B列に "GAP_FAILED" を記録したダミー行を追加。
 */
function refetchSymbolGap(symbol, startDate, endDate) {
  const ss = SpreadsheetApp.openById(
    PropertiesService.getScriptProperties().getProperty("SPREADSHEET_ID")
  );
  const ohlcvSh = ss.getSheetByName(CONFIG.OHLCV_SHEET);
  if (!ohlcvSh) throw new Error("ohlcv_4h シートが見つかりません");

  // 日付文字列を Date に変換（JST基準）
  const start = parseJstDate_(startDate, false);
  const end   = parseJstDate_(endDate, true);
  if (!isValidOhlcvTimestamp_(start) || !isValidOhlcvTimestamp_(end)) {
    throw new Error("日付の形式が不正です。yyyy/mm/dd 形式で指定してください。");
  }

  // マージンを付与（前後3日）
  const MARGIN_DAYS = 3;
  const fetchStart = new Date(start.getTime() - MARGIN_DAYS * 24 * 60 * 60 * 1000);
  const fetchEnd   = new Date(end.getTime()   + MARGIN_DAYS * 24 * 60 * 60 * 1000);

  const startSec = Math.floor(fetchStart.getTime() / 1000);
  const endSec   = Math.floor(fetchEnd.getTime() / 1000);

  console.log(`[refetchSymbolGap] ${symbol} : 指定=${Utilities.formatDate(start, "JST", "MM/dd")}～${Utilities.formatDate(end, "JST", "MM/dd")} / 取得範囲=${Utilities.formatDate(fetchStart, "JST", "MM/dd")}～${Utilities.formatDate(fetchEnd, "JST", "MM/dd")}`);

  // fetchOHLCVRecent を直接呼び出し
  const result = fetchOHLCVRecent(symbol, null, startSec, endSec);
  
  // 有効なデータ行を抽出（対象日以外の前後マージン分は成功扱いにしない）
  const MIN_VALID_DATE = new Date("1990-01-01T00:00:00Z");
  const targetDateSet = buildDateSetBetween_(start, end);
  const validRows = result ? result.filter(row => {
    const ts = row[0];
    if (!(ts instanceof Date) || isNaN(ts.getTime()) || ts < MIN_VALID_DATE) return false;
    const dateStr = Utilities.formatDate(ts, CONFIG.TIMEZONE, "yyyy/MM/dd");
    return !!targetDateSet[dateStr];
  }) : [];

  if (validRows.length > 0) {
    // 成功：B列に "GAP_REPAIR" をセット
    validRows.forEach(row => { row[1] = "GAP_REPAIR"; });
    
    // シートに追記
    const appended = appendRowsToSheet_(ohlcvSh, validRows);
    if (appended === 0) throw new Error("有効なタイムスタンプ付きOHLCV行がありませんでした");
    SpreadsheetApp.flush();

    // 重複削除（timestamp + symbol）
    const newLastRow = ohlcvSh.getLastRow();
    if (newLastRow > 1) {
      ohlcvSh.getRange(2, 1, newLastRow - 1, 8).removeDuplicates([1, 3]);
      const afterDedup = ohlcvSh.getLastRow();
      if (afterDedup > 1) {
        ohlcvSh.getRange(2, 1, afterDedup - 1, 8)
          .sort([{ column: 1, ascending: true }, { column: 3, ascending: true }]);
      }
    }

    console.log(`[refetchSymbolGap] ${symbol} : ${validRows.length}行を追記・重複排除完了`);
    return { ok: true, symbol, rows: validRows.length };
  } else {
    // 失敗：ダミー行を追加（B列に "GAP_FAILED"）
    const dummyRows = makeGapFailedRows_(symbol, startDate, {}, {});
    const appended = appendRowsToSheet_(ohlcvSh, dummyRows);
    if (appended === 0) throw new Error("GAP_FAILED行のタイムスタンプ生成に失敗しました");
    SpreadsheetApp.flush();
    
    console.log(`[refetchSymbolGap] ${symbol} : データなし → ダミー行を追加しました`);
    return { ok: false, symbol, rows: 0, dummy: true };
  }
}

function repairCloseSnapshotRowsSilent_(targets) {
  const ss = SpreadsheetApp.openById(
    PropertiesService.getScriptProperties().getProperty("SPREADSHEET_ID")
  );
  const sh = ss.getSheetByName(CONFIG.OHLCV_SHEET);
  if (!sh) throw new Error("ohlcv_4h シートが見つかりません");

  const summary = {
    ok: true,
    updatedRows: 0,
    appendedRows: 0,
    dedupedAndSorted: false,
    results: []
  };

  (targets || []).forEach(target => {
    const symbol = String(target.symbol || "").trim();
    const dateStr = normalizeDateStrSlash_(target.dateStr);
    if (!symbol || !dateStr) {
      summary.ok = false;
      summary.results.push({ symbol: symbol, dateStr: dateStr, ok: false, error: "INVALID_TARGET" });
      return;
    }

    const start = parseJstDate_(dateStr, false);
    const end = parseJstDate_(dateStr, true);
    if (!isValidOhlcvTimestamp_(start) || !isValidOhlcvTimestamp_(end)) {
      summary.ok = false;
      summary.results.push({ symbol: symbol, dateStr: dateStr, ok: false, error: "INVALID_DATE" });
      return;
    }

    const fetchedRows = fetchOHLCVRecent(
      symbol,
      null,
      Math.floor(start.getTime() / 1000),
      Math.floor(end.getTime() / 1000)
    ) || [];
    const pmRow = fetchedRows.find(row => {
      const ts = row[0];
      return ts instanceof Date &&
        Utilities.formatDate(ts, CONFIG.TIMEZONE, "yyyy/MM/dd") === dateStr &&
        getOhlcvSessionName_(ts) === "PM";
    });

    if (!pmRow) {
      summary.ok = false;
      summary.results.push({ symbol: symbol, dateStr: dateStr, ok: false, error: "PM_ROW_NOT_FOUND" });
      return;
    }

    const close = Number(pmRow[6]);
    if (Number.isFinite(target.expectedClose) && close !== target.expectedClose) {
      summary.ok = false;
      summary.results.push({
        symbol: symbol,
        dateStr: dateStr,
        ok: false,
        error: "CLOSE_MISMATCH",
        actualClose: close,
        expectedClose: target.expectedClose
      });
      return;
    }

    const normalizedPmRow = normalizeOhlcvRowForWrite_(pmRow.slice());
    if (!normalizedPmRow) {
      summary.ok = false;
      summary.results.push({ symbol: symbol, dateStr: dateStr, ok: false, error: "INVALID_TIMESTAMP" });
      return;
    }

    const rowNums = findOhlcvRowNumsByTimestampSymbol_(sh, normalizedPmRow[0], symbol);
    if (rowNums.length > 0) {
      rowNums.forEach(rowNum => {
        const writeRow = normalizedPmRow.slice();
        const existingAlertId = sh.getRange(rowNum, 2).getValue();
        writeRow[1] = existingAlertId || "CLOSE_SNAPSHOT_REPAIR";
        sh.getRange(rowNum, 1, 1, writeRow.length).setValues([writeRow]);
        sh.getRange(rowNum, 1).setNumberFormat("yyyy/mm/dd hh:mm");
      });
      summary.updatedRows += rowNums.length;
    } else {
      const writeRow = normalizedPmRow.slice();
      writeRow[1] = "CLOSE_SNAPSHOT_REPAIR";
      appendRowsToSheet_(sh, [writeRow]);
      summary.appendedRows++;
    }

    summary.results.push({
      symbol: symbol,
      dateStr: dateStr,
      ok: true,
      close: close,
      rowNums: rowNums
    });
  });

  if (summary.updatedRows > 0 || summary.appendedRows > 0) {
    dedupeAndSortOhlcv_(sh);
    SpreadsheetApp.flush();
    summary.dedupedAndSorted = true;
  }

  console.log("[repairCloseSnapshotRowsSilent] " + JSON.stringify(summary));
  return summary;
}

function findOhlcvRowNumsByTimestampSymbol_(sheet, targetTs, symbol) {
  const rowNums = [];
  if (!(targetTs instanceof Date) || isNaN(targetTs.getTime())) return rowNums;

  const targetTime = targetTs.getTime();
  const targetDateKey = Utilities.formatDate(targetTs, CONFIG.TIMEZONE, "yyyy/MM/dd");
  const targetSymbol = String(symbol || "").trim();
  const chunkSize = 5000;
  let row = sheet.getLastRow();

  while (row >= 2) {
    const numRows = Math.min(chunkSize, row - 1);
    const startRow = row - numRows + 1;
    const values = sheet.getRange(startRow, 1, numRows, 3).getValues();
    let sawOlderThanTarget = false;

    for (let i = values.length - 1; i >= 0; i--) {
      const ts = values[i][0];
      if (!(ts instanceof Date) || isNaN(ts.getTime())) continue;

      const dateKey = Utilities.formatDate(ts, CONFIG.TIMEZONE, "yyyy/MM/dd");
      if (dateKey < targetDateKey) {
        sawOlderThanTarget = true;
        break;
      }

      const rowSymbol = String(values[i][2] || "").trim();
      if (dateKey === targetDateKey && ts.getTime() === targetTime && rowSymbol === targetSymbol) {
        rowNums.push(startRow + i);
      }
    }

    if (sawOlderThanTarget) break;
    row = startRow - 1;
  }

  return rowNums.sort((a, b) => a - b);
}

function repairHistoricalOhlcvVolumesLegacyInternal_(options) {
  const opts = options || {};
  const dryRun = !!opts.dryRun;
  const props = PropertiesService.getScriptProperties();
  const startTime = Date.now();

  let state = readHistoricalVolumeRepairState_(props);
  if (opts.reset || !state || state.dryRun !== dryRun) {
    state = {
      dryRun: dryRun,
      nextIndex: 0,
      totalSymbols: 0,
      updatedRows: 0,
      missingRows: 0,
      fetchedSymbols: 0,
      deferredSymbols: 0,
      parseErrors: 0,
      batches: 0,
      startedAt: new Date().toISOString(),
      updatedAt: new Date().toISOString()
    };
    props.setProperty(HISTORICAL_VOLUME_REPAIR_STATE_KEY, JSON.stringify(state));
  }

  const ss = SpreadsheetApp.openById(
    PropertiesService.getScriptProperties().getProperty("SPREADSHEET_ID")
  );
  const sh = ss.getSheetByName(CONFIG.OHLCV_SHEET);
  if (!sh) throw new Error("ohlcv_4h シートが見つかりません");

  console.log(`[historicalVolumeRepair] start dryRun=${dryRun}, nextIndex=${state.nextIndex}`);
  const context = buildHistoricalVolumeRepairContext_(sh);
  const symbols = context.symbols;
  state.totalSymbols = symbols.length;

  if (symbols.length === 0 || state.nextIndex >= symbols.length) {
    props.deleteProperty(HISTORICAL_VOLUME_REPAIR_STATE_KEY);
    deleteTriggersByHandler_(HISTORICAL_VOLUME_REPAIR_RESUME_HANDLER);
    const complete = buildHistoricalVolumeRepairSummary_(state, true);
    console.log(`[historicalVolumeRepair] complete ${JSON.stringify(complete)}`);
    return complete;
  }

  while (state.nextIndex < symbols.length) {
    if (Date.now() - startTime > HISTORICAL_VOLUME_REPAIR_MAX_RUN_MS) break;

    const batchSymbols = symbols.slice(
      state.nextIndex,
      Math.min(state.nextIndex + HISTORICAL_VOLUME_REPAIR_BATCH_SYMBOLS, symbols.length)
    );
    const result = fetchHistoricalVolumeRepairRowsForSymbols_(batchSymbols, context.ranges);

    if (result.transient) {
      state.deferredSymbols += result.deferredSymbols;
      state.updatedAt = new Date().toISOString();
      props.setProperty(HISTORICAL_VOLUME_REPAIR_STATE_KEY, JSON.stringify(state));
      scheduleHistoricalVolumeRepairResume_();
      console.log(`[historicalVolumeRepair] transient pause nextIndex=${state.nextIndex}: ${result.message}`);
      return buildHistoricalVolumeRepairSummary_(state, false);
    }

    const updates = buildHistoricalVolumeRepairUpdates_(result.rows, context.rowMapByKey);
    const writeSummary = writeHistoricalVolumeRepairUpdates_(sh, updates, dryRun);

    state.updatedRows += writeSummary.rows;
    state.missingRows += countHistoricalVolumeRepairMissingRows_(batchSymbols, context.rowMapByKey, updates.seenKeys);
    state.fetchedSymbols += result.fetchedSymbols;
    state.parseErrors += result.parseErrors;
    state.nextIndex += batchSymbols.length;
    state.batches++;
    state.updatedAt = new Date().toISOString();
    props.setProperty(HISTORICAL_VOLUME_REPAIR_STATE_KEY, JSON.stringify(state));

    console.log(`[historicalVolumeRepair] batch nextIndex=${state.nextIndex}/${symbols.length}, rows=${writeSummary.rows}, parseErrors=${result.parseErrors}`);
    if (Date.now() - startTime > HISTORICAL_VOLUME_REPAIR_MAX_RUN_MS) break;
    Utilities.sleep(200);
  }

  if (state.nextIndex < symbols.length) {
    scheduleHistoricalVolumeRepairResume_();
    return buildHistoricalVolumeRepairSummary_(state, false);
  }

  if (!dryRun) {
    dedupeAndSortOhlcv_(sh);
    SpreadsheetApp.flush();
  }
  props.deleteProperty(HISTORICAL_VOLUME_REPAIR_STATE_KEY);
  deleteTriggersByHandler_(HISTORICAL_VOLUME_REPAIR_RESUME_HANDLER);
  const summary = buildHistoricalVolumeRepairSummary_(state, true);
  console.log(`[historicalVolumeRepair] complete ${JSON.stringify(summary)}`);
  return summary;
}

function buildHistoricalVolumeRepairContext_(sheet) {
  const lastRow = sheet.getLastRow();
  const ranges = {};
  const rowMapByKey = {};
  if (lastRow < 2) return { symbols: [], ranges: ranges, rowMapByKey: rowMapByKey };

  const values = sheet.getRange(2, 1, lastRow - 1, 3).getValues();
  for (let i = 0; i < values.length; i++) {
    const rowNum = i + 2;
    const ts = values[i][0];
    const symbol = String(values[i][2] || "").trim();
    if (!isHistoricalVolumeRepairSessionTs_(ts) || !symbol) continue;

    const key = makeHistoricalVolumeRepairKey_(symbol, ts);
    if (!rowMapByKey[key]) rowMapByKey[key] = [];
    rowMapByKey[key].push(rowNum);

    if (!ranges[symbol]) ranges[symbol] = { min: ts, max: ts };
    if (ts.getTime() < ranges[symbol].min.getTime()) ranges[symbol].min = ts;
    if (ts.getTime() > ranges[symbol].max.getTime()) ranges[symbol].max = ts;
  }

  return {
    symbols: Object.keys(ranges).sort(),
    ranges: ranges,
    rowMapByKey: rowMapByKey
  };
}

function isHistoricalVolumeRepairSessionTs_(ts) {
  if (!(ts instanceof Date) || isNaN(ts.getTime())) return false;
  const hour = parseInt(Utilities.formatDate(ts, CONFIG.TIMEZONE, "HH"), 10);
  const minute = parseInt(Utilities.formatDate(ts, CONFIG.TIMEZONE, "mm"), 10);
  return minute === 0 && (hour === 9 || hour === 13);
}

function makeHistoricalVolumeRepairKey_(symbol, ts) {
  return String(symbol || "").trim() + "|" + ts.getTime();
}

function fetchHistoricalVolumeRepairRowsForSymbols_(symbols, ranges) {
  const requests = [];
  const metaList = [];
  symbols.forEach(symbol => {
    const range = ranges[symbol];
    if (!range || !range.min || !range.max) return;
    const start = new Date(range.min.getTime() - 24 * 60 * 60 * 1000);
    const end = new Date(range.max.getTime() + 24 * 60 * 60 * 1000);
    const pair = buildOhlcvRequestPair_(
      symbol,
      null,
      Math.floor(start.getTime() / 1000),
      Math.floor(end.getTime() / 1000)
    );
    metaList.push({
      symbol: symbol,
      minTime: range.min.getTime(),
      maxTime: range.max.getTime(),
      intraIdx: requests.length
    });
    requests.push({ url: pair.intraUrl, muteHttpExceptions: true, headers: { "User-Agent": "Mozilla/5.0" } });
  });

  const summary = {
    rows: [],
    fetchedSymbols: 0,
    deferredSymbols: 0,
    parseErrors: 0,
    transient: false,
    message: ""
  };
  if (requests.length === 0) return summary;

  let responses;
  try {
    responses = UrlFetchApp.fetchAll(requests);
  } catch (e) {
    if (isTransientError_(e)) {
      summary.transient = true;
      summary.deferredSymbols = symbols.length;
      summary.message = e.message;
      return summary;
    }
    throw e;
  }

  metaList.forEach(meta => {
    try {
      const intraRes = responses[meta.intraIdx];
      if (isTransientHttpResponse_(intraRes)) {
        summary.transient = true;
        summary.deferredSymbols++;
        summary.message = "transient HTTP response";
        return;
      }

      const rows = parseIntraResponse_(meta.symbol, intraRes, null) || [];
      let added = 0;
      rows.forEach(row => {
        const ts = row[0];
        if (!(ts instanceof Date) || isNaN(ts.getTime())) return;
        const time = ts.getTime();
        if (time < meta.minTime || time > meta.maxTime) return;
        summary.rows.push(row);
        added++;
      });
      if (added > 0) summary.fetchedSymbols++;
    } catch (e) {
      summary.parseErrors++;
      console.log(`[historicalVolumeRepair] parse error: ${meta.symbol} - ${e.message}`);
    }
  });

  return summary;
}

function buildHistoricalVolumeRepairUpdates_(fetchedRows, rowMapByKey) {
  const byRowNum = {};
  const seenKeys = {};
  (fetchedRows || []).forEach(row => {
    const ts = row[0];
    const symbol = String(row[2] || "").trim();
    if (!(ts instanceof Date) || isNaN(ts.getTime()) || !symbol) return;

    const key = makeHistoricalVolumeRepairKey_(symbol, ts);
    const rowNums = rowMapByKey[key] || [];
    if (rowNums.length === 0) return;

    seenKeys[key] = true;
    const values = row.slice(3, 8);
    rowNums.forEach(rowNum => {
      byRowNum[rowNum] = values.slice();
    });
  });
  return { byRowNum: byRowNum, seenKeys: seenKeys };
}

function writeHistoricalVolumeRepairUpdates_(sheet, updates, dryRun) {
  const rowNums = Object.keys(updates.byRowNum).map(Number).sort((a, b) => a - b);
  if (rowNums.length === 0) return { rows: 0, chunks: 0 };
  if (dryRun) return { rows: rowNums.length, chunks: 0 };

  let chunks = 0;
  let startRow = rowNums[0];
  let rows = [updates.byRowNum[startRow]];

  for (let i = 1; i < rowNums.length; i++) {
    const rowNum = rowNums[i];
    if (rowNum === rowNums[i - 1] + 1) {
      rows.push(updates.byRowNum[rowNum]);
      continue;
    }
    sheet.getRange(startRow, 4, rows.length, 5).setValues(rows);
    chunks++;
    startRow = rowNum;
    rows = [updates.byRowNum[rowNum]];
  }

  sheet.getRange(startRow, 4, rows.length, 5).setValues(rows);
  chunks++;
  SpreadsheetApp.flush();
  return { rows: rowNums.length, chunks: chunks };
}

function countHistoricalVolumeRepairMissingRows_(symbols, rowMapByKey, seenKeys) {
  const symbolSet = {};
  symbols.forEach(symbol => symbolSet[symbol] = true);
  let missing = 0;
  Object.keys(rowMapByKey).forEach(key => {
    const symbol = key.split("|")[0];
    if (!symbolSet[symbol] || seenKeys[key]) return;
    missing += rowMapByKey[key].length;
  });
  return missing;
}

function scheduleHistoricalVolumeRepairResume_() {
  deleteTriggersByHandler_(HISTORICAL_VOLUME_REPAIR_RESUME_HANDLER);
  ScriptApp.newTrigger(HISTORICAL_VOLUME_REPAIR_RESUME_HANDLER)
    .timeBased()
    .after(HISTORICAL_VOLUME_REPAIR_RESUME_AFTER_MS)
    .create();
}

function repairHistoricalOhlcvVolumesChunked_(options) {
  const opts = options || {};
  const dryRun = !!opts.dryRun;
  const props = PropertiesService.getScriptProperties();
  const startTime = Date.now();
  const guardMs = 30 * 1000;

  let state = readHistoricalVolumeRepairState_(props);
  if (opts.reset || !state || state.version !== 2 || state.dryRun !== dryRun) {
    state = makeHistoricalVolumeRepairChunkedState_(dryRun);
    props.setProperty(HISTORICAL_VOLUME_REPAIR_STATE_KEY, JSON.stringify(state));
  }

  const ss = SpreadsheetApp.openById(
    PropertiesService.getScriptProperties().getProperty("SPREADSHEET_ID")
  );
  const sh = ss.getSheetByName(CONFIG.OHLCV_SHEET);
  if (!sh) throw new Error("ohlcv_4h シートが見つかりません");

  console.log(`[historicalVolumeRepair] start chunked dryRun=${dryRun}, rowStart=${state.rowStart}, nextSymbolIndex=${state.nextSymbolIndex}`);

  while (Date.now() - startTime < HISTORICAL_VOLUME_REPAIR_MAX_RUN_MS - guardMs) {
    const lastRow = sh.getLastRow();
    state.totalRows = Math.max(0, lastRow - 1);

    if (state.rowStart > lastRow) {
      if (!dryRun) {
        dedupeAndSortOhlcv_(sh);
        SpreadsheetApp.flush();
      }
      props.deleteProperty(HISTORICAL_VOLUME_REPAIR_STATE_KEY);
      deleteTriggersByHandler_(HISTORICAL_VOLUME_REPAIR_RESUME_HANDLER);
      const complete = buildHistoricalVolumeRepairChunkedSummary_(state, true);
      console.log(`[historicalVolumeRepair] complete ${JSON.stringify(complete)}`);
      return complete;
    }

    if (!state.chunkSymbols || state.chunkSymbols.length === 0) {
      const chunk = readHistoricalVolumeRepairChunkSymbols_(sh, state.rowStart, lastRow);
      state.rowEnd = chunk.endRow;
      state.chunkSymbols = chunk.symbols;
      state.nextSymbolIndex = 0;
      state.updatedAt = new Date().toISOString();
      props.setProperty(HISTORICAL_VOLUME_REPAIR_STATE_KEY, JSON.stringify(state));

      if (state.chunkSymbols.length === 0) {
        state.scannedRows += Math.max(0, state.rowEnd - state.rowStart + 1);
        state.rowStart = state.rowEnd + 1;
        state.rowEnd = 0;
        state.updatedAt = new Date().toISOString();
        props.setProperty(HISTORICAL_VOLUME_REPAIR_STATE_KEY, JSON.stringify(state));
        continue;
      }
    }

    const batchSymbols = state.chunkSymbols.slice(
      state.nextSymbolIndex,
      Math.min(state.nextSymbolIndex + HISTORICAL_VOLUME_REPAIR_FETCH_BATCH_SYMBOLS, state.chunkSymbols.length)
    );
    if (batchSymbols.length === 0) {
      state.scannedRows += Math.max(0, state.rowEnd - state.rowStart + 1);
      state.rowStart = state.rowEnd + 1;
      state.rowEnd = 0;
      state.chunkSymbols = [];
      state.nextSymbolIndex = 0;
      state.updatedAt = new Date().toISOString();
      props.setProperty(HISTORICAL_VOLUME_REPAIR_STATE_KEY, JSON.stringify(state));
      continue;
    }

    const context = buildHistoricalVolumeRepairChunkContext_(sh, state.rowStart, state.rowEnd, batchSymbols);
    const result = fetchHistoricalVolumeRepairRowsForSymbols_(batchSymbols, context.ranges);
    if (result.transient) {
      state.deferredSymbols += result.deferredSymbols || batchSymbols.length;
      state.updatedAt = new Date().toISOString();
      props.setProperty(HISTORICAL_VOLUME_REPAIR_STATE_KEY, JSON.stringify(state));
      scheduleHistoricalVolumeRepairResume_();
      console.log(`[historicalVolumeRepair] transient pause rowStart=${state.rowStart}, nextSymbolIndex=${state.nextSymbolIndex}: ${result.message}`);
      return buildHistoricalVolumeRepairChunkedSummary_(state, false);
    }

    const updates = buildHistoricalVolumeRepairUpdates_(result.rows, context.rowMapByKey);
    const writeSummary = writeHistoricalVolumeRepairUpdates_(sh, updates, dryRun);

    state.updatedRows += writeSummary.rows;
    state.missingRows += countHistoricalVolumeRepairMissingRows_(batchSymbols, context.rowMapByKey, updates.seenKeys);
    state.fetchedSymbols += result.fetchedSymbols;
    state.parseErrors += result.parseErrors;
    state.nextSymbolIndex += batchSymbols.length;
    state.batches++;
    state.updatedAt = new Date().toISOString();
    props.setProperty(HISTORICAL_VOLUME_REPAIR_STATE_KEY, JSON.stringify(state));

    console.log(`[historicalVolumeRepair] batch row=${state.rowStart}-${state.rowEnd}, symbols=${state.nextSymbolIndex}/${state.chunkSymbols.length}, rows=${writeSummary.rows}, parseErrors=${result.parseErrors}`);

    if (state.nextSymbolIndex >= state.chunkSymbols.length) {
      state.scannedRows += Math.max(0, state.rowEnd - state.rowStart + 1);
      state.rowStart = state.rowEnd + 1;
      state.rowEnd = 0;
      state.chunkSymbols = [];
      state.nextSymbolIndex = 0;
      state.updatedAt = new Date().toISOString();
      props.setProperty(HISTORICAL_VOLUME_REPAIR_STATE_KEY, JSON.stringify(state));
    }

    Utilities.sleep(200);
  }

  scheduleHistoricalVolumeRepairResume_();
  console.log(`[historicalVolumeRepair] pause rowStart=${state.rowStart}, nextSymbolIndex=${state.nextSymbolIndex}, updatedRows=${state.updatedRows}`);
  return buildHistoricalVolumeRepairChunkedSummary_(state, false);
}

    if (ts.getTime() < ranges[symbol].min.getTime()) ranges[symbol].min = ts;
    if (ts.getTime() > ranges[symbol].max.getTime()) ranges[symbol].max = ts;
  }

  return { ranges: ranges, rowMapByKey: rowMapByKey };
}

/**
 * 直近N日間のOHLCVデータを高速スキャンし、セッション不足の日を抽出する。
 * 休場日（土日祝・年末年始）は診断対象から除外する。
 * B列に GAP_REPAIR / GAP_FAILED がある日も除外する。
 */
function quickScanMissingSessions(daysBack = 14, minSessions = 2, options) {
  const opts = options || {};
  const ss = SpreadsheetApp.openById(
    PropertiesService.getScriptProperties().getProperty("SPREADSHEET_ID")
  );
  const sh = ss.getSheetByName(CONFIG.OHLCV_SHEET);
  if (!sh) throw new Error("ohlcv_4h シートが見つかりません");

  const now = getOhlcvBusinessDate_();
  const cutoff = new Date(now.getTime() - daysBack * 24 * 60 * 60 * 1000);
  const cutoffDateKey = Utilities.formatDate(cutoff, CONFIG.TIMEZONE, "yyyy/MM/dd");
  const nowDateKey = Utilities.formatDate(now, CONFIG.TIMEZONE, "yyyy/MM/dd");

  const lastRow = sh.getLastRow();
  if (lastRow < 2) return { count: 0, details: {} };

  // 休場日セットを取得
  const holidaySet = getMarketHolidaySet_();

  const symbols = getQuickRepairTargetSymbols_(ss);
  const symbolLookup = {};
  symbols.forEach(sym => symbolLookup[sym] = true);

  // ohlcv_4h はA列(timestamp)昇順に整列している前提。直近分だけを末尾から読む。
  const values = readRecentOhlcvRowsByTimestamp_(sh, cutoffDateKey, nowDateKey, 3, {
    deadlineMs: opts.deadlineMs || 0,
    maxRows: opts.maxRows || getQuickRepairRecentScanMaxRows_(symbols.length, daysBack)
  });
  const scanStats = values.scanStats || {};
  if (scanStats.complete === false) {
    writeProcessLog_("ギャップ修復", "直近OHLCVの確認が時間内に終わらなかったため、今回の実行を中断して次回へ回します。", {
      businessDate: formatYmd_(now),
      targetSymbols: symbols.length,
      scannedRows: scanStats.scannedRows || 0,
      maxRows: scanStats.maxRows || 0,
      stoppedBy: scanStats.stoppedBy || ""
    });
    return { count: 0, details: {}, incomplete: true, reason: "scan_incomplete", scanStats: scanStats };
  }

  // 日付×銘柄 ごとの有効セッション数カウント
  const countMap = {};
  const periodMap = {};
  const hasGapMarker = {};
  const dateStrByDayKey = {};

  for (const row of values) {
    const ts = row[0];
    const alertId = String(row[1] || "").trim();
    const sym = String(row[2] || "").trim();
    if (!(ts instanceof Date) || !sym || !symbolLookup[sym]) continue;

    const dayKey = toJstDayKeyFromDate_(ts);
    const dateStr = dateStrByDayKey[dayKey] || Utilities.formatDate(ts, CONFIG.TIMEZONE, "yyyy/MM/dd");
    dateStrByDayKey[dayKey] = dateStr;
    if (dateStr < cutoffDateKey || dateStr > nowDateKey) continue;
    const key = sym + "|" + dateStr;

    // GAP_FAILED はカウントしない（マーカーのみ記録）
    if (alertId === "GAP_FAILED") {
      hasGapMarker[key] = true;
      continue;
    }

    const period = getOhlcvSessionName_(ts);
    if (!period) continue;
    if (!periodMap[key]) periodMap[key] = {};
    periodMap[key][period] = true;
    countMap[key] = Object.keys(periodMap[key]).length;
    if (alertId === "GAP_REPAIR") {
      hasGapMarker[key] = true;
    }
  }

  // 診断対象期間内の全日付を走査し、休場日以外でデータが不足している日を探す
  const missing = {};
  const currentDate = new Date(cutoff);
  while (currentDate <= now) {
    const dateStr = Utilities.formatDate(currentDate, CONFIG.TIMEZONE, "yyyy/MM/dd");
    const isHoliday = isMarketHolidayJst_(currentDate, holidaySet);

    // 休場日は診断しない
    if (!isHoliday) {
      symbols.forEach(sym => {
        const key = sym + "|" + dateStr;
        const cnt = countMap[key] || 0;
        if (cnt < minSessions && !hasGapMarker[key]) {
          if (!missing[sym]) missing[sym] = [];
          missing[sym].push(dateStr);
        }
      });
    }
    currentDate.setDate(currentDate.getDate() + 1);
  }

  const symList = Object.keys(missing);
  writeProcessLog_("ギャップ修復", `直近 ${daysBack} 日を確認しました。休場日を除き、まだ補填対象の不足がある銘柄は ${symList.length} 件です。`, {
    businessDate: formatYmd_(now),
    targetSymbols: symbols.length,
    missingSymbols: symList.length
  });
  return {
    count: symList.length,
    details: missing,
    sessionInfo: {
      countMap: countMap,
      periodMap: periodMap,
      markerMap: hasGapMarker
    },
    scanStats: scanStats
  };
}

/**
 * JST日付文字列（yyyy/MM/dd）を JST日キー（epoch day）に変換する。
 */
function toJstDayKeyFromDateStr_(dateStr) {
  const normalized = String(dateStr || "").trim().replace(/\//g, "-");
  const d = new Date(normalized + "T00:00:00+09:00");
  if (isNaN(d.getTime())) return null;
  return Math.floor((d.getTime() + (9 * 60 * 60 * 1000)) / 86400000);
}

/**
 * Date を JST日キー（epoch day）に変換する。
 */
function toJstDayKeyFromDate_(dateObj) {
  return Math.floor((dateObj.getTime() + (9 * 60 * 60 * 1000)) / 86400000);
}

/**
 * 残タスクに必要なキーだけを対象にセッション数マップを構築する（キー: symbol|yyyy/MM/dd）。
 * GAP_FAILED は除外してカウントする。
 */
function buildSessionCountMapForTasks_(sheet, tasks, startIndex) {
  const targetSymbols = {};
  const targetDayToDateStr = {};
  for (let i = startIndex; i < tasks.length; i++) {
    const t = tasks[i];
    targetSymbols[t.symbol] = true;
    const dayKey = toJstDayKeyFromDateStr_(t.dateStr);
    if (dayKey === null) continue;
    targetDayToDateStr[dayKey] = t.dateStr;
  }

  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return {};
  const values = sheet.getRange(2, 1, lastRow - 1, 3).getValues();
  const map = {};
  const periodMap = {};
  for (const row of values) {
    const ts = row[0];
    const alertId = String(row[1] || "").trim();
    const sym = String(row[2] || "").trim();
    if (!(ts instanceof Date) || !sym || !targetSymbols[sym]) continue;
    if (alertId === "GAP_FAILED") continue;
    const rowDayKey = toJstDayKeyFromDate_(ts);
    const rowDateStr = targetDayToDateStr[rowDayKey];
    if (!rowDateStr) continue;
    const key = sym + "|" + rowDateStr;
    const period = getOhlcvSessionName_(ts);
    if (!period) continue;
    if (!periodMap[key]) periodMap[key] = {};
    periodMap[key][period] = true;
    map[key] = Object.keys(periodMap[key]).length;
  }
  return map;
}

function normalizeDateStrSlash_(value) {
  if (value instanceof Date && !isNaN(value.getTime())) {
    return Utilities.formatDate(value, CONFIG.TIMEZONE, "yyyy/MM/dd");
  }
  const s = String(value || "").trim().replace(/-/g, "/");
  const m = s.match(/^(\d{4})\/(\d{1,2})\/(\d{1,2})/);
  if (!m) return "";
  return m[1] + "/" + String(m[2]).padStart(2, "0") + "/" + String(m[3]).padStart(2, "0");
}

function parseJstDate_(value, endOfDay) {
  const dateStr = normalizeDateStrSlash_(value);
  if (!dateStr) return null;
  const isoDate = dateStr.replace(/\//g, "-");
  return new Date(isoDate + (endOfDay ? "T23:59:59+09:00" : "T00:00:00+09:00"));
}

function buildDateSetBetween_(startDate, endDate) {
  const start = parseJstDate_(startDate, false);
  const end = parseJstDate_(endDate, false);
  const set = {};
  if (!start || !end || isNaN(start.getTime()) || isNaN(end.getTime())) return set;
  for (let d = new Date(start.getTime()); d.getTime() <= end.getTime(); d = new Date(d.getTime() + 24 * 60 * 60 * 1000)) {
    set[Utilities.formatDate(d, CONFIG.TIMEZONE, "yyyy/MM/dd")] = true;
  }
  return set;
}

function getOhlcvSessionName_(ts) {
  if (!isValidOhlcvTimestamp_(ts)) return "";
  const hour = parseInt(Utilities.formatDate(ts, CONFIG.TIMEZONE, "HH"), 10);
  const minute = parseInt(Utilities.formatDate(ts, CONFIG.TIMEZONE, "mm"), 10);
  if (hour === 9 && minute === 0) return "AM";
  if (hour === 13 && minute === 0) return "PM";
  return "";
}

function isOhlcvSessionTimestamp_(ts) {
  return getOhlcvSessionName_(ts) !== "";
}

function makeJstSessionTimestamp_(dateStr, sessionName) {
  const normalized = normalizeDateStrSlash_(dateStr);
  if (!normalized) return null;
  const isoDate = normalized.replace(/\//g, "-");
  const hour = sessionName === "PM" ? "13" : "09";
  const ts = new Date(`${isoDate}T${hour}:00:00+09:00`);
  return isValidOhlcvTimestamp_(ts) ? ts : null;
}

function addSessionPeriod_(periodMap, key, period) {
  if (!period) return;
  if (!periodMap[key]) periodMap[key] = {};
  periodMap[key][period] = true;
}

function countSessionPeriods_(periodMap, key) {
  return periodMap[key] ? Object.keys(periodMap[key]).length : 0;
}

function countMergedSessionPeriods_(basePeriodMap, extraPeriodMap, key) {
  const merged = {};
  if (basePeriodMap[key]) Object.keys(basePeriodMap[key]).forEach(p => merged[p] = true);
  if (extraPeriodMap[key]) Object.keys(extraPeriodMap[key]).forEach(p => merged[p] = true);
  return Object.keys(merged).length;
}

function makeGapFailedRows_(symbol, dateStr, basePeriodMap, extraPeriodMap) {
  const key = symbol + "|" + normalizeDateStrSlash_(dateStr);
  const existing = {};
  if (basePeriodMap && basePeriodMap[key]) Object.keys(basePeriodMap[key]).forEach(p => existing[p] = true);
  if (extraPeriodMap && extraPeriodMap[key]) Object.keys(extraPeriodMap[key]).forEach(p => existing[p] = true);

  const sessions = [];
  if (!existing.AM) sessions.push("AM");
  if (!existing.PM) sessions.push("PM");
  if (sessions.length === 0) return [];

  return sessions.map(sessionName => {
    const ts = makeJstSessionTimestamp_(dateStr, sessionName);
    if (!isValidOhlcvTimestamp_(ts)) throw new Error("Invalid GAP_FAILED timestamp: " + dateStr);
    return [ts, "GAP_FAILED", symbol, "", "", "", "", ""];
  });
}

function flattenMissingTasks_(details) {
  const tasks = [];
  Object.keys(details || {}).sort().forEach(sym => {
    const seen = {};
    (details[sym] || []).forEach(dateStr => {
      const normalized = normalizeDateStrSlash_(dateStr);
      if (!normalized || seen[normalized]) return;
      seen[normalized] = true;
      tasks.push({ symbol: sym, dateStr: normalized });
    });
  });
  return tasks;
}

function groupQuickRepairTasksBySymbol_(tasks) {
  const map = {};
  tasks.forEach(t => {
    if (!map[t.symbol]) map[t.symbol] = {};
    map[t.symbol][t.dateStr] = true;
  });
  return Object.keys(map).sort().map(sym => ({
    symbol: sym,
    dates: Object.keys(map[sym]).sort()
  }));
}

function countQuickRepairGroupTasks_(groups) {
  return (groups || []).reduce((sum, g) => sum + (g.dates ? g.dates.length : 0), 0);
}

function buildQuickRepairSessionInfo_(sheet, tasks) {
  const targetSymbols = {};
  const targetDayToDateStr = {};
  tasks.forEach(t => {
    targetSymbols[t.symbol] = true;
    const dayKey = toJstDayKeyFromDateStr_(t.dateStr);
    if (dayKey !== null) targetDayToDateStr[dayKey] = t.dateStr;
  });

  const info = { countMap: {}, periodMap: {}, markerMap: {} };
  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return info;

  const values = sheet.getRange(2, 1, lastRow - 1, 3).getValues();
  for (const row of values) {
    const ts = row[0];
    const alertId = String(row[1] || "").trim();
    const sym = String(row[2] || "").trim();
    if (!(ts instanceof Date) || !sym || !targetSymbols[sym]) continue;

    const rowDateStr = targetDayToDateStr[toJstDayKeyFromDate_(ts)];
    if (!rowDateStr) continue;
    const key = sym + "|" + rowDateStr;

    if (alertId === "GAP_FAILED") {
      info.markerMap[key] = true;
      continue;
    }
    if (alertId === "GAP_REPAIR") info.markerMap[key] = true;

    addSessionPeriod_(info.periodMap, key, getOhlcvSessionName_(ts));
    info.countMap[key] = countSessionPeriods_(info.periodMap, key);
  }
  return info;
}

function isTransientHttpResponse_(res) {
  if (!res) return true;
  try {
    const code = res.getResponseCode();
    return code === 429 || code === 500 || code === 502 || code === 503 || code === 504;
  } catch (e) {
    return true;
  }
}

function getQuickRepairRecentScanMaxRows_(symbolCount, daysBack) {
  const count = Math.max(1, Number(symbolCount) || 1);
  const days = Math.max(1, Number(daysBack) || 14) + 7;
  const estimated = Math.ceil(count * days * QUICK_REPAIR_SCAN_ROWS_PER_SYMBOL_DAY);
  return Math.min(QUICK_REPAIR_SCAN_MAX_ROWS, Math.max(QUICK_REPAIR_SCAN_MIN_ROWS, estimated));
}

function purgeInvalidOhlcvTimestampRows_(sheet, context, maxRows) {
  const lastRow = sheet.getLastRow();
  if (lastRow <= 1) return { deletedRows: 0, repairSymbols: 0 };

  const scanRows = Math.min(Math.max(1, Number(maxRows) || QUICK_REPAIR_TAIL_CLEANUP_ROWS), lastRow - 1);
  const startRow = lastRow - scanRows + 1;
  const values = sheet.getRange(startRow, 1, scanRows, 3).getValues();
  const deleteRows = [];
  const repairSymbolSet = {};
  const samples = [];

  values.forEach((row, index) => {
    const ts = row[0];
    const invalidReason = !isValidOhlcvTimestamp_(ts)
      ? "invalid_timestamp"
      : (!isOhlcvSessionTimestamp_(ts) ? "non_session_timestamp" : "");
    if (!invalidReason) return;

    const sheetRow = startRow + index;
    const symbol = String(row[2] || "").trim();
    deleteRows.push(sheetRow);
    if (symbol) repairSymbolSet[symbol] = true;
    if (samples.length < 10) {
      samples.push({
        row: sheetRow,
        timestamp: ts == null ? "" : String(ts),
        reason: invalidReason,
        symbol: symbol
      });
    }
  });

  if (deleteRows.length === 0) return { deletedRows: 0, repairSymbols: 0 };

  deleteRowsByRowNums_(sheet, deleteRows);
  const repairSymbols = Object.keys(repairSymbolSet).sort();
  if (repairSymbols.length > 0) setOhlcvRepairSymbols_(repairSymbols);
  writeProcessLog_("OHLCV", "空・無効・対象外時刻のOHLCV行を削除し、対象銘柄を再取得キューへ入れました。", {
    context: context || "",
    scannedRows: scanRows,
    deletedRows: deleteRows.length,
    repairSymbols: repairSymbols.length,
    samples: samples
  });
  return { deletedRows: deleteRows.length, repairSymbols: repairSymbols.length };
}

function maybePurgeInvalidOhlcvTimestampRows_(sheet, context, maxRows, businessDateKey) {
  const props = PropertiesService.getScriptProperties();
  const lastRow = sheet.getLastRow();
  const key = String(businessDateKey || getOhlcvBusinessDateKey_());
  try {
    const raw = props.getProperty(QUICK_REPAIR_CLEANUP_STATE_KEY);
    const state = raw ? JSON.parse(raw) : null;
    if (state && state.businessDate === key && state.checked === true) {
      return { skipped: true, deletedRows: 0, repairSymbols: 0, lastCheckedRow: state.lastRow || 0 };
    }
  } catch (e) {}

  const result = purgeInvalidOhlcvTimestampRows_(sheet, context, maxRows);
  if ((result.deletedRows || 0) === 0) {
    props.setProperty(QUICK_REPAIR_CLEANUP_STATE_KEY, JSON.stringify({
      businessDate: key,
      checked: true,
      lastRow: lastRow,
      maxRows: maxRows,
      checkedAt: new Date().toISOString()
    }));
  } else {
    props.deleteProperty(QUICK_REPAIR_CLEANUP_STATE_KEY);
  }
  return result;
}

function dedupeAndSortOhlcv_(sheet) {
  purgeInvalidOhlcvTimestampRows_(sheet, "dedupeAndSortOhlcv_", QUICK_REPAIR_TAIL_CLEANUP_ROWS);
  const lastRow = sheet.getLastRow();
  if (lastRow <= 1) return;
  sheet.getRange(2, 1, lastRow - 1, 8).removeDuplicates([1, 3]);
  const afterDedup = sheet.getLastRow();
  if (afterDedup > 1) {
    sheet.getRange(2, 1, afterDedup - 1, 8)
      .sort([{ column: 1, ascending: true }, { column: 3, ascending: true }]);
  }
}

function addQuickRepairFailedMarkers_(failedRows, sessionInfo, group) {
  group.dates.forEach(dateStr => {
    const key = group.symbol + "|" + dateStr;
    if (sessionInfo.markerMap[key]) return;
    makeGapFailedRows_(group.symbol, dateStr, sessionInfo.periodMap, null)
      .forEach(row => failedRows.push(row));
    sessionInfo.markerMap[key] = true;
  });
}

function processQuickRepairGroups_(sheet, groups, sessionInfo, minSessions) {
  const MARGIN_DAYS = 3;
  const requests = [];
  const metaList = [];
  const pendingGroups = [];

  groups.forEach(group => {
    const pendingDates = group.dates.filter(dateStr => {
      const key = group.symbol + "|" + dateStr;
      return !sessionInfo.markerMap[key] && (sessionInfo.countMap[key] || 0) < minSessions;
    });
    if (pendingDates.length === 0) return;

    const start = parseJstDate_(pendingDates[0], false);
    const end = parseJstDate_(pendingDates[pendingDates.length - 1], true);
    if (!start || !end) return;

    const fetchStart = new Date(start.getTime() - MARGIN_DAYS * 24 * 60 * 60 * 1000);
    const fetchEnd = new Date(end.getTime() + MARGIN_DAYS * 24 * 60 * 60 * 1000);
    const pair = buildOhlcvRequestPair_(
      group.symbol,
      null,
      Math.floor(fetchStart.getTime() / 1000),
      Math.floor(fetchEnd.getTime() / 1000)
    );
    const targetDateSet = {};
    pendingDates.forEach(d => targetDateSet[d] = true);
    metaList.push({
      symbol: group.symbol,
      dates: pendingDates,
      targetDateSet: targetDateSet,
      intraIdx: requests.length
    });
    pendingGroups.push({ symbol: group.symbol, dates: pendingDates });
    requests.push({ url: pair.intraUrl, muteHttpExceptions: true, headers: { "User-Agent": "Mozilla/5.0" } });
  });

  const summary = {
    groups: pendingGroups.length,
    tasks: countQuickRepairGroupTasks_(pendingGroups),
    repairedDates: 0,
    failedDates: 0,
    deferredDates: 0,
    rows: 0,
    parseErrors: 0
  };
  if (requests.length === 0) return summary;

  const responses = UrlFetchApp.fetchAll(requests);
  const repairRows = [];
  const failedRows = [];
  const fetchedPeriodMap = {};

  metaList.forEach(meta => {
    try {
      const intraRes = responses[meta.intraIdx];
      if (isTransientHttpResponse_(intraRes)) {
        summary.deferredDates += meta.dates.length;
        return;
      }

      const result = parseIntraResponse_(meta.symbol, intraRes, null);
      (result || []).forEach(row => {
        const ts = row[0];
        if (!(ts instanceof Date) || isNaN(ts.getTime())) return;
        const rowDateStr = Utilities.formatDate(ts, CONFIG.TIMEZONE, "yyyy/MM/dd");
        if (!meta.targetDateSet[rowDateStr]) return;
        row[1] = "GAP_REPAIR";
        repairRows.push(row);
        addSessionPeriod_(fetchedPeriodMap, meta.symbol + "|" + rowDateStr, getOhlcvSessionName_(ts));
      });

      meta.dates.forEach(dateStr => {
        const key = meta.symbol + "|" + dateStr;
        const afterCount = countMergedSessionPeriods_(sessionInfo.periodMap, fetchedPeriodMap, key);
        if (afterCount >= minSessions) {
          summary.repairedDates++;
        } else {
          makeGapFailedRows_(meta.symbol, dateStr, sessionInfo.periodMap, fetchedPeriodMap)
            .forEach(row => failedRows.push(row));
          sessionInfo.markerMap[key] = true;
          summary.failedDates++;
        }
      });
    } catch (e) {
      summary.parseErrors++;
      summary.deferredDates += meta.dates.length;
      writeProcessLog_("ギャップ修復", `${meta.symbol} の補填レスポンスを解析できませんでした。この銘柄の日付は次回へ回します。`, {
        symbol: meta.symbol,
        dates: meta.dates,
        message: e.message
      });
    }
  });

  Object.keys(fetchedPeriodMap).forEach(key => {
    Object.keys(fetchedPeriodMap[key]).forEach(period => addSessionPeriod_(sessionInfo.periodMap, key, period));
    sessionInfo.countMap[key] = countSessionPeriods_(sessionInfo.periodMap, key);
  });

  const rowsToAppend = repairRows.concat(failedRows);
  if (rowsToAppend.length > 0) {
    summary.rows = appendRowsToSheet_(sheet, rowsToAppend);
  }
  return summary;
}

function readOhlcvRepairSymbols_() {
  try {
    const repairSymsJson = PropertiesService.getScriptProperties().getProperty("OHLCV_REPAIR_SYMBOLS");
    return repairSymsJson ? JSON.parse(repairSymsJson).map(sym => String(sym || "").trim()).filter(Boolean) : [];
  } catch (e) {
    return [];
  }
}

function setOhlcvRepairSymbols_(symbols) {
  const set = {};
  readOhlcvRepairSymbols_().forEach(sym => { if (sym) set[sym] = true; });
  (symbols || []).forEach(sym => {
    const s = String(sym || "").trim();
    if (s) set[s] = true;
  });
  const merged = Object.keys(set).sort();
  if (merged.length > 0) {
    PropertiesService.getScriptProperties().setProperty("OHLCV_REPAIR_SYMBOLS", JSON.stringify(merged));
  }
  return merged;
}

function getQuickRepairTargetSymbols_(ss) {
  const rawSh = ss.getSheetByName(CONFIG.RAW_SHEET);
  if (!rawSh) return [];
  const lastRow = rawSh.getLastRow();
  if (lastRow < dataStartRow_()) return [];

  const rowCount = lastRow - dataStartRow_() + 1;
  const symbolCodeCol = RAW_IDX.symbol_code + 1;
  const tvSymbolCol = RAW_IDX.tv_symbol + 1;
  const firstCol = Math.min(symbolCodeCol, tvSymbolCol);
  const lastCol = Math.max(symbolCodeCol, tvSymbolCol);
  const values = rawSh.getRange(dataStartRow_(), firstCol, rowCount, lastCol - firstCol + 1).getValues();
  const symbolCodeOffset = symbolCodeCol - firstCol;
  const tvSymbolOffset = tvSymbolCol - firstCol;
  const set = {};

  values.forEach(row => {
    const symbolCode = String(row[symbolCodeOffset] || "").trim();
    const tvSymbol = String(row[tvSymbolOffset] || "").trim();
    const sym = symbolCode || tvSymbol;
    if (sym) set[sym] = true;
  });

  readOhlcvRepairSymbols_().forEach(sym => { if (sym) set[sym] = true; });

  return Object.keys(set).sort();
}

function fetchQuickRepairRecentRowsForSymbols_(symbols, fetchStartSec, fetchEndSec, targetStartKey, targetEndKey) {
  const requests = [];
  const metaList = [];
  symbols.forEach(sym => {
    try {
      const pair = buildOhlcvRequestPair_(sym, null, fetchStartSec, fetchEndSec);
      metaList.push({ symbol: sym, intraIdx: requests.length });
      requests.push({ url: pair.intraUrl, muteHttpExceptions: true, headers: { "User-Agent": "Mozilla/5.0" } });
    } catch (e) {
      writeProcessLog_("ギャップ修復", `${sym} の補填取得URLを作れませんでした。`, { symbol: sym, message: e.message });
    }
  });

  const summary = { symbols: symbols.length, fetchedSymbols: 0, rows: 0, deferredSymbols: 0, parseErrors: 0 };
  if (requests.length === 0) return { rows: [], summary: summary };

  const responses = UrlFetchApp.fetchAll(requests);
  const rows = [];
  metaList.forEach(meta => {
    try {
      const intraRes = responses[meta.intraIdx];
      if (isTransientHttpResponse_(intraRes)) {
        summary.deferredSymbols++;
        return;
      }

      const result = parseIntraResponse_(meta.symbol, intraRes, null);
      let added = 0;
      (result || []).forEach(row => {
        const ts = row[0];
        if (!(ts instanceof Date) || isNaN(ts.getTime())) return;
        const dateStr = Utilities.formatDate(ts, CONFIG.TIMEZONE, "yyyy/MM/dd");
        if (dateStr < targetStartKey || dateStr > targetEndKey) return;
        row[1] = "GAP_REPAIR";
        rows.push(row);
        added++;
      });
      if (added > 0) {
        summary.fetchedSymbols++;
        summary.rows += added;
      }
    } catch (e) {
      summary.parseErrors++;
      writeProcessLog_("ギャップ修復", `${meta.symbol} の補填レスポンスをOHLCV行に変換できませんでした。`, {
        symbol: meta.symbol,
        message: e.message
      });
    }
  });

  return { rows: rows, summary: summary };
}

function readRecentOhlcvRowsByTimestamp_(sheet, cutoffDateKey, nowDateKey, numCols, options) {
  const opts = options || {};
  const lastRow = sheet.getLastRow();
  if (lastRow < 2) {
    const emptyRows = [];
    emptyRows.scanStats = { complete: true, scannedRows: 0, maxRows: 0, stoppedBy: "", mode: "empty" };
    return emptyRows;
  }
  const chunkSize = Math.max(100, Number(opts.chunkSize) || QUICK_REPAIR_READ_CHUNK_ROWS);
  const rows = [];
  const maxRows = Math.max(0, Number(opts.maxRows) || 0);
  const deadlineMs = Math.max(0, Number(opts.deadlineMs) || 0);
  const cutoffDayKey = toJstDayKeyFromDateStr_(cutoffDateKey);
  const nowDayKey = toJstDayKeyFromDateStr_(nowDateKey);

  if (cutoffDayKey !== null && nowDayKey !== null && opts.forceChunked !== true) {
    const boundary = findFirstOhlcvRowOnOrAfterDayKey_(sheet, cutoffDayKey, lastRow, deadlineMs);
    if (boundary.incomplete) {
      rows.scanStats = {
        complete: false,
        scannedRows: 0,
        chunkSize: chunkSize,
        maxRows: maxRows,
        stoppedBy: boundary.stoppedBy || "deadline",
        lastRow: lastRow,
        probes: boundary.probes || 0,
        mode: "binary_bulk"
      };
      return rows;
    }

    if (boundary.row > lastRow) {
      rows.scanStats = {
        complete: true,
        scannedRows: 0,
        chunkSize: 0,
        maxRows: maxRows,
        stoppedBy: "",
        lastRow: lastRow,
        startRow: boundary.row,
        probes: boundary.probes || 0,
        mode: "binary_bulk"
      };
      return rows;
    }

    let startRow = Math.max(2, boundary.row - QUICK_REPAIR_SCAN_BOUNDARY_PADDING_ROWS);
    let readRows = lastRow - startRow + 1;
    let stoppedBy = "";
    const effectiveMaxRows = maxRows ? maxRows + QUICK_REPAIR_SCAN_BOUNDARY_PADDING_ROWS : 0;
    if (effectiveMaxRows && readRows > effectiveMaxRows) {
      startRow = lastRow - effectiveMaxRows + 1;
      readRows = effectiveMaxRows;
      stoppedBy = "maxRows";
    }

    if (deadlineMs && deadlineMs - Date.now() < QUICK_REPAIR_MIN_READ_REMAINING_MS) {
      rows.scanStats = {
        complete: false,
        scannedRows: 0,
        chunkSize: readRows,
        maxRows: maxRows,
        stoppedBy: "deadline",
        lastRow: lastRow,
        startRow: startRow,
        probes: boundary.probes || 0,
        mode: "binary_bulk"
      };
      return rows;
    }

    const values = sheet.getRange(startRow, 1, readRows, numCols).getValues();
    for (let i = 0; i < values.length; i++) {
      const ts = values[i][0];
      if (!(ts instanceof Date) || isNaN(ts.getTime())) continue;
      const dayKey = toJstDayKeyFromDate_(ts);
      if (dayKey < cutoffDayKey || dayKey > nowDayKey) continue;
      rows.push(values[i]);
    }

    rows.scanStats = {
      complete: !stoppedBy,
      scannedRows: readRows,
      chunkSize: readRows,
      maxRows: maxRows,
      stoppedBy: stoppedBy,
      lastRow: lastRow,
      startRow: startRow,
      probes: boundary.probes || 0,
      mode: "binary_bulk"
    };
    return rows;
  }

  let row = lastRow;
  let scannedRows = 0;
  let stoppedBy = "";

  while (row >= 2) {
    if (deadlineMs && deadlineMs - Date.now() < QUICK_REPAIR_MIN_READ_REMAINING_MS) {
      stoppedBy = "deadline";
      break;
    }
    if (maxRows && scannedRows >= maxRows) {
      stoppedBy = "maxRows";
      break;
    }

    const remainingMaxRows = maxRows ? Math.max(0, maxRows - scannedRows) : chunkSize;
    const numRows = Math.min(chunkSize, row - 1, remainingMaxRows);
    if (numRows <= 0) {
      stoppedBy = "maxRows";
      break;
    }
    const startRow = row - numRows + 1;
    const values = sheet.getRange(startRow, 1, numRows, numCols).getValues();
    let sawOlderThanCutoff = false;
    scannedRows += numRows;

    for (let i = values.length - 1; i >= 0; i--) {
      const ts = values[i][0];
      if (!(ts instanceof Date) || isNaN(ts.getTime())) continue;
      const dayKey = cutoffDayKey !== null ? toJstDayKeyFromDate_(ts) : null;
      const dateStr = dayKey === null ? Utilities.formatDate(ts, CONFIG.TIMEZONE, "yyyy/MM/dd") : "";
      if ((dayKey !== null && dayKey < cutoffDayKey) || (dayKey === null && dateStr < cutoffDateKey)) {
        sawOlderThanCutoff = true;
        break;
      }
      if ((dayKey !== null && (nowDayKey === null || dayKey <= nowDayKey)) || (dayKey === null && dateStr <= nowDateKey)) {
        rows.push(values[i]);
      }
    }

    if (sawOlderThanCutoff) break;
    if (deadlineMs && deadlineMs - Date.now() < QUICK_REPAIR_MIN_READ_REMAINING_MS) {
      stoppedBy = "deadline";
      break;
    }
    row = startRow - 1;
  }

  rows.scanStats = {
    complete: !stoppedBy,
    scannedRows: scannedRows,
    chunkSize: chunkSize,
    maxRows: maxRows,
    stoppedBy: stoppedBy,
    lastRow: lastRow,
    mode: "chunked"
  };
  return rows;
}

function findFirstOhlcvRowOnOrAfterDayKey_(sheet, cutoffDayKey, lastRow, deadlineMs) {
  let lo = 2;
  let hi = lastRow;
  let answer = lastRow + 1;
  let probes = 0;

  while (lo <= hi) {
    if (deadlineMs && deadlineMs - Date.now() < QUICK_REPAIR_MIN_READ_REMAINING_MS) {
      return { incomplete: true, stoppedBy: "deadline", probes: probes };
    }

    const mid = Math.floor((lo + hi) / 2);
    const value = sheet.getRange(mid, 1).getValue();
    probes++;
    const dayKey = (value instanceof Date && !isNaN(value.getTime())) ? toJstDayKeyFromDate_(value) : null;

    if (dayKey === null || dayKey >= cutoffDayKey) {
      answer = mid;
      hi = mid - 1;
    } else {
      lo = mid + 1;
    }
  }

  return { row: answer, probes: probes };
}

/**
 * 一時的なGASサーバーエラーか判定する
 */
function isTransientError_(e) {
  const msg = (e.message || '').toLowerCase();
  return msg.includes("server error") ||
         msg.includes("we're sorry") ||
         msg.includes("javascript engine") ||
         msg.includes("unexpected error") ||
         msg.includes("internal") ||
         msg.includes("service unavailable") ||
         msg.includes("quota") ||
         msg.includes("timeout") ||
         msg.includes("503") ||
         msg.includes("502");
}

/**
 * 一時的エラーに対して指数バックオフでリトライするラッパー
 * @param {Function} fn
 * @param {number} maxRetries - デフォルト3回
 * @param {number} baseDelayMs - 初回待機ms（デフォルト2000）
 */
function withRetry_(fn, maxRetries, baseDelayMs) {
  maxRetries = maxRetries || 3;
  baseDelayMs = baseDelayMs || 2000;
  for (var attempt = 0; attempt <= maxRetries; attempt++) {
    try {
      return fn();
    } catch (e) {
      if (attempt < maxRetries && isTransientError_(e)) {
        var delay = baseDelayMs * Math.pow(2, attempt); // 2s→4s→8s
        writeProcessLog_("リトライ", "一時的なエラーのため、少し待って同じ処理を再試行します。", {
          attempt: attempt + 1,
          maxAttempts: maxRetries + 1,
          waitMs: delay,
          message: e.message
        });
        Utilities.sleep(delay);
      } else {
        throw e;
      }
    }
  }
}

/**
 * ギャップ自動修復（再開機能付き）
 * 毎日20:00にトリガー実行されるメイン関数
 */
function quickRepairRecentGaps(options) {
  const opts = options || {};
  const daysBack = opts.daysBack || 14;
  const minSessions = opts.minSessions || 2;
  const batchSymbols = opts.batchSymbols || QUICK_REPAIR_BATCH_SYMBOLS;
  const STATE_VERSION = QUICK_REPAIR_STATE_VERSION;
  const businessDateKey = getOhlcvBusinessDateKey_();

  const props = PropertiesService.getScriptProperties();
  writeProcessLog_("ギャップ修復", `直近 ${daysBack} 日のOHLCV不足セッションを確認し、必要な銘柄だけ補填します。`, {
    businessDate: businessDateKey,
    minSessions: minSessions,
    batchSymbols: batchSymbols,
    isManual: !!getManualOhlcvBusinessDateKey_()
  });

  // ★ ハードタイムアウト対策: 先にリトライトリガーをセット（10分後）
  // 正常完了時は最後に削除。GASに強制終了されても自動再開できる。
  scheduleQuickRepairResume_(QUICK_REPAIR_SAFETY_AFTER_MS);

  let lock = null;
  let lockAcquired = false;
  try {
    lock = LockService.getScriptLock();
    if (!lock.tryLock(QUICK_REPAIR_LOCK_WAIT_MS)) {
      writeProcessLog_("ギャップ修復", "別のギャップ修復が実行中のため、1分後に再試行します。", {
        businessDate: businessDateKey
      });
      scheduleQuickRepairResume_(QUICK_REPAIR_RESUME_AFTER_MS);
      return { ok: true, incomplete: true, reason: "lock_busy" };
    }
    lockAcquired = true;

    const startTime = Date.now();
    const ss = SpreadsheetApp.openById(
      PropertiesService.getScriptProperties().getProperty("SPREADSHEET_ID")
    );
    const ohlcvSh = ss.getSheetByName(CONFIG.OHLCV_SHEET);
    if (!ohlcvSh) throw new Error("ohlcv_4h シートが見つかりません");

    const state = {
      version: STATE_VERSION,
      businessDate: businessDateKey,
      daysBack: daysBack,
      minSessions: minSessions,
      totalRows: 0,
      processedGroups: 0,
      processedTasks: 0,
      repairedDates: 0,
      failedDates: 0,
      deferredDates: 0,
      deferredSymbols: 0,
      parseErrors: 0,
      batches: 0,
      lastScanSymbols: 0,
      lastScanTasks: 0,
      startedAt: new Date().toISOString()
    };
    props.setProperty(QUICK_REPAIR_STATE_KEY, JSON.stringify(state));

    const DEDUPE_GUARD_MS = 60 * 1000;
    const scanDeadlineMs = Math.min(
      startTime + QUICK_REPAIR_SCAN_BUDGET_MS,
      startTime + QUICK_REPAIR_MAX_RUN_MS - DEDUPE_GUARD_MS
    );
    maybePurgeInvalidOhlcvTimestampRows_(ohlcvSh, "quickRepairRecentGaps.entry", QUICK_REPAIR_ENTRY_CLEANUP_ROWS, businessDateKey);
    const scan = quickScanMissingSessions(daysBack, minSessions, { deadlineMs: scanDeadlineMs });
    if (scan.incomplete) {
      state.deferredDates += 1;
      state.updatedAt = new Date().toISOString();
      props.setProperty(QUICK_REPAIR_STATE_KEY, JSON.stringify(state));
      scheduleQuickRepairResume_(QUICK_REPAIR_RESUME_AFTER_MS);
      return {
        ok: true,
        incomplete: true,
        reason: scan.reason || "scan_incomplete",
        scanStats: scan.scanStats || {}
      };
    }
    const tasks = flattenMissingTasks_(scan.details);
    const groups = groupQuickRepairTasksBySymbol_(tasks);
    state.lastScanSymbols = scan.count;
    state.lastScanTasks = tasks.length;
    state.updatedAt = new Date().toISOString();
    props.setProperty(QUICK_REPAIR_STATE_KEY, JSON.stringify(state));
    writeProcessLog_("ギャップ修復", `補填候補をスキャンしました。対象銘柄グループは ${groups.length}、対象日数は ${tasks.length} です。`, {
      groups: groups.length,
      tasks: tasks.length
    });

    if (groups.length === 0) {
      writeProcessLog_("ギャップ修復", "補填が必要な不足はありません。最後にOHLCVシートの重複排除と並べ替えだけ実行します。", { rowsAddedThisRun: state.totalRows });
      dedupeAndSortOhlcv_(ohlcvSh);
      SpreadsheetApp.flush();
      writeProcessLog_("ギャップ修復", "ギャップ修復が完了しました。今回補填対象はありませんでした。", {
        rowsAddedThisRun: state.totalRows,
        deferredDates: state.deferredDates,
        parseErrors: state.parseErrors
      });
      props.deleteProperty(QUICK_REPAIR_STATE_KEY);
      deleteTriggersByHandler_("resumeQuickRepair");
      clearManualOhlcvBusinessDate_();
      return { ok: true, complete: true, processedGroups: state.processedGroups, processedTasks: state.processedTasks, rows: state.totalRows, deferredDates: state.deferredDates, parseErrors: state.parseErrors };
    }

    const sessionInfo = scan.sessionInfo || { countMap: {}, periodMap: {}, markerMap: {} };
    let groupIndex = 0;
    while (groupIndex < groups.length) {
      if (Date.now() - startTime > QUICK_REPAIR_MAX_RUN_MS - DEDUPE_GUARD_MS) break;

      const batchGroups = groups.slice(groupIndex, groupIndex + batchSymbols);
      const batchTasks = flattenQuickRepairGroupsToTasks_(batchGroups);
      writeProcessLog_("ギャップ修復", `不足セッションを補填中です。銘柄グループ ${groupIndex + 1}〜${groupIndex + batchGroups.length} / ${groups.length} を処理します。`, {
        tasks: batchTasks.length
      });
      const result = processQuickRepairGroups_(ohlcvSh, batchGroups, sessionInfo, minSessions);
      SpreadsheetApp.flush();

      groupIndex += batchGroups.length;
      state.processedGroups += batchGroups.length;
      state.processedTasks += countQuickRepairGroupTasks_(batchGroups);
      state.repairedDates += result.repairedDates || 0;
      state.failedDates += result.failedDates || 0;
      state.deferredDates += result.deferredDates || 0;
      state.totalRows += result.rows || 0;
      state.parseErrors += result.parseErrors || 0;
      state.batches++;
      state.updatedAt = new Date().toISOString();
      props.setProperty(QUICK_REPAIR_STATE_KEY, JSON.stringify(state));

      writeProcessLog_("ギャップ修復", `補填バッチが完了しました。進捗は ${groupIndex}/${groups.length} 銘柄グループです。`, {
        rowsAdded: result.rows || 0,
        repairedDates: result.repairedDates || 0,
        failedDates: result.failedDates || 0,
        deferredDates: result.deferredDates || 0,
        parseErrors: result.parseErrors || 0
      });
      if ((result.deferredDates || 0) > 0) {
        scheduleQuickRepairResume_(QUICK_REPAIR_RESUME_AFTER_MS);
        return { ok: true, incomplete: true, reason: "deferred_dates", processedGroups: state.processedGroups, processedTasks: state.processedTasks, rows: state.totalRows, deferredDates: state.deferredDates };
      }
      if (Date.now() - startTime > QUICK_REPAIR_MAX_RUN_MS - DEDUPE_GUARD_MS) break;
      Utilities.sleep(200);
    }

    if (groupIndex < groups.length) {
      scheduleQuickRepairResume_(QUICK_REPAIR_RESUME_AFTER_MS);
      writeProcessLog_("ギャップ修復", `実行時間が近づいたため、${groupIndex}/${groups.length} 銘柄グループまで保存して再開します。`, {
        rowsAddedThisRun: state.totalRows
      });
      return { ok: true, incomplete: true, groupIndex: groupIndex, totalGroups: groups.length, rows: state.totalRows };
    }

    writeProcessLog_("ギャップ修復", "補填行の追記が終わりました。最後にOHLCVシートを重複排除してtimestamp昇順に並べ替えます。", { rowsAddedThisRun: state.totalRows });
    dedupeAndSortOhlcv_(ohlcvSh);
    SpreadsheetApp.flush();
    writeProcessLog_("ギャップ修復", "ギャップ修復が完了しました。", {
      groups: groups.length,
      tasks: tasks.length,
      rowsAddedThisRun: state.totalRows,
      repairedDates: state.repairedDates,
      failedDates: state.failedDates,
      deferredDates: state.deferredDates,
      parseErrors: state.parseErrors
    });

    props.deleteProperty(QUICK_REPAIR_STATE_KEY);
    deleteTriggersByHandler_("resumeQuickRepair");
    clearManualOhlcvBusinessDate_();
    return { ok: true, complete: true, totalGroups: groups.length, totalTasks: tasks.length, rows: state.totalRows, processedGroups: state.processedGroups, processedTasks: state.processedTasks, repairedDates: state.repairedDates, failedDates: state.failedDates, deferredDates: state.deferredDates, parseErrors: state.parseErrors };

  } catch (e) {
    if (isTransientError_(e)) {
      // 一時的サーバーエラー: throwせずトリガーで再開（GASに「失敗」と記録させない）
      writeProcessLog_("ギャップ修復", "Yahoo FinanceまたはGoogle側の一時的なエラーです。失敗扱いにせず、再開トリガーに任せます。", { message: e.message });
      scheduleQuickRepairResume_(QUICK_REPAIR_RESUME_AFTER_MS);
      return { ok: false, transient: true, message: e.message };
    }
    // 予期しないエラー: 1分後にリトライ
    console.error(`[quickRepair] unexpected error: ${e.message}`);
    writeProcessLog_("ギャップ修復", "予期しないエラーが発生しました。再開トリガーを設定したうえで、GAS側にも失敗として記録します。", { message: e.message });
    scheduleQuickRepairResume_(QUICK_REPAIR_RESUME_AFTER_MS);
    throw e;
  } finally {
    if (lockAcquired && lock) {
      try { lock.releaseLock(); } catch (releaseErr) {}
    }
  }
}

/**
 * 再開用トリガーハンドラ
 */
function resumeQuickRepair() {
  deleteTriggersByHandler_("resumeQuickRepair");
  try {
    return quickRepairRecentGaps();
  } catch (e) {
    if (isTransientError_(e)) {
      writeProcessLog_("ギャップ修復", "再開トリガーで一時的なエラーが発生したため、1分後に再試行します。", {
        message: e.message
      });
      scheduleQuickRepairResume_(QUICK_REPAIR_RESUME_AFTER_MS);
      return { ok: false, transient: true, retried: true, message: e.message };
    }
    throw e;
  }
}

/**
 * GAP修復の監査用。
 * quickScanMissingSessions は「未処理タスク」だけを見るため、GAP_FAILED/GAP_REPAIR
 * マーカーが付いた未充足日を意図的に除外する。この関数はマーカー付きの未充足も数える。
 */
function auditGapRepairCoverage(daysBack, minSessions) {
  daysBack = daysBack || 14;
  minSessions = minSessions || 2;

  const ss = SpreadsheetApp.openById(
    PropertiesService.getScriptProperties().getProperty("SPREADSHEET_ID")
  );
  const sh = ss.getSheetByName(CONFIG.OHLCV_SHEET);
  if (!sh) throw new Error("ohlcv_4h シートが見つかりません");

  const now = getOhlcvBusinessDate_();
  const cutoff = new Date(now.getTime() - daysBack * 24 * 60 * 60 * 1000);
  const cutoffDateKey = Utilities.formatDate(cutoff, CONFIG.TIMEZONE, "yyyy/MM/dd");
  const nowDateKey = Utilities.formatDate(now, CONFIG.TIMEZONE, "yyyy/MM/dd");
  const holidaySet = getMarketHolidaySet_();
  const lastRow = sh.getLastRow();
  if (lastRow < 2) return { ok: true, daysBack: daysBack, untreatedShort: 0, attemptedButShort: 0 };

  const targetSymbols = getQuickRepairTargetSymbols_(ss);
  const targetSymbolSet = {};
  targetSymbols.forEach(sym => targetSymbolSet[sym] = true);
  const values = readRecentOhlcvRowsByTimestamp_(sh, cutoffDateKey, nowDateKey, 3);
  const periodMap = {};
  const markerMap = {};
  const failedMarkerMap = {};
  const repairMarkerMap = {};

  for (const row of values) {
    const ts = row[0];
    const alertId = String(row[1] || "").trim();
    const sym = String(row[2] || "").trim();
    if (!(ts instanceof Date) || !sym || !targetSymbolSet[sym]) continue;

    const dateStr = Utilities.formatDate(ts, CONFIG.TIMEZONE, "yyyy/MM/dd");
    if (dateStr < cutoffDateKey || dateStr > nowDateKey) continue;

    const key = sym + "|" + dateStr;
    if (alertId === "GAP_FAILED") {
      markerMap[key] = true;
      failedMarkerMap[key] = true;
      continue;
    }
    if (alertId === "GAP_REPAIR") {
      markerMap[key] = true;
      repairMarkerMap[key] = true;
    }
    addSessionPeriod_(periodMap, key, getOhlcvSessionName_(ts));
  }

  const summary = {
    ok: true,
    daysBack: daysBack,
    minSessions: minSessions,
    symbols: targetSymbols.length,
    scannedRecentRows: values.length,
    untreatedShort: 0,
    attemptedButShort: 0,
    failedMarkedShort: 0,
    repairMarkedShort: 0,
    filledDays: 0,
    samplesUntreated: [],
    samplesAttemptedShort: []
  };

  const symbols = targetSymbols;
  for (let d = new Date(cutoff.getTime()); d <= now; d = new Date(d.getTime() + 24 * 60 * 60 * 1000)) {
    if (isMarketHolidayJst_(d, holidaySet)) continue;
    const dateStr = Utilities.formatDate(d, CONFIG.TIMEZONE, "yyyy/MM/dd");
    symbols.forEach(sym => {
      const key = sym + "|" + dateStr;
      const cnt = countSessionPeriods_(periodMap, key);
      if (cnt >= minSessions) {
        summary.filledDays++;
        return;
      }
      const sample = { symbol: sym, dateStr: dateStr, sessions: cnt };
      if (markerMap[key]) {
        summary.attemptedButShort++;
        if (failedMarkerMap[key]) summary.failedMarkedShort++;
        if (repairMarkerMap[key]) summary.repairMarkedShort++;
        if (summary.samplesAttemptedShort.length < 20) summary.samplesAttemptedShort.push(sample);
      } else {
        summary.untreatedShort++;
        if (summary.samplesUntreated.length < 20) summary.samplesUntreated.push(sample);
      }
    });
  }

  console.log(`[auditGapRepair] ${JSON.stringify(summary)}`);
  return summary;
}

function auditEvaluationOhlcvCoverage(options) {
  const opts = options || {};
  const daysBack = opts.daysBack || OHLCV_DEFAULT_LOOKBACK_DAYS;
  const minSessions = opts.minSessions || 2;
  const scan = quickScanMissingSessions(daysBack, minSessions);
  const audit = auditGapRepairCoverage(daysBack, minSessions);

  const summary = Object.assign({}, audit, {
    pendingSymbols: scan.count,
    pendingTasks: pendingTasks.length,
    pendingSamples: pendingTasks.slice(0, 20)
  });
  console.log(`[auditEvaluationOhlcvCoverage] ${JSON.stringify(summary)}`);
  return summary;
}

function repairEvaluationOhlcvCoverageInternal_(options) {
  const opts = options || {};
  const daysBack = opts.daysBack || OHLCV_DEFAULT_LOOKBACK_DAYS;
  const minSessions = opts.minSessions || 2;
  const isResume = opts.resume === true;
  const props = PropertiesService.getScriptProperties();
  const startTime = Date.now();
  const guardMs = 30 * 1000;

  let state = isResume ? readEvaluationOhlcvCoverageState_(props) : null;
  if (!state || state.version !== 1 || state.daysBack !== daysBack || state.minSessions !== minSessions) {
    state = makeEvaluationOhlcvCoverageState_(daysBack, minSessions);
  }
  state.runs++;
  state.updatedAt = new Date().toISOString();
  props.setProperty(EVAL_OHLCV_COVERAGE_STATE_KEY, JSON.stringify(state));
  scheduleEvaluationOhlcvCoverageResume_(EVAL_OHLCV_COVERAGE_SAFETY_AFTER_MS);

  try {
    const ss = SpreadsheetApp.openById(
      PropertiesService.getScriptProperties().getProperty("SPREADSHEET_ID")
    );
    const sh = ss.getSheetByName(CONFIG.OHLCV_SHEET);
    if (!sh) throw new Error("ohlcv_4h シートが見つかりません");

    while (Date.now() - startTime < EVAL_OHLCV_COVERAGE_MAX_RUN_MS - guardMs) {
      const scan = quickScanMissingSessions(daysBack, minSessions);
      const tasks = flattenMissingTasks_(scan.details);
      state.lastScanSymbols = scan.count;
      state.lastScanTasks = tasks.length;
      state.updatedAt = new Date().toISOString();
      props.setProperty(EVAL_OHLCV_COVERAGE_STATE_KEY, JSON.stringify(state));

      if (tasks.length === 0) {
        dedupeAndSortOhlcv_(sh);
        SpreadsheetApp.flush();
        props.deleteProperty(EVAL_OHLCV_COVERAGE_STATE_KEY);
        deleteTriggersByHandler_(EVAL_OHLCV_COVERAGE_RESUME_HANDLER);
        const complete = buildEvaluationOhlcvCoverageSummary_(state, true, { remainingTasks: 0 });
        console.log(`[repairEvaluationOhlcvCoverage] 完了: ${JSON.stringify(complete)}`);
        return complete;
      }

      const groups = groupQuickRepairTasksBySymbol_(tasks);
      const batchGroups = groups.slice(0, EVAL_OHLCV_COVERAGE_BATCH_GROUPS);
      const batchTasks = flattenQuickRepairGroupsToTasks_(batchGroups);
      const sessionInfo = buildRecentSessionInfoForTasks_(sh, batchTasks, daysBack);
      const result = processQuickRepairGroups_(sh, batchGroups, sessionInfo, minSessions);
      SpreadsheetApp.flush();

      state.processedGroups += batchGroups.length;
      state.processedTasks += countQuickRepairGroupTasks_(batchGroups);
      state.repairedDates += result.repairedDates || 0;
      state.failedDates += result.failedDates || 0;
      state.deferredDates += result.deferredDates || 0;
      state.rows += result.rows || 0;
      state.parseErrors += result.parseErrors || 0;
      state.batches++;
      state.lastResult = result;
      state.updatedAt = new Date().toISOString();
      props.setProperty(EVAL_OHLCV_COVERAGE_STATE_KEY, JSON.stringify(state));

      console.log(`[repairEvaluationOhlcvCoverage] batch=${state.batches} groups=${batchGroups.length} result=${JSON.stringify(result)}`);
      if ((result.deferredDates || 0) > 0) {
        scheduleEvaluationOhlcvCoverageResume_(EVAL_OHLCV_COVERAGE_RESUME_AFTER_MS);
        return buildEvaluationOhlcvCoverageSummary_(state, false, { reason: "deferred_dates" });
      }
      Utilities.sleep(200);
    }

    scheduleEvaluationOhlcvCoverageResume_(EVAL_OHLCV_COVERAGE_RESUME_AFTER_MS);
    return buildEvaluationOhlcvCoverageSummary_(state, false, { reason: "time_limit" });
  } catch (e) {
    scheduleEvaluationOhlcvCoverageResume_(EVAL_OHLCV_COVERAGE_RESUME_AFTER_MS);
    if (isTransientError_(e)) {
      return buildEvaluationOhlcvCoverageSummary_(state, false, { transient: true, message: e.message });
    }
    throw e;
  }
}

function makeEvaluationOhlcvCoverageState_(daysBack, minSessions) {
  return {
    version: 1,
    daysBack: daysBack,
    minSessions: minSessions,
    runs: 0,
    processedGroups: 0,
    processedTasks: 0,
    repairedDates: 0,
    failedDates: 0,
    deferredDates: 0,
    rows: 0,
    parseErrors: 0,
    batches: 0,
    lastScanSymbols: 0,
    lastScanTasks: 0,
    startedAt: new Date().toISOString(),
    updatedAt: new Date().toISOString()
  };
}

function readEvaluationOhlcvCoverageState_(props) {
  try {
    const raw = props.getProperty(EVAL_OHLCV_COVERAGE_STATE_KEY);
    return raw ? JSON.parse(raw) : null;
  } catch (e) {
    return null;
  }
}

function buildEvaluationOhlcvCoverageSummary_(state, complete, extra) {
  return Object.assign({
    ok: true,
    complete: complete,
    daysBack: state.daysBack,
    minSessions: state.minSessions,
    runs: state.runs,
    processedGroups: state.processedGroups,
    processedTasks: state.processedTasks,
    repairedDates: state.repairedDates,
    failedDates: state.failedDates,
    deferredDates: state.deferredDates,
    rows: state.rows,
    parseErrors: state.parseErrors,
    batches: state.batches,
    lastScanSymbols: state.lastScanSymbols,
    lastScanTasks: state.lastScanTasks
  }, extra || {});
}

function scheduleEvaluationOhlcvCoverageResume_(delayMs) {
  deleteTriggersByHandler_(EVAL_OHLCV_COVERAGE_RESUME_HANDLER);
  ScriptApp.newTrigger(EVAL_OHLCV_COVERAGE_RESUME_HANDLER)
    .timeBased()
    .after(delayMs)
    .create();
}

function flattenQuickRepairGroupsToTasks_(groups) {
  const tasks = [];
  (groups || []).forEach(group => {
    (group.dates || []).forEach(dateStr => tasks.push({ symbol: group.symbol, dateStr: dateStr }));
  });
  return tasks;
}

function buildRecentSessionInfoForTasks_(sheet, tasks, daysBack, options) {
  const opts = options || {};
  const info = { countMap: {}, periodMap: {}, markerMap: {} };
  if (!tasks || tasks.length === 0) return info;

  const targetSymbols = {};
  const targetDayToDateStr = {};
  tasks.forEach(t => {
    targetSymbols[t.symbol] = true;
    const dayKey = toJstDayKeyFromDateStr_(t.dateStr);
    if (dayKey !== null) targetDayToDateStr[dayKey] = t.dateStr;
  });

  const now = getOhlcvBusinessDate_();
  const cutoff = new Date(now.getTime() - daysBack * 24 * 60 * 60 * 1000);
  const cutoffDateKey = Utilities.formatDate(cutoff, CONFIG.TIMEZONE, "yyyy/MM/dd");
  const nowDateKey = Utilities.formatDate(now, CONFIG.TIMEZONE, "yyyy/MM/dd");
  const values = readRecentOhlcvRowsByTimestamp_(sheet, cutoffDateKey, nowDateKey, 3, {
    deadlineMs: opts.deadlineMs || 0,
    maxRows: opts.maxRows || getQuickRepairRecentScanMaxRows_(Object.keys(targetSymbols).length, daysBack)
  });
  const scanStats = values.scanStats || {};
  if (scanStats.complete === false) {
    info.incomplete = true;
    info.scanStats = scanStats;
    return info;
  }

  for (const row of values) {
    const ts = row[0];
    const alertId = String(row[1] || "").trim();
    const sym = String(row[2] || "").trim();
    if (!(ts instanceof Date) || !sym || !targetSymbols[sym]) continue;

    const rowDateStr = targetDayToDateStr[toJstDayKeyFromDate_(ts)];
    if (!rowDateStr) continue;
    const key = sym + "|" + rowDateStr;

    if (alertId === "GAP_FAILED") {
      info.markerMap[key] = true;
      continue;
    }
    if (alertId === "GAP_REPAIR") info.markerMap[key] = true;

    addSessionPeriod_(info.periodMap, key, getOhlcvSessionName_(ts));
    info.countMap[key] = countSessionPeriods_(info.periodMap, key);
  }
  return info;
}

/**
 * 既存の不良 GAP_REPAIR 行（V=0 かつ O=H=L=C）を一括削除する one-shot 関数。
 * 一回だけ手動実行する想定。
 *
 * 削除条件: B=GAP_REPAIR AND V=0 AND O===H===L===C
 * → これに該当しない正常な補填行（V>0 や OHLC バラつきあり）は保持される
 */
function purgeBogusGapRepairRows() {
  const ss = SpreadsheetApp.openById(
    PropertiesService.getScriptProperties().getProperty("SPREADSHEET_ID")
  );
  const sh = ss.getSheetByName(CONFIG.OHLCV_SHEET);
  if (!sh) { console.log("ohlcv_4h が見つかりません"); return; }
  const lastRow = sh.getLastRow();
  if (lastRow < 2) { console.log("データなし"); return; }
  const values = sh.getRange(2, 1, lastRow - 1, 8).getValues();
  const deleteRows = [];
  values.forEach((r, i) => {
    if (String(r[1]) !== "GAP_REPAIR") return;
    const o = Number(r[3]), h = Number(r[4]), l = Number(r[5]), c = Number(r[6]);
    const v = Number(r[7]);
    const vIsZero = (v === 0 || !Number.isFinite(v));
    const ohlcSame = Number.isFinite(o) && o === h && h === l && l === c;
    if (vIsZero && ohlcSame) deleteRows.push(i + 2);
  });
  if (deleteRows.length === 0) { console.log("[purgeBogus] 削除対象なし"); return { ok: true, deleted: 0 }; }
  deleteRowsByRowNums_(sh, deleteRows);
  console.log(`[purgeBogus] 削除: ${deleteRows.length} 行`);
  return { ok: true, deleted: deleteRows.length };
}

/**
 * 旧ロジック由来の残骸を安全に整理する one-shot（再開可能バッチ版）。
 * - 空timestamp行を削除
 * - 09:00 / 13:00 以外のOHLCV timestamp行を削除
 * - 長期滞留の GAP_FAILED を削除
 *
 * 使い方:
 *   cleanupLegacyGapFailedAndEmptyTimestamps(true)   // DryRun（書き込みなし）
 *   cleanupLegacyGapFailedAndEmptyTimestamps(false)  // 本番削除
 */
function cleanupLegacyGapFailedAndEmptyTimestamps(dryRun) {
  const STATE_KEY = "CLEANUP_LEGACY_STATE_V1";
  const RESUME_HANDLER = "resumeCleanupLegacyGapFailedAndEmptyTimestamps";
  const MAX_RUN_MS = 3.5 * 60 * 1000;
  const BATCH_SIZE = 8000;
  const RESUME_DELAY_MS = 60 * 1000;
  const FORCE_PURGE_DAYS = 90; // 長期滞留GAP_FAILEDのみ対象

  const isDryRun = dryRun !== false;
  const props = PropertiesService.getScriptProperties();
  const nowMs = Date.now();
  const forcePurgeMs = FORCE_PURGE_DAYS * 24 * 60 * 60 * 1000;
  const startTime = Date.now();

  let state = null;
  try {
    const raw = props.getProperty(STATE_KEY);
    state = raw ? JSON.parse(raw) : null;
  } catch (e) {
    state = null;
  }

  if (!state || state.dryRun !== isDryRun) {
    state = {
      dryRun: isDryRun,
      nextRow: 2,
      scannedRows: 0,
      emptyTimestamp: 0,
      invalidSessionTimestamp: 0,
      staleGapFailed: 0,
      deletedRows: 0,
      chunks: 0
    };
    props.setProperty(STATE_KEY, JSON.stringify(state));
  }

  const ss = SpreadsheetApp.openById(
    PropertiesService.getScriptProperties().getProperty("SPREADSHEET_ID")
  );
  const sh = ss.getSheetByName(CONFIG.OHLCV_SHEET);
  if (!sh) { console.log("ohlcv_4h が見つかりません"); return { ok: false, message: "sheet not found" }; }

  while (Date.now() - startTime < MAX_RUN_MS) {
    const lastRow = sh.getLastRow();
    if (state.nextRow > lastRow) {
      deleteTriggersByHandler_(RESUME_HANDLER);
      props.deleteProperty(STATE_KEY);
      const summary = {
        ok: true,
        complete: true,
        dryRun: isDryRun,
        scannedRows: state.scannedRows,
        emptyTimestamp: state.emptyTimestamp,
        invalidSessionTimestamp: state.invalidSessionTimestamp || 0,
        staleGapFailed: state.staleGapFailed,
        deletedRows: state.deletedRows,
        chunks: state.chunks
      };
      console.log(`[cleanupLegacy] 完了: ${JSON.stringify(summary)}`);
      return summary;
    }

    const numRows = Math.min(BATCH_SIZE, lastRow - state.nextRow + 1);
    const values = sh.getRange(state.nextRow, 1, numRows, 3).getValues();
    const deleteRows = [];
    const repairSymbolSet = {};
    let chunkEmpty = 0;
    let chunkInvalidSession = 0;
    let chunkStale = 0;

    for (let i = 0; i < values.length; i++) {
      const rowNum = state.nextRow + i;
      const ts = values[i][0];
      const alertId = String(values[i][1] || "").trim();
      const sym = String(values[i][2] || "").trim();

      const isEmptyTs = !(ts instanceof Date) || isNaN(ts.getTime());
      if (isEmptyTs) {
        deleteRows.push(rowNum);
        if (sym) repairSymbolSet[sym] = true;
        chunkEmpty++;
        continue;
      }

      if (!isOhlcvSessionTimestamp_(ts)) {
        deleteRows.push(rowNum);
        if (sym) repairSymbolSet[sym] = true;
        chunkInvalidSession++;
        continue;
      }

      if (alertId !== "GAP_FAILED") continue;
      const ageMs = nowMs - ts.getTime();
      if (ageMs >= forcePurgeMs) {
        deleteRows.push(rowNum);
        if (sym) repairSymbolSet[sym] = true;
        chunkStale++;
      }
    }

    if (!isDryRun && deleteRows.length > 0) {
      deleteRowsByRowNums_(sh, deleteRows);
      const repairSymbols = Object.keys(repairSymbolSet).sort();
      if (repairSymbols.length > 0) setOhlcvRepairSymbols_(repairSymbols);
      state.deletedRows += deleteRows.length;
      state.nextRow = state.nextRow + numRows - deleteRows.length;
    } else {
      state.nextRow += numRows;
    }

    state.scannedRows += numRows;
    state.emptyTimestamp += chunkEmpty;
    state.invalidSessionTimestamp = (state.invalidSessionTimestamp || 0) + chunkInvalidSession;
    state.staleGapFailed += chunkStale;
    state.chunks += 1;
    props.setProperty(STATE_KEY, JSON.stringify(state));

    console.log(`[cleanupLegacy] chunk=${state.chunks} scanned=${state.scannedRows} deleteCandidates=${chunkEmpty + chunkInvalidSession + chunkStale} invalidSession=${chunkInvalidSession} nextRow=${state.nextRow}`);
  }

  deleteTriggersByHandler_(RESUME_HANDLER);
  ScriptApp.newTrigger(RESUME_HANDLER).timeBased().after(RESUME_DELAY_MS).create();
  props.setProperty(STATE_KEY, JSON.stringify(state));
  console.log(`[cleanupLegacy] 中断: scanned=${state.scannedRows} nextRow=${state.nextRow} 1分後に再開`);
  return { ok: true, incomplete: true, scannedRows: state.scannedRows, nextRow: state.nextRow };
}

function resumeCleanupLegacyGapFailedAndEmptyTimestamps() {
  deleteTriggersByHandler_("resumeCleanupLegacyGapFailedAndEmptyTimestamps");
  cleanupLegacyGapFailedAndEmptyTimestamps();
}
